## wrap_long_lines_with_trailing_commas, in Roc on Shale's platform (roc_platform/README.md): long
## Roc lines shortened, and with `tokens=N` groups laid out by the stacking
## rule, by trailing commas that `roc fmt` reads. Its launcher,
## cookbook/roc/wrap_long_lines_with_trailing_commas.shale, is how it is run, and its header says
## the rule, the rewrite and the list it prints. check_goldens.shale, beside
## this, holds its output and the files it rewrites to its goldens.
##
##   cd <repo> && shale cookbook:roc/wrap_long_lines_with_trailing_commas.shale dir=roc width=100
##   cd <repo> && shale cookbook:roc/wrap_long_lines_with_trailing_commas.shale dir=app,platform tokens=6
##   shale cookbook:roc/wrap_long_lines_with_trailing_commas.shale self_test=1
##
## The arguments: `dir` (`roc`; several, comma-separated), `width` (100),
## `tokens` (0, wide lines alone), `write` (`write=0` reports and writes
## nothing), `self_test` and `work_base`, the folder the work copies go in,
## which the launcher gives.
##
## This root reads the arguments, rewrites each folder (Rewrite) and prints
## the lines left wide (WideLines); SelfTest is the self-test. Groups,
## Stacking, Packed and Bytes are the cookbook's Roc library,
## cookbook/roc/lib/, the one home of the stacking rule.
## budget: steps=1000000000 read_bytes=2000000000 arena_bytes=2000000000 output_bytes=4000000 changes=5000 roc_cpu_ms=600000
app [main!] { lib: "../lib/main.roc", pf: platform "../../../roc_platform/platform/main.roc" }

import pf.Shale
import lib.Stacking
import lib.Text
import Rewrite
import SelfTest
import WideLines

main! : {} => {}
main! = |{}| {
	self_test = Shale.arg!("self_test", "") == "1"
	work_base = Shale.arg!("work_base", "scratch:")
	dirs_given = Shale.arg!("dir", "roc")
	width = integer_of!("width", Shale.arg!("width", "100"))
	tokens = if self_test {
		Stacking.stack_tokens
	} else {
		integer_of!("tokens", Shale.arg!("tokens", "0"))
	}
	write_given = Shale.arg!("write", "1")
	if write_given != "0" and write_given != "1" {
		Shale.fail!("wrap_long_lines_with_trailing_commas: write=${write_given}: write must be 1 (the default) or 0")
	}
	write = self_test or write_given == "1"
	roc = Rewrite.roc_bin!(self_test)
	settings = { rule: { tokens, width }, roc, work_base }
	if self_test {
		SelfTest.run!(settings)
	} else {
		var $report = WideLines.fresh
		for dir in dirs_of!(dirs_given) {
			work = Rewrite.work_of!(dir, { self_test: Bool.False, work_base })
			done = match Rewrite.rewrite_in_work!(dir, { work, settings }) {
				Ok(finished) => finished
				Err(StillChanging(touched)) => {
					Shale.fail!(Rewrite.still_changing(touched))
					{ work, paths: [] }
				}
			}
			at = if write {
				Rewrite.write_back!(dir, done)
				dir
			} else {
				done.work
			}
			$report = WideLines.report_wide_lines!($report, { dir, at, paths: done.paths, width })
			Shale.remove_tree!(done.work)
		}
		remove_work_base!(work_base)
		report = $report
		said = if write {
			""
		} else {
			"report, nothing written: "
		}
		print_findings!("${said}${WideLines.summary(report, width)}", report.long)
		fail_if_refused!(report.refused)
	}
	if self_test {
		remove_work_base!(work_base)
	}
}

## The whole number an argument gives; another value stops the run, as
## Shale's `arg` does for an integer default. (Each `Shale.arg!` names its
## argument as a literal, so the launcher's check of the values finds it.)
integer_of! : Str, Str => U64
integer_of! = |name, given|
	match I64.from_str(given) {
		Ok(value) if value >= 0 => value.to_u64_wrap()
		Ok(_) => {
			Shale.fail!("wrap_long_lines_with_trailing_commas: ${name}=${given} is below 0")
			0
		}
		Err(_) => {
			Shale.fail!("`arg`: ${name}=${given} is not an integer, and its default is one")
			0
		}
	}

## The folders `dir=` names, each held to be a path and not an option.
dirs_of! : Str => List(Str)
dirs_of! = |given| {
	var $dirs = []
	for piece in given.split_on(",") {
		dir = Text.trim(piece)
		if dir.starts_with("-") {
			Shale.fail!("`wrap_long_lines_with_trailing_commas`'s `dir` is \"${dir}\": a value that starts with `-` would be one of the program's options")
		}
		if dir != "" {
			$dirs = $dirs.append(dir)
		}
	}
	if $dirs.is_empty() {
		Shale.fail!("wrap_long_lines_with_trailing_commas: dir= names no folder")
	}
	$dirs
}

## The work folder, when the launcher made a fresh one for want of a
## `scratch:` mount, removed once the run has finished.
remove_work_base! : Str => {}
remove_work_base! = |work_base| {
	if work_base != "scratch:" {
		Shale.remove_tree!(work_base)
	}
}

## The summary, each line listed, then the summary again
## (cookbook/findings.shale): the first copy survives a cut output, and a
## gate reads the last line.
print_findings! : Str, List(Str) => {}
print_findings! = |summary, listed| {
	Shale.print!(summary)
	if !(listed.is_empty()) {
		for line in listed {
			Shale.print!(line)
		}
		Shale.print!(summary)
	}
}

## The files Roc's parser refused, named, then a stop
## (cookbook/roc/parsed.shale's `fail_if_refused`); nothing when it refused
## none.
fail_if_refused! : List(Str) => {}
fail_if_refused! = |refused| {
	if !(refused.is_empty()) {
		for refusal in refused {
			Shale.print!("refused: ${refusal}")
		}
		refusals = "roc-ast refused ${refused.len().to_str()} files (named above)"
		why = "which have no tree to judge; every other file was judged"
		Shale.fail!("wrap_long_lines_with_trailing_commas: ${refusals}, ${why}")
	}
}

## The rewrite of a folder: its .roc files copied to a work copy, then pass
## after pass, each a trailing comma given to each line's group that should
## stack and taken from each stacked group that should sit on one line, with
## `roc fmt` run on the work copy after each, until a pass changes nothing.
## The changed files are written back only once the rewrite has finished, so
## a rewrite that never settles leaves the folder as it was. The rule and its
## pieces are the cookbook's Roc library's Stacking; the Roc is the tree's
## own (cookbook/roc/pinned.shale), run by its OS path.
import pf.Shale
import lib.Bytes
import lib.Stacking
import lib.Text

Rewrite := [].{

	## What a rewrite runs by: the stacking rule (`tokens` 0 wraps wide lines
	## alone), the Roc's OS path, and the folder the work copies go in.
	Settings : { rule : Stacking.Rule, roc : Str, work_base : Str }

	## A finished rewrite: the work copy that holds it, and the folder's .roc
	## files, each as its path inside the folder.
	Done : { work : Str, paths : List(Str) }

	## The most passes a rewrite makes.
	max_passes : U64
	max_passes = 8

	## The work copy of folder `dir`: the self-test's is a name of its own,
	## any other's is named for the folder's OS path.
	work_of! : Str, { self_test : Bool, work_base : Str } => Str
	work_of! = |dir, given|
		if given.self_test {
			Text.path_under(given.work_base, "roc_commas_work/self_test")
		} else {
			named = Shale.re_replace!(Shale.real!(dir), "[^A-Za-z0-9]", "_")
			Text.path_under(given.work_base, "roc_commas_work/${named}")
		}

	## The .roc files of folder `dir`, each as its path inside it.
	roc_paths_in_dir! : Str => List(Str)
	roc_paths_in_dir! = |dir| {
		cut = dir.count_utf8_bytes() + 1
		Shale.glob!("${dir}/**/*.roc").map(|path| Text.bytes_of(path, { start: cut, count: path.count_utf8_bytes() }))
	}

	## Rewrites the work copy `work` of folder `dir` until a pass changes
	## nothing, printing each pass. A rewrite still changing after
	## `max_passes` passes is `StillChanging`, with the files the last pass
	## changed, named as the folder's.
	rewrite_in_work! : Str, { work : Str, settings : Rewrite.Settings } => Try(Rewrite.Done, [StillChanging(List(Str))])
	rewrite_in_work! = |dir, at| {
		work = at.work
		rule = at.settings.rule
		paths = Rewrite.roc_paths_in_dir!(dir)
		fresh_work_copy!(dir, { work, paths })
		run_roc_fmt!(work, at.settings.roc)
		$pass : U64
		var $pass = 1
		$changed : U64
		var $changed = 1
		var $touched = []
		# A file's text when a pass last left it alone. A later pass would
		# leave the same text alone again, so it is not read for groups twice.
		var $settled = Dict.empty()
		while $changed > 0 and $pass <= Rewrite.max_passes {
			$changed = 0
			$touched = []
			for path in paths {
				text = Rewrite.read_text!("${work}/${path}")
				if $settled.get(path) != Ok(text) {
					n = rewrite_file!("${work}/${path}", { text, rule })
					if n > 0 {
						$changed = $changed + n
						$touched = $touched.append("${dir}/${path}")
					} else {
						$settled = $settled.insert(path, text)
					}
				}
			}
			Shale.print!("${dir}: pass ${$pass.to_str()}: ${$changed.to_str()} trailing commas added or taken out")
			run_roc_fmt!(work, at.settings.roc)
			$pass = $pass + 1
		}
		if $changed > 0 {
			Err(StillChanging($touched))
		} else {
			Ok({ work, paths })
		}
	}

	## The stop a rewrite still changing makes, naming the files it changed.
	still_changing : List(Str) -> Str
	still_changing = |touched|
		"wrap_long_lines_with_trailing_commas: still changing after ${Rewrite.max_passes.to_str()} passes, so no file was written: ${Str.join_with(touched, ", ")}"

	## Writes each file the rewrite changed back to folder `dir`, and says
	## how many.
	write_back! : Str, Rewrite.Done => {}
	write_back! = |dir, done| {
		var $changed = []
		for path in done.paths {
			text = Rewrite.read_text!("${done.work}/${path}")
			if text != Rewrite.read_text!("${dir}/${path}") {
				$changed = $changed.append({ path: "${dir}/${path}", text })
			}
		}
		for change in $changed {
			Shale.write!(change.path, change.text)
		}
		Shale.print!("${dir}: ${$changed.len().to_str()} files rewritten")
	}

	## A file's text; a file gone since the glob stops the run.
	read_text! : Str => Str
	read_text! = |path|
		match Shale.read!(path) {
			Found(text) => text
			Missing => {
				Shale.fail!("wrap_long_lines_with_trailing_commas: no file ${path}")
				""
			}
		}

	## Whether a line is a comment's, which no trailing comma shortens.
	is_comment : Str -> Bool
	is_comment = |text| Text.trim(text).starts_with("#")

	## The OS path of the Roc to rewrite with: in a self-test the pin every
	## recipe's self-test runs (pinned.shale's `SELF_TEST_PIN`, read there),
	## else the one the `roc-version` where the run starts names.
	roc_bin! : Bool => Str
	roc_bin! = |self_test| {
		pin = if self_test {
			self_test_pin!({})
		} else {
			if !(Shale.exists!("roc-version")) {
				Shale.fail!("wrap_long_lines_with_trailing_commas: no roc-version where the run starts: run it from the repository's root, whose roc-version pins its Roc")
			}
			Text.trim(Rewrite.read_text!("roc-version"))
		}
		roc = "tools:roc-${pin}/roc"
		if !(Shale.exists!(roc)) {
			Shale.fail!("wrap_long_lines_with_trailing_commas: no ${roc}: install Roc ${pin} into ~/tools/roc-${pin}/ (roc-apps's toolchain release carries it)")
		}
		real = Shale.real!(roc)
		said = Shale.run!({ argv: [real, "version"], keep: Whole(4000), timeout_ms: 30000, want: "" })
		if Text.trim(said.out) != "Roc compiler version ${pin}" {
			Shale.fail!("wrap_long_lines_with_trailing_commas: ${roc} says ${Text.trim(said.out)}, not Roc compiler version ${pin}")
		}
		real
	}
}

## The work copy made afresh: the folder's .roc files copied into it.
fresh_work_copy! : Str, { work : Str, paths : List(Str) } => {}
fresh_work_copy! = |dir, copy| {
	if Shale.exists!(copy.work) {
		Shale.remove_tree!(copy.work)
	}
	Shale.mkdir!(copy.work)
	for path in copy.paths {
		Shale.write!("${copy.work}/${path}", Rewrite.read_text!("${dir}/${path}"))
	}
}

## `roc fmt` run on the work copy; a file that does not parse stops the run,
## after its output.
run_roc_fmt! : Str, Str => {}
run_roc_fmt! = |work, roc| {
	result = Shale.run!({
		argv: [roc, "fmt", Shale.real!(work)],
		keep: Tail(20),
		timeout_ms: 60000,
		want: "",
	})
	if result.code != 0 {
		Shale.print!(result.out)
		Shale.fail!("roc fmt failed")
	}
}

## One pass over one file, whose text is `text`: the groups that should
## unstack lose their trailing commas, then each code line gets one for its
## group that should stack, if any; the file is written when any changed,
## and the answer is how many did.
rewrite_file! : Str, { text : Str, rule : Stacking.Rule } => U64
rewrite_file! = |path, given| {
	rule = given.rule
	source_lines = given.text.split_on("\n")
	unstacked = if rule.tokens > 0 {
		match Stacking.groups_to_unstack(source_lines, rule) {
			Ok(found) => found
			Err(NoComma(line)) => {
				Shale.fail!("groups: line ${line.to_str()} does not end in its group's trailing comma")
				[]
			}
		}
	} else {
		[]
	}
	unstacked_lines = Stacking.without_trailing_commas(source_lines, unstacked)
	reachable = Stacking.without_targets_section(unstacked_lines)
	var $lines = unstacked_lines
	$changed : U64
	var $changed = unstacked.len()
	$index : U64
	var $index = 0
	for text in unstacked_lines {
		if !(Rewrite.is_comment(text)) {
			match line_with_comma(reachable.get($index) ?? "", rule) {
				Ok(with_comma) => {
					$lines = $lines.set($index, with_comma) ?? crash "wrap_long_lines_with_trailing_commas: a line past the file's end"
					$changed = $changed + 1
				}
				Err(_) => {}
			}
		}
		$index = $index + 1
	}
	if $changed > 0 {
		Shale.write!(path, Str.join_with($lines, "\n"))
	}
	$changed
}

## The line with a trailing comma for its outermost group that should
## stack, or else, when it is wider than the width, for the group that
## stacking brings within it. A line of data gets none.
line_with_comma : Str, Stacking.Rule -> Try(Str, [NoComma])
line_with_comma = |text, rule| {
	few_commas = rule.tokens == 0 or commas_before_the_last(text) < Stacking.stack_items - 1
	if Bytes.line_width(text) <= rule.width and few_commas {
		Err(NoComma)
	} else {
		match Stacking.group_to_stack(text, rule) {
			Ok(group) => Ok(Stacking.with_trailing_comma({ line: text, group }))
			Err(_) => Err(NoComma)
		}
	}
}

## How many commas a line holds, not counting one that ends it. A group of
## `stack_items` items that opens and closes on the line holds one fewer at
## least, so a line that fits with fewer has no group to stack. Counting is
## far cheaper than reading the line's groups, and most of a data file's
## lines are like `{ x: 0.5, y: 0.25 },`.
commas_before_the_last : Str -> U64
commas_before_the_last = |text| {
	bytes = text.to_utf8()
	commas = bytes.count_if(|byte| byte == ',')
	if Bytes.trimmed_end(bytes).last() == Ok(',') {
		commas - 1
	} else {
		commas
	}
}

## pinned.shale's `SELF_TEST_PIN`, the one home of the self-tests' Roc.
self_test_pin! : {} => Str
self_test_pin! = |{}| {
	pinned = "cookbook:roc/pinned.shale"
	groups = match Shale.re_find!(
		Rewrite.read_text!(pinned),
		"\nconst SELF_TEST_PIN = \"([^\"]+)\"",
	) {
		Found(found) => found.groups
		NoMatch => []
	}
	match groups {
		[pin, ..] => pin
		[] => {
			Shale.fail!("wrap_long_lines_with_trailing_commas: no SELF_TEST_PIN in ${pinned}")
			""
		}
	}
}

## The recipe's self-test, the Shale original's: the plants in
## cookbook/roc/plants/wrap_long_lines_with_trailing_commas/ are copied into the work folder and
## rewritten there with the rule's six tokens and the self-tests' Roc. The
## rewrite must end, each plant must come out as its `.want` file, no group
## may be left to stack or to unstack, and the lines left wide must be
## exactly `long_plants`' listed, with `wide_string_plants` wide strings and
## `data_plants` lines of data counted. Then `too_deep/too_deep.roc`, one
## group deeper than the passes reach, must stop the rewrite, naming its
## file, and be left as it was.
import pf.Shale
import lib.Stacking
import lib.Text
import Rewrite
import WideLines

SelfTest := [].{

	## The self-test, its copies and work copies under `settings.work_base`.
	run! : Rewrite.Settings => {}
	run! = |settings| {
		dir = Text.path_under(settings.work_base, "wrap_long_lines_with_trailing_commas")
		work = Rewrite.work_of!(dir, { self_test: Bool.True, work_base: settings.work_base })
		fresh_copies!(dir, plants)
		match Rewrite.rewrite_in_work!(dir, { work, settings }) {
			Ok(done) => {
				Rewrite.write_back!(dir, done)
				Shale.remove_tree!(done.work)
			}
			Err(StillChanging(touched)) => Shale.fail!(Rewrite.still_changing(touched))
		}
		$bad : U64
		var $bad = 0
		for plant in Shale.glob!("${plants}/*.roc") {
			$bad = $bad + plant_wrong!(plant, { dir, rule: settings.rule })
		}
		$bad = $bad + report_wrong!(dir, settings.rule.width)
		if $bad > 0 {
			Shale.fail!("${$bad.to_str()} checks failed")
		}
		if Shale.exists!(work) {
			Shale.fail!("wrap_long_lines_with_trailing_commas: a finished rewrite should remove its work copy, ${work}")
		}
		fresh_copies!(dir, "${plants}/too_deep")
		named = Rewrite.still_changing(["${dir}/too_deep.roc"])
		stopped = match Rewrite.rewrite_in_work!(dir, { work, settings }) {
			Ok(_) => Bool.False
			Err(StillChanging(touched)) => Rewrite.still_changing(touched) == named
		}
		if !(stopped) {
			Shale.fail!("wrap_long_lines_with_trailing_commas: too_deep.roc should stop with `${named.drop_prefix("wrap_long_lines_with_trailing_commas: ")}`")
		}
		copy = Rewrite.read_text!("${dir}/too_deep.roc")
		if copy != Rewrite.read_text!("${plants}/too_deep/too_deep.roc") {
			Shale.fail!("wrap_long_lines_with_trailing_commas: a rewrite that stops should leave too_deep.roc as it was")
		}
		Shale.print!("ok   too_deep.roc: stops naming its file, and is left as it was")
		Shale.remove_tree!(work)
		Shale.remove_tree!(dir)
		Shale.print!("PASS: the rewrite ends, every plant comes out as its .want, no group is left to stack or to unstack, the lines left wide are listed or counted, and a rewrite past its passes stops naming its file and changes none")
	}
}

plants = "cookbook:roc/plants/wrap_long_lines_with_trailing_commas"

## The plant lines left wide with nothing to stack.
long_plants : List(Str)
long_plants = ["function_type.roc:4"]

## How many plant lines are wide only for their strings.
wide_string_plants : U64
wide_string_plants = 1

data_plants : U64
data_plants = 2

## The start of a verdict's line.
verdict : Bool -> Str
verdict = |ok|
	if ok {
		"ok   "
	} else {
		"FAIL "
	}

## Folder `dir` made afresh with a copy of each `*.roc` in folder `from`.
fresh_copies! : Str, Str => {}
fresh_copies! = |dir, from| {
	if Shale.exists!(dir) {
		Shale.remove_tree!(dir)
	}
	Shale.mkdir!(dir)
	for plant in Shale.glob!("${from}/*.roc") {
		Shale.write!("${dir}/${Text.basename(plant)}", Rewrite.read_text!(plant))
	}
}

## One plant's rewrite held to its `.want`, and to no group left to stack or
## to unstack, a line printed for each wrong: the answer is how many were.
plant_wrong! : Str, { dir : Str, rule : Stacking.Rule } => U64
plant_wrong! = |plant, given| {
	name = Text.basename(plant)
	want = "${plants}/${name.drop_suffix(".roc")}.want"
	got = Rewrite.read_text!("${given.dir}/${name}")
	ok = got == Rewrite.read_text!(want)
	Shale.print!("${verdict(ok)}${name}: comes out as its .want")
	$bad : U64
	var $bad = if ok {
		0
	} else {
		1
	}
	got_lines = got.split_on("\n")
	$number : U64
	var $number = 0
	for text in Stacking.without_targets_section(got_lines) {
		$number = $number + 1
		left = !(Rewrite.is_comment(text)) and Stacking.group_to_stack(text, given.rule) != Err(NoGroup)
		if left {
			Shale.print!("FAIL ${name}:${$number.to_str()}: a group left to stack: ${Text.trim(text)}")
			$bad = $bad + 1
		}
	}
	unstacked = match Stacking.groups_to_unstack(got_lines, given.rule) {
		Ok(found) => found
		Err(NoComma(line)) => {
			Shale.fail!("groups: line ${line.to_str()} does not end in its group's trailing comma")
			[]
		}
	}
	for found in unstacked {
		Shale.print!("FAIL ${name}:${found.group.line.to_str()}: a group left to unstack")
		$bad = $bad + 1
	}
	$bad
}

## The lines the rewrite left wide held to `long_plants`, and the wide
## strings and data counted, a line printed for each: the answer is how many
## were wrong.
report_wrong! : Str, U64 => U64
report_wrong! = |dir, width| {
	report = WideLines.report_wide_lines!(
		WideLines.fresh,
		{ dir, at: dir, paths: Rewrite.roc_paths_in_dir!(dir), width },
	)
	cut = dir.count_utf8_bytes() + 1
	listed = report.long.map(

-- Shorten long Roc lines by letting `roc fmt` lay them out. roc fmt never
-- wraps a line, but it puts the items of a record, list, call or lambda's
-- arguments one to a line when the last item has a trailing comma. For each
-- code line wider than `width`, this script adds that comma to the widest
-- group of two or more items whose stacking brings the line within `width`:
-- the text up to its opening bracket fits, and so does its closing bracket
-- with what follows it (groups.shale's `group_to_wrap`). A group beside the
-- wide part, as a list pattern in a `match` arm whose string makes the line
-- wide, is left alone, and so are a function type's arguments, which can't
-- take the comma. It runs roc fmt, and repeats until no line changes; a
-- folder still changing after 8 passes stops the run, naming its files. The
-- passes work on a copy of the folder's .roc files under
-- `scratch:roc_commas_work/`, and the changed files are written back only
-- once the rewrite has finished, so a run that stops, on a pass too many or
-- on a budget, leaves the folder as it was. Under a mounts file with no
-- `scratch:` mount, the copies go in a fresh hidden folder at the root
-- instead (`.wrap_long_lines.XXXXXX`, named in the output), removed when the
-- run ends; a run that stops leaves it, for `git status` to show.
--
-- Then it lists each code line still too wide, as `path:line (width): why:
-- text`, `why` being `nothing to stack`: no group whose stacking would bring
-- it within `width`. A line wide only for the text of its strings, by Roc's
-- parser (as check_stacking.shale's `wide string`), and a comment line are
-- counted, not listed, as no trailing comma shortens either; so is a line
-- of numeric data, which stays packed (below). The list is
-- printed between its counts, as cookbook/findings.shale describes. It runs
-- the Roc that the repository's roc-version file names.
--
-- With `tokens=N`, the script holds the folder to roc-style.md's stacking
-- rule, whose six tokens are `tokens=6`, whatever a line's width: a group of
-- three or more items and N or more tokens gets the trailing comma, and a
-- stacked group that should sit on one line loses it, which is how `roc fmt`
-- is asked to join it. A group and its tokens are read as check_stacking.shale reads
-- them (cookbook/roc/groups.shale), so after `tokens=6` check_stacking.shale finds no
-- group left to stack or to unstack. A line's outermost group goes first, and
-- the next pass reaches the groups inside it. A function type's arguments and
-- a row of literals are left alone, as check_stacking.shale leaves them, and so is a
-- platform header's `targets:` section, which `roc fmt` lays out by itself.
--
-- The arguments:
--   dir     the folders to rewrite, `roc` by default; several may be given,
--           separated by commas, each rewritten in turn.
--   width   the widest a line may be, 100 by default.
--   tokens  the stacking rule's tokens, as above; 0 (the default) wraps
--           wide lines alone.
--   write   with `write=0`, the rewrite runs in the work copy, the list is
--           of the lines it would leave, and no folder is written: a
--           report.
--

--   cd <repo> && shale <shale>/cookbook/roc/wrap_long_lines.shale dir=app,platform tokens=6

-- How many commas line `l` holds, not counting one that ends it. A group of
-- STACK_ITEMS items that opens and closes on the line holds STACK_ITEMS - 1 of
-- them at least, and a comma that ends the line is inside no such group, so a
-- line that fits with fewer has no group to stack. Counting is far cheaper
-- than reading the line's groups, and most of a data file's lines are like
-- `{ x: 0.5, y: 0.25 },`.
function commas_before_the_last(l)
  local commas = #split(l, ",") - 1
  if ends_with(trim_end(l), ",") then commas = commas - 1 end
  return commas
end

-- Line `l` with a trailing comma for its outermost group that should stack,
-- or else, when it is wider than WIDTH, for groups.shale's `group_to_wrap`.
-- nil when it gets none. A line of data, by groups.shale's is_data_line,
-- gets none.
function line_with_comma(l)
  if line_width(l) <= WIDTH and (TOKENS == 0 or commas_before_the_last(l) < STACK_ITEMS - 1) then return nil end
  local g = group_to_stack(l, TOKENS, WIDTH)
  if g == nil then return nil end
  return with_trailing_comma(l, g)
end

-- Takes the trailing comma from each group of the file at `path`, whose text
-- is `text`, that should unstack, adds one to each code line that should get
-- one, and returns how many groups and lines it changed.
function rewrite_trailing_commas(path, text)
  local ls = split(text, "\n")
  local unstacked = {}
  if TOKENS > 0 then unstacked = groups_to_unstack(ls, TOKENS, WIDTH) end
  ls = without_trailing_commas(ls, unstacked)
  local reachable = without_targets_section(ls)
  local changed = #unstacked
  for i = 1, #ls do
    if not is_comment(ls[i]) then
      local with_comma = line_with_comma(reachable[i])
      if with_comma ~= nil then
        ls[i] = with_comma
        changed = changed + 1
      end
    end
  end
  if changed > 0 then replace(path, join(ls, "\n")) end
  return changed
end

-- Rewrites commas and runs roc fmt in `dir`'s work copy until a pass changes
-- nothing. Files still changing after MAX_PASSES stop the run, since the list
-- of long lines would otherwise judge a rewrite that never finished. Returns
-- `{work, paths}`, the work copy holding the rewrite.
function rewrite_in_work(dir)
  local work = work_of(dir)
  local paths = roc_paths_in_dir(dir)
  fresh_work_copy(dir, work, paths)
  run_roc_fmt(work)
  local pass = 1
  local changed = 1
  local touched = {}
  -- A file's text when a pass last left it alone. A later pass would leave
  -- the same text alone again, so it is not read for groups a second time.
  local settled = {}
  while changed > 0 and pass <= MAX_PASSES do
    changed = 0
    touched = {}
    for p in paths do
      local text = read(work .. "/" .. p)
      if not has_key(settled, p) or settled[p] ~= text then
        local n = rewrite_trailing_commas(work .. "/" .. p, text)
        if n > 0 then
          changed = changed + n
          push(touched, dir .. "/" .. p)
        else
          settled[p] = text
        end
      end
    end
    print(dir .. ": pass " .. pass .. ": " .. changed .. " trailing commas added or taken out")
    run_roc_fmt(work)
    pass = pass + 1
  end
  if changed > 0 then fail("wrap_long_lines: still changing after " .. MAX_PASSES .. " passes, so no file was written: " .. join(touched, ", ")) end

end
