## map_file, in Roc on Shale's platform (roc_platform/README.md): a file of
## Shale's C mapped, each top-level item one line, `first-last kind name`,
## from the splitter the C recipes share (cookbook/c/items/, a package),
## which reads the dialect c/check_discipline.shale holds. Its launcher,
## cookbook/c/map_file.shale, is how it is run, and its header says the
## arguments. check_goldens.shale, beside this, holds its output on the
## plants to its goldens, line for line.
##
##   shale cookbook:c/map_file.shale file=c/script/vm.c
##   shale cookbook:c/map_file.shale tree=c
##   shale cookbook:c/map_file.shale self_test=1
##
## The effects: `Shale.arg!` for the arguments, `Shale.glob!` for a tree's
## files, `Shale.read!` for each one's text, and `Shale.print!` and
## `Shale.fail!` for the verdict. The splitting is the package's, plain
## Roc; SelfTest, beside this, holds it to the plants.
## budget: steps=50000000 arena_bytes=1000000000 roc_cpu_ms=600000
app [main!] {
	items: "../items/main.roc",
	pf: platform "../../../roc_platform/platform/main.roc",
}

import pf.Shale
import items.Item
import items.Paths
import items.Split
import items.Words
import SelfTest

main! : {} => {}
main! = |{}| {
	if integer_or_stop!("self_test", Shale.arg!("self_test", "0")) == 1 {
		SelfTest.run!({})
	} else {
		file = Shale.arg!("file", "")
		tree = Shale.arg!("tree", "")
		if (file == "") == (tree == "") {
			Shale.fail!("map_file: give `file=<a .c or .h>` or `tree=<a folder>`, one of them")
		} else if file != "" {
			for line in map_lines(split_or_stop!(file, read_text!(file))) {
				Shale.print!(line)
			}
		} else {
			map_tree!(tree)
		}
	}
}

		Ok(value) => value
		Err(_) => {
			Shale.fail!("map_file: ${name}=${given} is not an integer")

	}
}

## The C files under the folder, sorted, any folder named `out` (c/out, the
## build's output) left out. A folder that is not there stops the run, in
## the words of the Shale original's walk: a check that can't run fails.
c_files_under! : Str => List(Str)
c_files_under! = |folder| {
	if Shale.list!(folder) == Missing {
		Shale.fail!("`walk(\"${folder}\")`: not found (FileNotFound)")
	}
	paths = Shale.glob!("${folder}/**/*.c").concat(Shale.glob!("${folder}/**/*.h"))
	paths.keep_if(|path| !(Paths.is_in_out_folder(folder, path))).sort_with(Words.byte_order)
}

## Every C file under the folder split and accounted for, in one line: the
## files, their lines and their items by kind.
map_tree! : Str => {}
map_tree! = |folder| {
	files = c_files_under!(folder)
	var $kinds = []
	$line_count : U64
	var $line_count = 0
	$item_count : U64
	var $item_count = 0
	for path in files {
		text = read_text!(path)
		items = split_or_stop!(path, text)
		match accounted(path, Split.lines(text), items) {
			Ok(count) => {
				$line_count = $line_count + count
			}
			Err(why) => Shale.fail!(why)
		}
		$kinds = $kinds.concat(items.map(|item| Item.kind_name(item.kind)))
		$item_count = $item_count + items.len()
	}
	names = distinct($kinds).sort_with(Words.byte_order)
	parts = names.map(|name| "${$kinds.count_if(|kind| kind == name).to_str()} ${name}")
	lines = "${$line_count.to_str()} lines, each in one item or between them"
	counts = "${$item_count.to_str()} items: ${Str.join_with(parts, ", ")}"
	Shale.print!("PASS c items: ${files.len().to_str()} files under ${folder}, ${lines}; ${counts}")
}

## Each line of the file in exactly one of the items, or blank, or a `//`
## comment line between them: the count of its lines, or the error naming
## the line that is not.
accounted : Str, List(List(U8)), List(Item) -> Try(U64, Str)
accounted = |path, lines, items| {
	# Each line's owner, by its number (place 0 is no line's).
	var $owners = List.repeat(Unowned, lines.len() + 1)
	for item in items {
		var $at = item.first
		while $at <= item.last {
			match $owners.get($at) {
				Ok(Owned(owner)) => return Err("${path}:${$at.to_str()}: in two items, `${owner}` and `${item.name}`")
				_ => {}
			}
			$owners = match $owners.set($at, Owned(item.name)) {
				Ok(owners) => owners
				Err(_) => crash "map_file: an item past its file's last line"
			}
			$at = $at + 1
		}
	}
	$number : U64
	var $number = 0
	for raw in lines {
		$number = $number + 1
		is_owned = $owners.get($number) != Ok(Unowned)
		if !(is_owned) and !(raw.is_empty()) and !(raw.starts_with(['/', '/'])) {
			return Err("${path}:${$number.to_str()}: in no item: ${Str.from_utf8_lossy(raw)}")
		}
	}
	Ok($number)
}

## Each text once, in the order first seen.
distinct : List(Str) -> List(Str)
distinct = |texts|
	texts.fold(
		[],
		|seen, text|
			if seen.contains(text) {
				seen
			} else {
				seen.append(text)
			},
	)

-- Maps a file of Shale's C: each top-level item, one a line, as
-- `first-last kind name`, from cookbook/c/items.shale's splitter, which
-- reads the dialect c/check_discipline.shale holds (that library's header
-- has the kinds and what each item carries).
--
--   shale cookbook:c/map_file.shale file=c/script/vm.c
--   shale cookbook:c/map_file.shale tree=c         every .c and .h under c/
--                                                  (c/out left out) split,
--                                                  each line accounted for
--   shale cookbook:c/map_file.shale self_test=1    the plants
--
-- `tree=` splits every file and proves the split: each line is in exactly
-- one item, or blank, or a `//` comment line between items, and it prints
-- one line, the files, lines and items by kind. A line the dialect does not
-- account for stops the run naming its file and line, in either mode.
--
-- `tree=c` takes about 17 million steps and a seventh of a second.
--
-- `self_test=1` maps the plants in cookbook/c/plants/map_file/ (a header and
-- a .c file holding an item of each kind) and holds each to its map, its
-- comments, heads, bodies and uses, then splits each plant in breaks/,
-- which must stop naming its file and the line it breaks.
--
-- budget: steps=50000000

use("./items.shale")

const PLANTS = "cookbook/c/plants/map_file"

-- The map of the items in `items`, one `first-last kind name` a line.
function map_lines(items)
  local out = {}
  for it in items do push(out, it.first .. "-" .. it.last .. " " .. it.kind .. " " .. it.name) end
  return out
end

-- The C files under `dir`, sorted, c/out (the build's output) left out.
function c_files_under(dir)
  local files = {}
  for e in walk_all(dir, {kind = "file", skip = "out"}) do
    if ends_with(e.path, ".c") or ends_with(e.path, ".h") then push(files, e.path) end
  end
  sort(files)
  return files
end

-- Each line of `p` in exactly one of `items`, or blank, or a `//` comment
-- line between them, or the run stops naming the line. Gives the count of
-- lines.
function accounted(p, items)
  local all = lines(read(p))
  local owner = {}
  for it in items do
    local k = it.first
    while k <= it.last do
      if has_key(owner, "" .. k) then fail(p .. ":" .. k .. ": in two items, `" .. owner["" .. k] .. "` and `" .. it.name .. "`") end
      owner["" .. k] = it.name
      k = k + 1
    end
  end
  local n = 0
  for raw in all do
    n = n + 1
    if not has_key(owner, "" .. n) and raw ~= "" and not starts_with(raw, "//") then fail(p .. ":" .. n .. ": in no item: " .. raw) end
  end
  return n
end

-- Every C file under `dir` split and accounted for, in one line.
function map_tree(dir)
  local files = c_files_under(dir)
  local kinds = {}
  local total = 0
  local count = 0
  for p in files do
    local items = c_items(p)
    total = total + accounted(p, items)
    for it in items do incr(kinds, it.kind) end
    count = count + #items
    yield()
  end
  local parts = {}
  local names = keys(kinds)
  sort(names)
  for k in names do push(parts, kinds[k] .. " " .. k) end
  print("PASS c items: " .. #files .. " files under " .. dir .. ", " .. total .. " lines, each in one item or between them; " .. count .. " items: " .. join(parts, ", "))
  return nil
end

-- `got` equal to `want`, or the self-test fails, naming `what`.
function same_or_fail(what, got, want)
  if got ~= want then fail("map_file self-test: " .. what .. " is\n" .. got .. "\nit should be\n" .. want) end
  return nil
end

-- The item named `name` of kind `kind` in `items`.
function item_named(items, kind, name)
  for it in items do
    if it.kind == kind and it.name == name then return it end
  end
  fail("map_file self-test: no " .. kind .. " `" .. name .. "`")
end

-- An item's lines as `comment a-b, sig c-d, body e-f`.

  return "comment " .. it.comment_first .. "-" .. it.comment_last .. ", sig " .. it.sig_first .. "-" .. it.sig_last .. ", body " .. it.body_first .. "-" .. it.body_last
end

  same_or_fail("kinds_sum", sum.storage .. " | " .. sum.type .. " | " .. spans(sum) .. " | " .. join(sum.uses, ","), "static inline | uint32_t | comment 0-0, sig 47-47, body 47-47 | p")
  local name = item_named(hi, "function", "kinds_name")
  same_or_fail("kinds_name", name.type .. " | " .. spans(name) .. " | " .. join(name.uses, ","), "Str | comment 49-50, sig 51-51, body 52-57 | KINDS_A,KINDS_LINUX,S,k,kinds_label")
  local check = item_named(hi, "prototype", "kinds_check")
  same_or_fail("kinds_check", check.type .. " | " .. spans(check), "Status | comment 0-0, sig 60-61, body 0-0")
  same_or_fail("KINDS_NAMES's spans", spans(item_named(hi, "data", "KINDS_NAMES")), "comment 0-0, sig 63-63, body 64-65")
  checks = checks + 11
  local c = PLANTS .. "/kinds.c"
  local ci = c_items(c, {uses = true})
  accounted(c, ci)
  same_or_fail(c .. "'s map", join(map_lines(ci), "\n"), join({
    "2-2 include test.h", "4-4 data kinds_hook", "6-11 function kinds_wrapped", "13-15 test kinds_test",
    "17-19 test_list TEST_MAIN", "21-23 function main"}, "\n"))
  checks = checks + 1
  local wrapped = item_named(ci, "function", "kinds_wrapped")
  same_or_fail("kinds_wrapped", spans(wrapped) .. " | " .. join(wrapped.uses, ","), "comment 0-0, sig 6-7, body 8-10 | a,b,kinds_add,other,p,sum")
  same_or_fail("kinds_test's uses", join(item_named(ci, "test", "kinds_test").uses, ","), "EXPECT,kinds_wrapped")
  same_or_fail("TEST_MAIN's uses", join(item_named(ci, "test_list", "TEST_MAIN").uses, ","), "TEST_MAIN,kinds_test")
  same_or_fail("main's spans", spans(item_named(ci, "function", "main")), "comment 0-0, sig 21-21, body 22-22")
  checks = checks + 4
  -- Each break stops, naming its file, its line and why.
  local breaks = {
    {f = "indented.c", said = "indented.c:3: an indented line outside every item"},
    {f = "column0.c", said = "column0.c:2: a line at column 0 that opens no item the dialect knows"},
    {f = "in_body.c", said = "in_body.c:3: a line at column 0 inside the function `f` opened at line 2, which only a `}` may close"},
    {f = "unclosed.c", said = "unclosed.c:2: the function `f` that no `}` at column 0 closes"},
    {f = "no_guard.h", said = "no_guard.h:1: a header with no `#ifndef SHALE_NO_GUARD_H`"},
    {f = "no_endif.h", said = "no_endif.h:5: a header whose last item is not its guard's `#endif`"},
    {f = "bad_test.c", said = "bad_test.c:2: a `TEST(` line that is not `TEST(name) {`"},
    {f = "wrapped_head.c", said = "wrapped_head.c:3: a line at column 0 inside the head opened at line 2"},
    {f = "bad_generator.c", said = "bad_generator.c:2: a generator line whose name and type this reader can't find"},
  }
  for b in breaks do
    local p = PLANTS .. "/breaks/" .. b.f
    local r = attempt(c_items, p)
    if r.ok then fail("map_file self-test: " .. p .. " split, with " .. #r.value .. " items; it should stop with " .. b.said) end
    if not starts_with(r.error, PLANTS .. "/breaks/" .. b.said .. ": ") then fail("map_file self-test: " .. p .. " stopped with `" .. r.error .. "`; it should stop with `" .. PLANTS .. "/breaks/" .. b.said .. ": <the line>`") end
    checks = checks + 1
  end
  print("PASS map_file self-test: " .. checks .. " checks, each item kind mapped and " .. #breaks .. " shape breaks stopped at their lines")
  return nil
end

if arg("self_test", 0) == 1 then
  self_test()
else
  local file = arg("file", "")
  local tree = arg("tree", "")
  if (file == "") == (tree == "") then fail("map_file: give `file=<a .c or .h>` or `tree=<a folder>`, one of them") end
  if file ~= "" then
    for l in map_lines(c_items(file)) do print(l) end
  else
    map_tree(tree)
  end
end
