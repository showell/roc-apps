## find_awkward_names, in Roc on Shale's platform (roc_platform/README.md): awkward
## names in every .roc file of a tree, from Roc's own parser's trees. Its
## launcher, cookbook/roc/find_awkward_names.shale, is how it is run, and its header
## says the checks, from `state_effect` to `stale_doc_name`.
## check_goldens.shale, beside this, holds its output on the plants to its
## goldens, line for line.
##
##   cd <repo> && shale cookbook:roc/find_awkward_names.shale glob=app/**/*.roc
##   shale cookbook:roc/find_awkward_names.shale self_test=1
##
## The arguments: `glob` (`**/*.roc`), `words` (`words=1` prints the census
## of effects' first words instead), `house` (`house=0` leaves out the house
## style), `types` (`1` asks Roc's type checker), `root` (the app or apps the
## types are read from), `terms` (a project's terms) and `self_test`
## (`self_test=1` checks the plants, the typed ones with the types).
##
## This root reads the word lists (Words), globs the files, gathers the
## types (Types), has Roc's parser read the files, reads each file's
## definitions (Reading), judges them (Judge, with ListNames and Makers),
## then the bindings (ShortNames), the docs (DocNames) and what is judged
## over every file (Passes), and prints the verdict; SelfTest judges the
## plants. Tree, Text, Finding and RocTypes are the cookbook's Roc library,
## cookbook/roc/lib/.
## budget: steps=800000000 arena_bytes=3000000000 output_bytes=4000000 roc_cpu_ms=600000 programs=100
app [main!] { lib: "../lib/main.roc", pf: platform "../../../roc_platform/platform/main.roc" }

			Kept => Bool.True
			LeftOut => !(house_style.contains(finding.check))
		}

		[]
	} else {
		reported.map(Finding.line_of)
	}

		census = Passes.census_lines(Passes.census(read.definitions, lists))

	}

	if self_test {
		census = Passes.census(read.definitions, lists)
		SelfTest.run!({ paths, findings: reported, census, finding_lines }, checks)
	}
}

## The files Roc's parser read, each with its text cut at each newline, and
## a `path: why` line for each it refused.
parse! : List(Str) => { files : List(Reading.FileRead), refused : List(Str) }
parse! = |paths| {
	var $files = []
	var $refused = []
	for file in Shale.roc_tree!(paths) {
		if file.err == "" {
			source = match Shale.read!(file.path) {
				Found(text) => text
				Missing => ""
			}
			$files = $files.append({ path: file.path, nodes: file.nodes, source_lines: source.split_on("\n") })
		} else if file.err.starts_with(file.path) {
			$refused = $refused.append(file.err)
		} else {
			$refused = $refused.append("${file.path}: ${file.err}")
		}
	}
	{ files: $files, refused: $refused }
}

## Every file's definitions, in file order, and the facts their reading
## gathers.
definitions_of : List(Reading.FileRead), RocTypes.Typing -> { definitions : List(Reading.Definition), facts : Facts.Gathered }
definitions_of = |files, typing|
	files.fold(
		{ definitions: [], facts: Facts.none },
		|so_far, file| {
			pattern_types = pattern_types_of(typing, file.path, file.nodes)
			read = Reading.file_definitions(file, pattern_types)
			{ definitions: so_far.definitions.concat(read.definitions), facts: Facts.joined(so_far.facts, read.facts) }
		},
	)

## The checker's type of each plain-name pattern of the file at `path`, by
## its node's place: none when no root reached the file.
pattern_types_of : RocTypes.Typing, Str, List(Tree.Node) -> Dict(U64, Str)
pattern_types_of = |typing, path, nodes| {
	if !(typing.real.contains(path)) {
		return Dict.empty()
	}
	var $typed = Dict.empty()
	$place : U64
	var $place = 0
	for node in nodes {
		$place = $place + 1
		if node.kind == "p-ident" {
			match RocTypes.type_at(typing, path, node) {
				Ok(written) => {
					$typed = $typed.insert($place, written)
				}
				Err(NoType) => {}
			}
		}
	}
	$typed
}

## Every finding, in the Shale original's order: each definition's, each
## file's `where` clauses and bindings, each file's docs, the type and
## field names, and the preposition splits.
judged! : Asked => List(Finding)
judged! = |asked| {
	lists = asked.lists
	read = asked.read
	facts = read.facts
	by_definition = read.definitions.map(|definition| Judge.findings(definition, lists, facts)).join()
	by_file = asked.files
		.map(|file| Passes.where_name_findings(file.path, file.nodes, lists.fixed_names).concat(ShortNames.findings(file.path, file.nodes)))
		.join()
	sources = Words.word_set!("cookbook:words/cited_sources.txt")
	# Each plant opens by naming its check, so in the self-test the check
	# names, and this recipe's, count as names.
	known = if asked.self_test {
		Set.from_list(checks).insert("find_awkward_names")
	} else {
		Set.empty()
	}
	doc_files = asked.files.map(|file| { path: file.path, nodes: file.nodes, source_lines: file.source_lines })
	by_doc = DocNames.findings!(doc_files, DocNames.seen_of(doc_files, known, sources))
	by_name = facts.names
		.map(
			|named| {
				subject = { path: named.path, line: named.line, name: named.name, kind: "field", is_function: named.is_function, first: named.first, gives: named.gives }
				listed = if named.is_field {
					ListNames.findings(subject, lists)
				} else {
					[]
				}
				Passes.container_findings(named, lists).concat(listed)
			},
		)
		.join()
	splits = Passes.preposition_splits(read.definitions, lists, facts.keywords)
	by_definition.concat(by_file).concat(by_doc).concat(by_name).concat(splits)
}

## Whether `house=0` leaves the house style out (cookbook/roc/house.shale):
## `house` is read as an integer, 1 unless given.
read_house! : {} => [Kept, LeftOut]
read_house! = |{}| {
	given = Shale.arg!("house", "1")
	match integer_of(given) {
		Ok(0) => LeftOut
		Ok(1) => Kept
		Ok(other) => {
			Shale.fail!("house must be 1 (the default) or 0, not ${other.to_str()}")
			Kept
		}
		Err(NotAnInteger) => {
			Shale.fail!("`arg`: house=${given} is not an integer, and its default is one")
			Kept
		}
	}
}

## A launch value read as an integer, as Shale's `arg` reads one whose
## default is: digits, a `-` before them or not.
integer_of : Str -> Try(I64, [NotAnInteger])
integer_of = |text| {
	bytes = text.to_utf8()
	digits = match bytes {
		['-', .. as rest] => rest
		_ => bytes
	}
	if digits.is_empty() or !(digits.all(Text.is_digit)) {
		return Err(NotAnInteger)
	}
	value = digits.fold(0, |sum, digit| sum * 10 + (digit - '0').to_i64())
	if bytes.first() == Ok('-') {
		Ok(-value)
	} else {
		Ok(value)
	}
}

## After the findings: the files that didn't parse, and those no root of the
## types reaches (judged without them), named, then a stop
## (cookbook/roc/parsed.shale's `fail_if_refused`).
fail_if_refused! : List(Str), List(Str) => {}
fail_if_refused! = |refused, unreached| {
	if !(refused.is_empty()) or !(unreached.is_empty()) {
		for refusal in refused {
			Shale.print!("refused: ${refusal}")
		}
		for missed in unreached {
			Shale.print!(missed)
		}
		var $why = []
		if !(refused.is_empty()) {
			$why = $why.append("roc-ast refused ${refused.len().to_str()} files (named above), which have no tree to judge")
		}
		if !(unreached.is_empty()) {
			$why = $why.append("${unreached.len().to_str()} files (named above) are reached by no root of the types, so were judged without them")
		}
		Shale.fail!("find_awkward_names: ${Str.join_with($why, "; ")}; every other file was judged")
	}
}

## What find_awkward_names' `stale_doc_name` reads (the Shale original's
## doc_names.shale): the names a file's comment lines put in backticks, and whether each still
## names something in the file, the modules it imports, or the module a
## `Module.name` names. Only a name a reader can't take for a prose word is
## judged: one of two words or more (`two_opt`), or a module's member
## (`Route.two_opt`). A name cited from another language or library, or
## with a possessive before it, is another's. Reading a module beside the
## file that the run didn't parse is an effect, so this module has them.
import pf.Shale
import lib.Finding
import lib.Text
import lib.Tree

DocNames := [].{

	## What the run has read of names: the self-test's own (`known`), each
	## module's by its name, each file's by its path (`Unread` for one that
	## couldn't be), and the sources a citation names.
	Seen : {
		known : Set(Str),
		by_module : Dict(Str, Set(Str)),
		by_path : Dict(Str, [Read(Set(Str)), Unread]),
		sources : Set(Str),
	}

	## A parsed file, its lines and its tree.
	FileRead : { path : Str, nodes : List(Tree.Node), source_lines : List(Str) }

	## The names the parsed files hold, by module and by path.
	seen_of : List(DocNames.FileRead), Set(Str), Set(Str) -> DocNames.Seen
	seen_of = |files, known, sources| {
		var $by_module = Dict.empty()
		var $by_path = Dict.empty()
		for file in files {
			module_named = Tree.module_name(file.path)
			own = names_in_tree(file.nodes)
			$by_module = $by_module.insert(module_named, ($by_module.get(module_named) ?? Set.empty()).union(own))
			$by_path = $by_path.insert(file.path, Read(own))
		}
		{ known, by_module: $by_module, by_path: $by_path, sources }
	}

	## Each backticked name in a comment of each file that names nothing
	## there, once a line, file by file.
	findings! : List(DocNames.FileRead), DocNames.Seen => List(Finding)
	findings! = |files, seen_before| {
		var $seen = seen_before
		var $found = []
		for file in files {
			judged = file_findings!(file, $seen)
			$seen = judged.seen
			$found = $found.concat(judged.found)
		}
		$found
	}
}

## A backticked name: its line, its module (`""` for a bare name), the name,
## and whether a possessive or a cited source says it is another's.
DocName : { line : U64, module_part : Str, name : Str, is_others : Bool }

## The file's stale names, and what the run has read of names since.
file_findings! : DocNames.FileRead, DocNames.Seen => { found : List(Finding), seen : DocNames.Seen }
file_findings! = |file, seen_before| {
	var $seen = seen_before
	own = match seen_before.by_path.get(file.path) {
		Ok(Read(names)) => names
		_ => names_in_tree(file.nodes)
	}
	base = Text.basename(file.path)
	folder = Str.from_utf8_lossy(file.path.to_utf8().take_first(file.path.count_utf8_bytes() - base.count_utf8_bytes()))
	own_module = Tree.module_name(file.path)
	is_platform = file.nodes.any(|node| node.kind == "platform")
	var $imported = []
	var $all_imports_known = Bool.True
	for imported in imports_in_tree(file.nodes) {
		looked = module_names!(folder, imported, $seen)
		$seen = looked.seen
		match looked.names {
			Ok(names) => {
				$imported = $imported.append(names)
			}
			Err(Unknown) => {
				$all_imports_known = Bool.False
			}
		}
	}
	var $said = Set.empty()
	var $found = []
	for doc_name in doc_names_in(file.source_lines, $seen.sources) {
		name = doc_name.name
		plain = name.drop_suffix("!")
		holds = |names| names.contains(name) or names.contains(plain)
		is_host_symbol = is_platform and plain.starts_with("roc_")
		is_bare = doc_name.module_part == ""
		var $judged = !(is_bare and (doc_name.is_others or !($all_imports_known) or is_host_symbol))
		var $names_something = Bool.False
		if is_bare or doc_name.module_part == own_module {
			if holds(own) or $seen.known.contains(plain) {
				$names_something = Bool.True
			}
			if is_bare and $imported.any(holds) {
				$names_something = Bool.True
			}
		} else {
			looked = module_names!(folder, doc_name.module_part, $seen)
			$seen = looked.seen
			match looked.names {
				Ok(names) => {
					if holds(names) {
						$names_something = Bool.True
					}
				}
				Err(Unknown) => {
					$judged = Bool.False
				}
			}
		}
		shown = if is_bare {
			name
		} else {
			"${doc_name.module_part}.${name}"
		}
		key = "${doc_name.line.to_str()}|${shown}"
		if $judged and !($names_something) and !($said.contains(key)) {
			$said = $said.insert(key)
			place = if is_bare {
				"this file or its imports"
			} else {
				doc_name.module_part
			}
			what = "a comment names `${shown}`, which names nothing in ${place}: a rename left it behind, or it names what is gone"
			$found = $found.append({ path: file.path, line: doc_name.line, check: "stale_doc_name", what })
		}
	}
	{ found: $found, seen: $seen }
}

## The names module `module` holds, or `Unknown` when they can't be known
## (a builtin's module, a package's the run didn't read): those of
## `<folder><module>.roc` beside the file, read once, else those of the
## files read that hold a module of that name.
module_names! : Str, Str, DocNames.Seen => { names : Try(Set(Str), [Unknown]), seen : DocNames.Seen }
module_names! = |folder, module_named, seen| {
	path = "${folder}${module_named}.roc"
	looked_up = match seen.by_path.get(path) {
		Ok(known) => { read: known, after: seen }
		Err(_) => {
			read_now = if Shale.exists!(path) {
				match Shale.roc_tree!([path]) {
					[parsed, ..] if parsed.err == "" => Read(names_in_tree(parsed.nodes))
					_ => Unread
				}
			} else {
				Unread
			}
			{ read: read_now, after: { ..seen, by_path: seen.by_path.insert(path, read_now) } }
		}
	}
	read = looked_up.read
	after = looked_up.after
	names = match read {
		Read(names_read) => Ok(names_read)
		Unread =>
			match seen.by_module.get(module_named) {
				Ok(names_held) => Ok(names_held)
				Err(_) => Err(Unknown)
			}
	}
	{ names, seen: after }
}

## Every name a file's tree holds: what it declares, binds, calls, each
## record field and method it names, and each symbol a platform's
## `provides` gives the host, with and without an effect's `!`.
names_in_tree : List(Tree.Node) -> Set(Str)
names_in_tree = |nodes| {
	var $names = Set.empty()
	for node in nodes {
		for attribute in node.attributes {
			if ["raw", "name", "field", "method", "symbol"].contains(attribute.name) {
				value = attribute.value.drop_prefix(".")
				if value != "" {
					$names = $names.insert(value)
					if value.ends_with("!") {
						$names = $names.insert(value.drop_suffix("!"))
					}
				}
			}
		}
	}
	$names
}

## The modules a file imports, by the name its code calls them (`Route`,
## and `Stdout` for `pf.Stdout`).
imports_in_tree : List(Tree.Node) -> List(Str)
imports_in_tree = |nodes|
	nodes.keep_if(|node| node.kind == "s-import").map(|node| Tree.last_part(Tree.attribute(node, "raw")))

## A file's extensions, so `Screen.roc` is a file's name, not a member's.
file_extensions : List(Str)
file_extensions = ["roc", "md", "txt", "zig", "ts", "js", "mjs", "json", "shale", "html", "tsv"]

## Each backticked name in the comment lines of a file (`#` or `##` first),
## as the header says which are judged.
doc_names_in : List(Str), Set(Str) -> List(DocName)
doc_names_in = |source_lines, sources| {
	var $out = []
	$number : U64
	var $number = 0
	for source_line in source_lines {
		$number = $number + 1
		line = Text.trim(source_line)
		if line.starts_with("#") {
			for span in backticked(line) {
				match doc_name_parts(span.text) {
					Ok(parts) => {
						before_on_line = Text.trim(without_hashes(Text.bytes_of(line, { start: 0, count: span.at })))
						before = if before_on_line == "" and $number > 1 {
							Text.trim(Text.line_at(source_lines, $number - 1))
						} else {
							before_on_line
						}
						is_others = is_possessive("${before} ") or cites_source(sentence_before(source_lines, $number, span.at), sources)
						is_file = parts.module_part != "" and file_extensions.contains(parts.name)
						if (parts.module_part != "" or parts.name.contains("_")) and !(is_file) {
							$out = $out.append({ line: $number, module_part: parts.module_part, name: parts.name, is_others })
						}
					}
					Err(NoName) => {}
				}
			}
		}
	}
	$out
}

## Each run between two backticks with something between them, left to
## right: `at` is how many bytes stand before its opening backtick.
backticked : Str -> List({ at : U64, text : Str })
backticked = |line| {
	bytes = line.to_utf8()
	var $spans = []
	$at : U64
	var $at = 0
	while $at < bytes.len() {
		if bytes.get($at) == Ok('`') {
			rest = bytes.drop_first($at + 1)
			match rest.find_first_index(|byte| byte == '`') {
				Ok(0) => {
					$at = $at + 1
				}
				Ok(length) => {
					$spans = $spans.append({ at: $at, text: Str.from_utf8_lossy(rest.take_first(length)) })
					$at = $at + length + 2
				}
				Err(_) => {
					$at = bytes.len()
				}
			}
		} else {
			$at = $at + 1
		}
	}
	$spans
}

## A backticked text's module and name when it is a doc name:
## `two_opt`, `read_line!`, `Route.two_opt(stops)`
## (the Shale original's DOC_NAME).
doc_name_parts : Str -> Try({ module_part : Str, name : Str }, [NoName])
doc_name_parts = |text| {
	bytes = text.to_utf8()
	match bytes {
		[first, .. as rest] if Text.is_capital(first) => {
			module_length = 1 + (rest.find_first_index(|byte| !(is_letter_or_digit(byte))) ?? rest.len())
			if bytes.get(module_length) != Ok('.') {
				return Err(NoName)
			}
			name = name_of(bytes.drop_first(module_length + 1))?
			Ok({ module_part: Str.from_utf8_lossy(bytes.take_first(module_length)), name })
		}
		_ => Ok({ module_part: "", name: name_of(bytes)? })
	}
}

## The name a doc name's bytes start with, `[a-z][a-z0-9]*(_[a-z0-9]+)*!?`,
## when nothing or a call's parentheses follow it.
name_of : List(U8) -> Try(Str, [NoName])
name_of = |bytes| {
	is_word_byte = |byte| Text.is_small_letter(byte) or Text.is_digit(byte)
	match bytes.first() {
		Ok(first) if Text.is_small_letter(first) => {}
		_ => return Err(NoName)
	}
	$length : U64
	var $length = 1
	while $length < bytes.len() and is_word_byte(bytes.get($length) ?? 0) {
		$length = $length + 1
	}
	while bytes.get($length) == Ok('_') and is_word_byte(bytes.get($length + 1) ?? 0) {
		$length = $length + 2
		while $length < bytes.len() and is_word_byte(bytes.get($length) ?? 0) {
			$length = $length + 1
		}
	}
	if bytes.get($length) == Ok('!') {
		$length = $length + 1
	}
	after = bytes.drop_first($length)
	is_call = after.len() >= 2 and after.first() == Ok('(') and after.last() == Ok(')')
	if after.is_empty() or is_call {
		Ok(Str.from_utf8_lossy(bytes.take_first($length)))
	} else {
		Err(NoName)
	}
}

## The places of a text's sentence ends: a `.`, `?` or `!` with a space or
## the text's end after it.

sentence_ends = |text| {
	bytes = text.to_utf8()

	$at : U64
	var $at = 0
	for byte in bytes {
		is_mark = byte == '.' or byte == '?' or byte == '!'
		is_last = $at + 1 == bytes.len()
		if is_mark and (is_last or is_space(bytes.get($at + 1) ?? 0)) {
			$ends = $ends.append($at)
		}
		$at = $at + 1
	}
	$ends
}

## The text of the sentence before byte `at` of comment line `number`:
## back across the comment lines above, their `#`s gone, until a `.`, `?`
## or `!` that ends a sentence, a blank comment line, or the comment's
## start.
sentence_before : List(Str), U64, U64 -> Str
sentence_before = |source_lines, number, at| {
	on_line = Text.bytes_of(Text.trim(Text.line_at(source_lines, number)), { start: 0, count: at })
	var $text = Text.trim(without_hashes(on_line))
	var $up = number - 1
	var $is_blank_above = Bool.False
	while !($is_blank_above) and sentence_ends($text).is_empty() and $up >= 1 and Text.trim(Text.line_at(source_lines, $up)).starts_with("#") {
		above = Text.trim(without_hashes(Text.trim(Text.line_at(source_lines, $up))))
		if above == "" {
			$is_blank_above = Bool.True
		} else {
			$text = "${above} ${$text}"
			$up = $up - 1
		}
	}
	match sentence_ends($text).last() {
		Ok(last) => Text.bytes_of($text, { start: last + 1, count: $text.count_utf8_bytes() })
		Err(_) => $text
	}
}

## The word at place `at` of `words`, counted from 1.
word_at : List(Str), U64 -> Str
word_at = |words, at| words.get(at - 1) ?? ""

## `words` with each run of words that spells a term joined into one word,
## the longest term first: `two`, `opt`, `stops` gives `two_opt`, `stops`.
joined_terms : List(Str), Words.Terms -> List(Str)
joined_terms = |words, terms| {
	term_words = terms.words.to_list().map(|(_, spelled)| spelled)
	var $joined = []
	$at : U64
	var $at = 0
	while $at < words.len() {
		rest = words.drop_first($at)
		taken = term_words.fold(
			1,
			|longest, spelled| if spelled.len() > longest and rest.take_first(spelled.len()) == spelled {
				spelled.len()
			} else {
				longest
			},
		)
		$joined = $joined.append(Str.join_with(rest.take_first(taken), "_"))
		$at = $at + taken
	}
	$joined
}

## The place of the word of a name that says what it is, counted from 1:
## the one before the first preposition or word list_names.txt calls
## `after`, past the name's first word (`trees_in_view` gives `trees`), else
## the last that is not digits alone (`corner_2` gives `corner`).
head_word_at : List(Str), Words.Lists -> U64
head_word_at = |words, lists| {
	$place : U64
	var $place = 2
	while $place <= words.len() {
		word = word_at(words, $place)
		if lists.prepositions.contains(word) or lists.list_names.get(word) == Ok("after") {
			return $place - 1
		}
		$place = $place + 1
	}
	var $at = words.len()
	while $at > 1 and is_digits(word_at(words, $at)) {
		$at = $at - 1
	}
	$at
}

is_digits : Str -> Bool
is_digits = |word| {
	bytes = word.to_utf8()
	!(bytes.is_empty()) and bytes.all(Text.is_digit)
}

## Whether a word reads as a plural. An `s` doesn't make one in `class`, in a
## word list_names.txt calls `one` (`axis`), or after letters that are no
## word (`pts`); after one letter it does (`xs`).
reads_as_plural : Str, Words.Lists -> Bool
reads_as_plural = |word, lists| {
	if lists.list_names.get(word) == Ok("one") or !(word.ends_with("s")) or word.ends_with("ss") {
		return Bool.False
	}
	stem = word.drop_suffix("s").to_utf8()
	stem.len() == 1 or stem.any(|byte| byte == 'a' or byte == 'e' or byte == 'i' or byte == 'o' or byte == 'u' or byte == 'y')
}

## A word's plural, as English spells most.
plural_of : Str -> Str
plural_of = |word| {
	is_vowel_before_last = match word.to_utf8() {
		[.., before, _] => before == 'a' or before == 'e' or before == 'i' or before == 'o' or before == 'u'
		_ => Bool.False
	}
	if word.ends_with("y") and !(is_vowel_before_last) {
		"${word.drop_suffix("y")}ies"
	} else if ["s", "x", "z", "ch", "sh"].any(|ending| word.ends_with(ending)) {
		"${word}es"
	} else {
		"${word}s"
	}
}

## The plural of what a list of `element` holds, for a name to say it, or
## `""` when the element has no word of its own: a type variable, a
## number, text.
held_plural : Str -> Str
held_plural = |element| {
	held = TypeText.element_word(element)
	starts_capital = match element.to_utf8().first() {
		Ok(byte) => Text.is_capital(byte)
		Err(_) => Bool.False
	}
	if held == "" or !(starts_capital) or plain_elements.contains(element) {
		""
	} else {
		plural_of(held)
	}
}

## The names `singular_list` offers for `bare`, whose head word is at `at`
## in `words`: the plural of what the list holds put after the head, as in
## `herd_cows`, or the head's own plural when the element is that word, a
## number or text; else the name with `_list`.
honest_list_names : Str, List(Str), U64, Str -> Str
honest_list_names = |bare, words, at, element| {
	held = held_plural(element)
	says_what_it_holds = held != "" and TypeText.element_word(element) != word_at(words, at)
	var $plural_words = []
	$place : U64
	var $place = 0
	for word in words {
		$place = $place + 1
		if $place != at {
			$plural_words = $plural_words.append(word)
		} else if says_what_it_holds {
			$plural_words = $plural_words.append(word).append(held)
		} else {
			$plural_words = $plural_words.append(plural_of(word))
		}
	}
	"`${Str.join_with($plural_words, "_")}`, or `${bare}_list`"
}

--                   well as the type: `Scene.in_view : View -> Scene` gives
--                   its own type and still reads as a question;
--                 - a conversion: `to_*` and `into_*`, which name what they
--                   give;
--                 - a `from_*` that takes or gives its own module's type, as
--                   above, even inside a record, a union or a list
--                   (`Along.from_start : Along -> Distance`,
--                   `Key.from_path : Str -> Try({ key : Key }, [NoKey])`). A
--                   `Try` gives its success only: an error that carries the
--                   type is no conversion to it;
--                 - a `from_<noun>` in a module that has no type of its own,
--                   one whose file declares only `Hex :: [].{ ... }`
--                   (`Hex.from_hex : Str -> Try(List(U8), [NotHex])`). Such a
--                   module has no type to take or give. A module with a type
--                   is held to the rule above, so
--                   `DateTime.from_clock : I64 -> Try(Str, [OutOfRange])`,
--                   which neither takes nor gives a `DateTime`, is reported.
--                   The next word has to be no preposition, so `from_behind`
--                   stays a phrase, and the type has to be known;
--                 - a changed copy: `with_*` or `without_*` that gives its
--                   first argument's type (`with_lean : Bike, Angle -> Bike`).
--                 A definition with no annotation shows none of this, so it
--                 is reported unless `types=1` supplies its type. A list that
--                 is no function, a value, a field or a parameter, named from
--                 a preposition, that `singular_list` would report
--                 (`by_slot : List(List(Stop))`) is reported here instead: it
--                 wants its noun (`routes_by_slot`), not a plural.
--   singular_list  A name that holds or gives a `List(...)`, or a `Try` of
--                 one, but reads as one thing: `triangle : List(Shape)`. The
--                 name ends in a plural of what the list holds (`triangles`,
--                 `herd_cows`) or is explicit (`triangle_list`). A collective
--                 noun is no plural: `herd`, `crowd` and `route` are reported.
--                 Read are top-level and member definitions, record fields,
--                 and the plain parameters of a top-level or member lambda.
--                 The word that counts is the one before a preposition or an
--                 `after` word of cookbook/words/list_names.txt, else the
--                 last (`trees_in_view`, `corners_ahead`). These pass:
--                 - a `List(U8)`, whose bytes are plumbing: the name says what
--                   they mean (`boundary`). A `List(F64)` is checked, since a
--                   flat run of coordinates can't be told from many readings;
--                 - a word list_names.txt calls `many`, one that can't be
--                   counted, and a word read as a participle (`clipped`);
--                 - a function whose whole name is a verb (`draw`) or a term
--                   (`two_opt`), and a changed copy: a function that gives
--                   its first argument's type, whatever its first word
--                   (`trim_hull`, `coalesce`, `stable_by_load_desc`);
--                 - a word the project's terms (`terms=`) say names a list
--                   of its element (`route` of `Stop`s);
--                 - one letter, and a value named by a number, which are
--                   `short_name`'s and `taken_word`'s to judge;
--                 - an accumulator's short idiom, as a whole name (`out`);
--                 - a name `phrase_name` reports, to be named again anyway.
--                 The word list, a participle and a verb don't pass a word
--                 that is the element's own (`route : List(Route)`), and a
--                 changed copy of more than one word does, as its object is
--                 one element (`set_run : List(Run), Run -> List(Run)`). A word
--                 verbs.txt holds that list_names.txt calls `one` is no verb
--                 here (`route`). An `s` makes no plural after `s` (`class`),
--                 after letters that are no word (`pts`), or in a `one` word
--                 (`axis`). A word list_names.txt calls `adjective`
--                 (`stable`, `cheapest`) is no noun, so the finding says the
--                 name has none and offers no plural of it. A term's words
--                 read as one (`two_opt_pass`'s word is `pass`). An alias of
--                 a list isn't seen, and with `types=1` an unannotated
--                 definition or parameter is read by its inferred type.
--   world_verb    A pure function whose first word is a verb of
--                 cookbook/words/world_verbs.txt, one that claims to act on
--                 something outside (`draw`, `paint_shape`). Nothing is drawn
--                 by a function that only makes a value, so name it for that
--                 value (`tree_shapes`). A verb that says what happens to its
--                 argument is fine (`plus`, `clamp`, `trim_hull`). An effect,
--                 which does act, and a value are not reported. A `##` doc
--                 directly above the definition (or its annotation) answers
--                 it, saying why the verb stands: a renderer named for what
--                 its output is drawn as, as `single`'s doc answers it in
--                 check_boundaries.
--   where_name    A method in a `where` clause named by one word
--                 (`[a.height : a -> U64]`). A clause matches by name alone,
--                 so the name is the whole contract: two words say what it is
--                 for (`scenery_height`). The methods Roc calls by name
--                 (`is_eq`, `plus`, `map`) are fine. A local's clause is read.
--   stale_doc_name  A backticked name in a comment line (`#` or `##` first)
--                 that names nothing in the file or the modules it imports:
--                 a rename leaves its comments behind (rename.shale counts
--                 and leaves them). A bare name is judged when it has two
--                 words or more (`cheapest_insert`), since a one-word one
--                 (`for`, `fold`) is often the language's or the prose's; a
--                 `Module.name` (`Route.two_opt`, a call's parentheses
--                 allowed) is judged against `Module.roc` beside the file,
--                 else a module of that name among the files read (a
--                 package's, `import game.SunPath`), and left when there is
--                 none, as for `List.map` (cookbook/roc/doc_names.shale).
--                 An import is read the same way, and one whose names can't
--                 be read leaves the file's bare names unjudged, as does a
--                 possessive before one (`roc-ray's`), or a language or
--                 library of cookbook/words/cited_sources.txt before it in
--                 its sentence ("Rust's `is_ascii_digit`", "Zig `isDigit`,
--                 Rust `is_ascii_digit`"): it is another's. A
--                 platform's `provides` symbols (`roc_plan`) are its names,
--                 and its bare `roc_` names (`roc_crashed`) the host's.
--
-- Each name is judged once: from its annotation when it has one, else from
-- its definition. An effect that takes an effectful function (`map2!`,
-- `on_err!`: an argument's type holds a `=>`, or a lambda argument's name
-- ends in `!`) has its `!` from that callback, so it is judged as the name
-- without the `!`. Names chosen elsewhere are skipped: the methods Roc calls
-- by name (cookbook/roc/headers.shale), and the top-level names a platform
-- dictates, which are what an app's header provides (`app [loaded, step]`)
-- and what a platform's header requires and provides.
--
-- The output is the counts, one line per finding,
-- `path:line: <check>: <why>`, and the counts again
-- (cookbook/findings.shale).
--
-- The word lists are in cookbook/words/, shared with the Zig name hunt.
-- participles.txt lists the irregular past participles; any other word
-- ending in `ed` counts as one unless ed_verbs.txt lists it. With `words=1`
-- the recipe prints a census instead of the findings: each first word the
-- effects start with, how many start with it, and its verdict: `verb`,
-- `participle`, `vague`, `asks` (`is`, `has`) or `refused` (what
-- `noun_effect` reports). A verb missing from verbs.txt shows up there as
-- `refused`.
--
-- With `types=1` the recipe asks Roc's type checker for each definition's
-- type (cookbook/roc/types.shale). It needs `root=` and the tree's
-- `roc-version` file. A root the checker refuses, or a roc-types built for
-- another Roc version, stops the run. A file that no root imports is
-- checked without types, listed after the findings, and fails the run.
--
--   cd <repo> && shale <shale>/cookbook/roc/hunt_names.shale glob=app/*.roc root=app/main.roc types=1
--
-- `self_test=1` runs the recipe over `cookbook/roc/plants/hunt_names/`. Each
-- plant is named for the check it holds (`question.roc`,
-- `question_where.roc`) and must make that check report, and no other. A
-- `quiet_*.roc` plant must make no check report. Every check needs a plant,
-- and a plant named for no check fails the test. A plant another imports is
-- a CamelCase module, read as snake case (`QuietDocShelf.roc` is
-- `quiet_doc_shelf.roc`); the plants' terms are `terms.txt` there; each plant
-- opens by naming its check, so the check names count as names for
-- `stale_doc_name`; and cookbook/roc/naming_plants.shale holds these
-- rules, with what a finding must say, or not. The census must also give
-- the right verdict for one known word of the plants per verdict. The
-- plants in `typed/` are checked with the types, from their app
-- `typed/QuietApp.roc`, and their CamelCase module names read as snake case
-- (`QuestionInferred` as `question_inferred`); the rest are checked
-- without. The self-test fails without roc-types.
--
-- The recipe reads each file's source as well as its tree, to tell `->`
-- from `=>`, which the tree writes alike. On a large tree that needs more
-- than the default arena.
--
-- What it reads of each definition is cookbook/roc/declared.shale's
-- `file_definitions`, and the checks judged over the whole tree once every
-- file is read are cookbook/roc/naming_passes.shale's.

function report(path, line, check, why)
  if house_leaves_out("hunt_names", check) then return nil end
  if not CENSUS_ONLY then push(finding_lines, path .. ":" .. line .. ": " .. check .. ": " .. why) end
  count_by_check[check] = count_by_check[check] + 1
  fired_in_file[path .. "|" .. check] = true
  return nil
end

function is_participle(word)
  return reads_as_participle(word, PARTICIPLES, ED_VERBS)
end

for word in keys(VERBS) do
  if is_participle(word) then fail("hunt_names: `" .. word .. "` is in cookbook/words/verbs.txt and reads as a participle, which would make its effects states") end
end

-- What an effect's first word says: `participle`, `verb`, `vague`, `asks`
-- (`is`, `has`) or `refused`.
function first_word_verdict(word)
  if is_participle(word) then return "participle" end
  if has_key(VERBS, word) then return "verb" end
  if has_key(VAGUE, word) then return "vague" end
  if word == "is" or word == "has" then return "asks" end
  return "refused"
end

-- The checker's type of each plain-name pattern in the parsed file `f`, by
-- its node's index as text: none unless USE_TYPES and the checker reached
-- the file.
function pattern_types_in(f)
  local typed = {}
  if not (USE_TYPES and has_key(CHECKER_TYPES.real, f.path)) then return typed end
  for i = 1, #f.nodes do
    if f.nodes[i].kind == "p-ident" then
      local checker_type = roc_type_at(CHECKER_TYPES, "pat", f.path, f.nodes[i])
      if checker_type ~= nil then typed[tostring(i)] = checker_type end
    end
  end
  return typed
end

-- Every top-level and module-member annotation and definition in the
-- parsed files, as declared.shale's `file_definitions` reads them, filling
-- FILE_FACTS along the way.
function read_definitions(parsed_files)
  local definitions = {}
  for f in parsed_files do
    local src = split(read(f.path), "\n")
    local known = {dictated = header_names(f.nodes).dictated, pattern_types = pattern_types_in(f), recursion_suffixes = RECURSION_SUFFIXES}
    for def in file_definitions(f, src, known, FILE_FACTS) do push(definitions, def) end
  end
  return definitions
end

function all_tags_past(tags)
  for t in tags do
    if not is_participle(tag_first_word(t)) then return false end
  end
  return true
end

-- What a definition's type says, for `question`, `taken_word` and
-- `phrase_name`: `{is_function, first, result, known, source_note}`. The type
-- is the checker's when it gave one, else the annotation's. `known` is false
-- for a definition with no annotation and no checker's type.
function definition_signature(def)
  local signature = {is_function = def.annotation and def.fn, first = get(def, "first", ""), result = get(def, "result", ""), known = def.annotation, source_note = ""}
  if has_key(def, "typed") then
    local checker = roc_type_parts(def.typed)
    signature = {is_function = checker.fn, first = first_argument(checker.args), result = checker.result, known = true, source_note = " (the checker's type)"}
    if has_key(BOOL_ALIASES, checker.result) then signature.result = "Bool" end
  end
  return signature
end

-- Reports a name whose type says many and whose head word says one:
-- `singular_list` in the header. `subject` is `{path, line, name, kind,
-- is_function, first, gives}`, `kind` being what the report calls it.
function judge_list_name(subject)
  local found = list_name_finding(subject, LIST_WORDS)
  if found ~= nil then report(subject.path, subject.line, found.check, found.why) end
  return nil
end

-- Reports what each check finds in one definition record, except
-- `container` and `preposition_split`, which are judged over all the files
-- afterwards.
function judge_definition(def)
  local path = def.path
  local line = def.line
  local full_name = def.name
  if has_key(LANGUAGE_FIXED_NAMES, full_name) then return nil end
  local has_bang = ends_with(full_name, "!")
  local name = has_bang and sub(full_name, 1, #full_name - 1) or full_name
  local is_effect = has_bang and not def.callback
  local ws = split(name, "_")
  local last_word = ws[#ws]
  local is_annotation = def.annotation
  -- A definition paired with an annotation was judged at the annotation.
  local judged_here = not def.paired
  local asks_inside = contains(name, "_is_") or contains(name, "_has_")
  local has_namespace = is_effect and #ws > 1 and has_key(NAMESPACES, ws[1])
  local act_word = has_namespace and ws[2] or ws[1]
  local act_verdict = first_word_verdict(act_word)
  if has_namespace and act_verdict == "refused" and has_key(NAMESPACES[ws[1]], act_word) then act_verdict = "verb" end
  local act_place = has_namespace and ("the word after its namespace `" .. ws[1] .. "_`") or "its first word"
  local named_as_state = is_participle(act_word) and not asks_inside
  if is_effect and judged_here then
    if not has_key(first_word_census, act_word) then first_word_census[act_word] = {word = act_word, count = 0, verdict = act_verdict} end
    first_word_census[act_word].count = first_word_census[act_word].count + 1
  end
  local arguments = get(def, "args") or {}
  local draws_on_surface = #arguments > 0 and has_key(SURFACES, get(arguments[1], "ref", ""))
  if is_effect and judged_here and act_verdict == "refused" and not draws_on_surface then
    if has_key(PREPOSITIONS, act_word) then report(path, line, "noun_effect", "`" .. full_name .. "` has an effect but " .. act_place .. " is a preposition, `" .. act_word .. "`: an effect is named for its act")
    else report(path, line, "noun_effect", "`" .. full_name .. "` has an effect but " .. act_place .. ", `" .. act_word .. "`, is no verb: it reads as a value, not an act") end
  end
  if judged_here and #ws > 1 and has_key(PREPOSITIONS, last_word) then note_trailing_preposition(def, ws) end
  local names_drawn_type = #ws > 1 and has_key(get(def, "types") or {}, camel_case(ws))
  if is_effect and judged_here and named_as_state and not names_drawn_type then report(path, line, "state_effect", "`" .. full_name .. "` has an effect but is named as a state, `" .. act_word .. "`: a verb says what it does") end
  local named_as_question = #ws > 1 and (ws[1] == "is" or ws[1] == "has" or ws[1] == "can" or ws[1] == "should")
  local signature = definition_signature(def)
  local returns_bool = signature.is_function and signature.result == "Bool"
  if named_as_question and judged_here and is_effect and not returns_bool then report(path, line, "question", "`" .. full_name .. "` asks a question but has an effect" .. (signature.is_function and (" and gives `" .. signature.result .. "`" .. signature.source_note) or "")) end
  if returns_bool and named_as_state and judged_here then report(path, line, "question", "`" .. full_name .. "` gives a `Bool`" .. signature.source_note .. " but is named as a state, `" .. ws[1] .. "`: ask it (`is_*`, `has_*`)") end
  if named_as_question and judged_here and signature.known and signature.is_function and not is_effect and signature.result ~= "Bool" and signature.result ~= "_" then report(path, line, "question", "`" .. full_name .. "` asks a question but gives `" .. signature.result .. "`" .. signature.source_note) end
  if judged_here then
    local all_vague = true
    for w in ws do
      if not has_key(VAGUE, w) then all_vague = false end
    end
    if #ws == 1 and has_key(ENTRY_POINT_NAMES, ws[1]) then all_vague = false end
    if all_vague then report(path, line, "vague", "`" .. full_name .. "` is only vague words: what does it handle, and how?") end
    if #name == 1 then report(path, line, "short_name", "`" .. full_name .. "` is one letter, read far from where it is defined: name what it holds") end
    local has_leading_noun = false
    for w in take(ws, max(#ws - 2, 0)) do
      if not has_key(PREPOSITIONS, w) then has_leading_noun = true end
    end
    if has_leading_noun and has_key(PREPOSITIONS, last_word) and has_key(PREPOSITIONS, ws[#ws - 1]) then report(path, line, "stacked_preposition", "`" .. full_name .. "` stacks two prepositions, `_" .. ws[#ws - 1] .. "_" .. last_word .. "`: let each read once, the entry named for what it gives (`rest`, `tail_after`) and the worker for what it adds (`rest_onto`)") end
    -- A function is known by its type, else by a lambda definition; a value
    -- by a type that is no function, else by a definition that is no
    -- lambda. An inferred annotation (`two : _`) with no checker's type is
    -- neither.
    local has_type = signature.known and not (is_annotation and get(def, "whole", "") == "_" and not has_key(def, "typed"))
    local is_lambda = get(def, "lambda", false)
    local is_function = signature.is_function or (not signature.known and is_lambda)
    local is_value = (has_type and not signature.is_function) or (not signature.known and not is_lambda)
    local taken_kind = get(TAKEN_WORDS, name, "")
    if taken_kind == "roc" and is_function then
      local result_parts = split(signature.result, ".")
      local returns_the_word = signature.is_function and result_parts[#result_parts] == camel_case({name})
      if not returns_the_word then report(path, line, "taken_word", "`" .. full_name .. "` is Roc's own word, `" .. name .. "`, for a function that gives no `" .. camel_case({name}) .. "`: name it for the domain's thing (`element`, `append`), and keep Roc's words for Roc") end
    end
    if taken_kind == "cardinal" and is_value then report(path, line, "taken_word", "`" .. full_name .. "` is a number's name for a value: it says how many, not what it holds") end
    -- An effect so named is `state_effect`'s, and a function so named that
    -- gives a `Bool` is `question`'s, so this check takes the rest.
    if not has_bang and #ws == 1 and is_participle(name) and is_function and not returns_bool then report(path, line, "bare_participle", "`" .. full_name .. "` is a function named by a bare past participle, which reads as a `Bool`: name it by a noun for what it gives (`user_by_name`) or a verb for what it does (`add_user`)") end
    -- An effect is `noun_effect`'s, and a result of `_` says nothing.
    if not has_bang and has_key(PREPOSITIONS, ws[1]) and is_function and not returns_bool and signature.result ~= "_" and not reads_as_maker(def, ws, signature, FILE_FACTS, PREPOSITIONS) then
      local gives = signature.known and ("gives `" .. signature.result .. "`" .. signature.source_note) or "has no annotation to show it makes its own type"
      report(path, line, "phrase_name", "`" .. full_name .. "` starts with a preposition, `" .. ws[1] .. "`, and " .. gives .. ": bare, it reads as a yes or no and hides what it gives; name the thing (`trees_in_view`, not `in_view`)")
    elseif signature.known then
      judge_list_name({path = path, line = line, name = full_name, kind = signature.is_function and "function" or "value", is_function = signature.is_function, first = signature.first, gives = signature.result})
    end
    if not has_bang and is_function and has_key(WORLD_VERBS, ws[1]) and not get(def, "has_doc", false) then report(path, line, "world_verb", "`" .. full_name .. "` is pure, and `" .. ws[1] .. "` claims to act on something outside: name it for the value it gives (`tree_shapes`, not `draw_tree`)") end
    local argument_types = get(def, "argument_types", {})
    local parameters = get(def, "parameters", {})
    for k = 1, #parameters do
      local parameter = parameters[k]
      local parameter_type = k <= #argument_types and argument_types[k] or get(parameter, "typed", "")
      if parameter.name ~= "" then judge_list_name({path = path, line = parameter.line, name = parameter.name, kind = "parameter", is_function = false, first = "", gives = parameter_type}) end
    end
  end
  -- `worker_name` is judged at the definition, which knows whether it calls
  -- itself, and at an annotation only when no definition follows it, as
  -- with a hosted effect.
  local annotation_alone = is_annotation and not get(def, "has_def", false)
  local helper_suffix = has_key(HELPER_SUFFIXES, last_word) and (annotation_alone or not is_annotation)
  local recursion_suffix = has_key(RECURSION_SUFFIXES, last_word) and get(def, "recurses", false)
  if #ws > 1 and (helper_suffix or recursion_suffix) then report(path, line, "worker_name", "`" .. full_name .. "` is named only as a helper, `_" .. last_word .. "`: name the worker for the parameter it adds (`_from` a start point, `_onto` an output built, `_with` other carried state) and give the entry function the plain name") end
  if is_annotation and not def.fn and def.whole ~= "_" then
    local annotated_type = def.whole
    if last_word == "count" and not has_key(INTEGER_TYPES, annotated_type) then report(path, line, "name_type", "`" .. full_name .. "` is a count of type `" .. annotated_type .. "`") end
    if last_word == "path" and annotated_type ~= "Str" and not contains(annotated_type, "Path") then report(path, line, "name_type", "`" .. full_name .. "` is a path of type `" .. annotated_type .. "`") end
    if annotated_type == "Bool" and (ws[1] == "no" or ws[1] == "not") and #ws > 1 then report(path, line, "negative", "`" .. full_name .. "` is a negative flag: `" .. full_name .. " = Bool.false` reads twice") end
  end
  if is_annotation and is_effect and def.fn then
    for argument in arguments do
      local union = get(argument, "union")
      local ref = get(argument, "ref")
      if union == nil and ref ~= nil then union = get(TAG_UNION_TYPES, ref) end
      if union ~= nil then
        local shown = (get(argument, "list") and "a list of " or "") .. "`[" .. join(union.names, ", ") .. "]`"
        if not union.payload and #union.names >= 2 and all_tags_past(union.names) then report(path, line, "command_tense", "`" .. full_name .. "` takes " .. shown .. ", an instruction named like an event: name its tags as commands (`Pinned` is `Pin`)") end
        local command_tags = {}
        local event_tags = {}
        for t in union.names do
          local w = tag_first_word(t)
          if is_participle(w) then push(event_tags, t) elseif has_key(VERBS, w) then push(command_tags, t) end
        end
        if #command_tags > 0 and #event_tags > 0 then report(path, line, "mixed_tense", "`" .. full_name .. "` takes " .. shown .. ": commands (" .. join(command_tags, ", ") .. ") beside events (" .. join(event_tags, ", ") .. "): name what it is told to do in one mood") end
      end
    end
  end
  if is_annotation and #ws > 2 and ws[1] == "is" and ws[2] == "not" then report(path, line, "negative", "`" .. full_name .. "` asks a negative: `!" .. full_name .. "(...)` reads twice") end
  return nil
end

-- A plant's name relative to PLANTS. A typed plant is a CamelCase module
-- (`typed/QuestionInferred.roc`), as is a plant another imports
-- (`QuietDocShelf.roc`), and its name is given in snake case
-- (`question_inferred.roc`).
function plant_name(path)
  local name = sub(path, #PLANTS + 1)
  if starts_with(name, "typed/") then name = sub(name, 7) end
  if re_find(name, "^[A-Z]") ~= nil then return snake_case_plant_name(name) end
  return name
end
