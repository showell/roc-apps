## rewrite_module_calls_as_method_calls, in Roc on Shale's platform (roc_platform/README.md):
## old-style Roc calls rewritten as method calls, `List.map(xs, f)` as
## `xs.map(f)`, wherever the compiler agrees it is the same call. Its
## launcher, cookbook/roc/rewrite_module_calls_as_method_calls.shale, is how it is run, and
## its header says which calls turn, how the types and the compiler judge
## them, and the arguments. check_goldens.shale, beside this, holds its
## output and the files it rewrites on the plants to its goldens.
##
##   cd <repo> && shale cookbook:roc/rewrite_module_calls_as_method_calls.shale glob=app/*.roc root=app/main.roc
##   cd <repo> && shale cookbook:roc/rewrite_module_calls_as_method_calls.shale glob=app/*.roc root=app/main.roc apply=1
##   shale cookbook:roc/rewrite_module_calls_as_method_calls.shale self_test=1
##
## This root reads the arguments, globs the files and prints the verdict.
## Run is the rewrite: the trees (Source), the candidates (Candidates, with
## Literals), the types (Types, judging by Judging and Peers), the trial and
## the confirm (Trial, by Compiler, writing by Rewritten), all on a sheet
## (Tally); Sweep is what the rewrites leave for a sweep; SelfTest runs the
## plants of Plants. Imports, Text and RocTypes are the cookbook's Roc
## library, cookbook/roc/lib/.
## budget: steps=900000000 output_bytes=4000000 arena_bytes=3000000000 kept_bytes=256000000 changes=100000 change_bytes=2000000000 programs=1100 roc_cpu_ms=3600000
app [main!] { lib: "../lib/main.roc", pf: platform "../../../roc_platform/platform/main.roc" }

import pf.Shale
import Compiler
import Run
import SelfTest
import Sweep
import lib.Text

main! : {} => {}
main! = |{}| {
	self_test = Shale.arg!("self_test", "") == "1"
	apply = self_test or Shale.arg!("apply", "") == "1"
	use_types = either!("types", Shale.arg!("types", "1"))
	count_only = !(either!("trials", Shale.arg!("trials", "1")))
	if count_only and apply and !(self_test) {
		Shale.fail!("rewrite_module_calls_as_method_calls: trials=0 writes nothing, so not with apply=1")
	}
	if count_only and !(use_types) {
		Shale.fail!("rewrite_module_calls_as_method_calls: trials=0 counts by the types, so not with types=0")
	}
	peers = Shale.arg!("peers", "")
	if peers != "" and peers != "undo" {
		Shale.fail!("rewrite_module_calls_as_method_calls: peers=${peers}: peers takes only `undo`")
	}
	undo = peers == "undo"
	if undo and (count_only or !(use_types)) {
		Shale.fail!("rewrite_module_calls_as_method_calls: peers=undo judges peers by the types and turns them in trials, so not with trials=0 or types=0 (trials=0 counts them as peer_receiver)")
	}
	plant = Shale.arg!("plant", "")
	if plant != "" and !(self_test) {
		Shale.fail!("rewrite_module_calls_as_method_calls: plant= is for the self-test's child runs, and needs self_test=1")
	} else if self_test and plant != "" {
		SelfTest.plant!(plant, Compiler.roc_bin!(Bool.True))
	} else if self_test {
		SelfTest.run!({})
	} else {
		rewrite!({ apply, use_types, count_only, undo })
	}
}

## Whether the argument `name`, given as `value`, is 1 (its default) rather
## than 0; any other value stops the run.
either! : Str, Str => Bool
either! = |name, value| {
	if value != "0" and value != "1" {
		Shale.fail!("rewrite_module_calls_as_method_calls: ${name}=${value}: ${name} must be 1 (the default) or 0")
	}
	value == "1"
}

## The rewrite of the glob's files, its summary, its findings and the
## summary again (cookbook/findings.shale), then the files no root reaches.
rewrite! : { apply : Bool, use_types : Bool, count_only : Bool, undo : Bool } => {}
rewrite! = |mode| {
	root = Shale.arg!("root", "")
	from = U64.from_str(Shale.arg!("from", "1")) ?? 1
	types_from = Shale.arg!("types_from", "")
	paths = SelfTest.paths_of_globs!(Shale.arg!("glob", "**/*.roc"))
	roc = Compiler.roc_bin!(Bool.False)
	total = Run.all!({
		paths,
		root,
		from,
		use_types: mode.use_types,
		types_roc: roc,
		judge: { roc, apply: mode.apply },
		count_only: mode.count_only,
		types_from,
		undo: mode.undo,
	})
	made = Sweep.made!(total.tally)
	summary = "${summary_of(mode, total, from, paths.len())}, made_unused_import ${made.unused.to_str()}, made_long_line ${made.long.to_str()}"
	kinds = total.tally.kind_counts
		.sort_with(|(first, _), (second, _)| Text.byte_order(first, second))
		.map(|(kind, count)| "${kind} ${count.to_str()}")
	judged = if mode.use_types {
		["types: the first arguments of the candidates the types judged: ${Str.join_with(kinds, ", ")}"]
	} else {
		[]
	}
	print_findings!(summary, total.lines.concat(made.lines).concat(judged))
	for path in total.unreached {
		Shale.print!("not reached by ${root}: ${path}")
	}
	if !(total.unreached.is_empty()) {
		Shale.fail!("rewrite_module_calls_as_method_calls: no root reaches ${total.unreached.len().to_str()} file(s) with a call to try, listed above: their calls were left untried")
	}
}

## The summary line's counts, and where to go on from when the programs ran
## out.
summary_of : { apply : Bool, use_types : Bool, count_only : Bool, undo : Bool }, Run.Total, U64, U64 -> Str
summary_of = |mode, total, from, file_count| {
	count = |n| n.to_str()
	programs = count(total.tally.programs_used)
	counted = if mode.count_only {
		"count: files ${count(total.files)}, candidates ${count(total.candidates)}, proven ${count(total.rewritten)}, unsettled ${count(total.unsettled)}, left ${count(total.left)}, peer_receiver ${count(total.peers)}, programs ${programs} (no trials: the types alone decide)"
	} else {
		undone = if mode.undo {
			"peers=undo, "
		} else {
			""
		}
		dry = if mode.apply {
			""
		} else {
			"dry run: "
		}
		checks = "checks: ${count(total.tally.confirms)} confirming proven calls, ${count(total.tally.rounds)} in trial rounds"
		"${undone}${dry}files ${count(total.files)}, candidates ${count(total.candidates)}, rewritten ${count(total.rewritten)}, left ${count(total.left)}, peer_receiver ${count(total.peers)}, programs ${programs} (${checks})"
	}
	if total.next > 0 {
		"${counted}, in files ${count(from)} to ${count(total.next - 1)} of ${count(file_count)}: a run holds ${count(Run.max_programs)} programs; go on with from=${count(total.next)}"
	} else {
		counted
	}
}

## The summary, each finding, then the summary again (cookbook/findings.shale):
## the first copy survives a cut output, and a gate reads the last line.
print_findings! : Str, List(Str) => {}
print_findings! = |summary, finding_lines| {
	Shale.print!(summary)
	if !(finding_lines.is_empty()) {
		for line in finding_lines {
			Shale.print!(line)
		}
		Shale.print!(summary)
	}
}

## The tree's own Roc, which judges each rewrite: `roc check` and its errors,
## `roc test`'s summary, and `roc fmt`. The Roc is the one the tree pins
## (its `roc-version`), since method lookup has changed
## between nightlies. Each refusal is worded as the Shale original's.
import pf.Shale
import Candidates
import lib.Text

Compiler := [].{

	## What `roc check` said: whether it passed, each error it reported (a
	## warning counts, since a warning fails the build), and its output.
	Check : { ok : Bool, errors : List(Candidates.CheckError), out : Str }

	## The OS path of the tree's Roc: the pin in `roc-version` where the run
	## starts, or in a self-test the pin pinned.shale names.
	roc_bin! : Bool => Str
	roc_bin! = |self_test| {
		pin = if self_test {
			Compiler.self_test_pin!({})
		} else if Shale.exists!("roc-version") {
			Text.trim(Compiler.read_text!("roc-version"))
		} else {
			Compiler.stop!("rewrite_module_calls_as_method_calls: no roc-version where the run starts: run it from the repository's root, whose roc-version pins its Roc")
		}
		roc = "tools:roc-${pin}/roc"
		if !(Shale.exists!(roc)) {
			Compiler.stop!("rewrite_module_calls_as_method_calls: no ${roc}: install Roc ${pin} into ~/tools/roc-${pin}/ (roc-apps's toolchain release carries it)")
		} else {
			real = Shale.real!(roc)
			said = Shale.run!({ argv: [real, "version"], keep: Whole(4000), timeout_ms: 30000, want: "" })
			if Text.trim(said.out) == "Roc compiler version ${pin}" {
				real
			} else {
				Compiler.stop!("rewrite_module_calls_as_method_calls: ${roc} says ${Text.trim(said.out)}, not Roc compiler version ${pin}")
			}
		}
	}

	## The pin every recipe's self-test runs, from cookbook/roc/pinned.shale's
	## `SELF_TEST_PIN`, its one home.
	self_test_pin! : {} => Str
	self_test_pin! = |{}| {
		pinned = Compiler.read_text!("cookbook:roc/pinned.shale")
		match Shale.re_find!(pinned, "\nconst SELF_TEST_PIN = \"([^\"]+)\"") {
			Found(matched) => matched.groups.first() ?? ""
			NoMatch => Compiler.stop!("rewrite_module_calls_as_method_calls: cookbook/roc/pinned.shale names no SELF_TEST_PIN")
		}
	}

	## `roc check` on the app `root`.
	check! : Str, Str => Compiler.Check
	check! = |roc, root| {
		ran = roc_run!([roc, "check", Shale.real!(root)], Whole(64000000), 900000)
		var $errors = []
		for said in Text.lines(ran.out) {
			# An error's heading is `── ✗ <title> ───── <path>:<line>:<col>`, and
			# a warning's has `●` in place of `✗`.
			if said.starts_with("── ✗ ") or said.starts_with("── ● ") {
				match Shale.re_find!(said, " ([^ ]+):([0-9]+):([0-9]+)$") {
					Found(at) => {
						heading = Str.from_utf8_lossy(said.to_utf8().drop_first(11))
						title = Text.trim(Text.before_first(heading, " ─"))
						place = at.groups.get(0) ?? ""
						line = U64.from_str(at.groups.get(1) ?? "") ?? 0
						$errors = $errors.append({ title, file: Text.basename(place), line })
					}
					NoMatch => {}
				}
			}
		}
		{ ok: ran.code == 0, errors: $errors, out: ran.out }
	}

	## The last line of `roc test` on `root`, without its time, and the exit
	## code: what a rewrite may not change.
	test_summary! : Str, Str => Str
	test_summary! = |roc, root| {
		ran = roc_run!([roc, "test", Shale.real!(root)], Whole(64000000), 900000)
		last = Text.lines(Text.trim(ran.out)).last() ?? ""
		summary = match Shale.re_find!(last, "^(.*) in [0-9.]+ ms\\.") {
			Found(matched) => matched.groups.first() ?? ""
			NoMatch => last
		}
		"${summary} (exit ${ran.code.to_str()})"
	}

	## Whether `roc fmt` finds the file as it would lay it out.
	is_fmt_clean! : Str, Str => Bool
	is_fmt_clean! = |roc, path| roc_run!([roc, "fmt", "--check", Shale.real!(path)], Tail(20), 120000).code == 0

	## The file laid out by `roc fmt`.
	fmt! : Str, Str => {}
	fmt! = |roc, path| {
		_ran = roc_run!([roc, "fmt", Shale.real!(path)], Tail(20), 120000)
		{}
	}

	## The first non-blank line of `out`, for a failure that names no error
	## (a crash).
	first_line_said : Str -> Str
	first_line_said = |out|
		match Text.lines(out).find_first(|said| Text.trim(said) != "") {
			Ok(said) => Text.trim(said)
			Err(_) => "nothing said"
		}

	## A file's text; a missing one stops the run.
	read_text! : Str => Str
	read_text! = |path|
		match Shale.read!(path) {
			Found(text) => text
			Missing => Compiler.stop!("rewrite_module_calls_as_method_calls: no ${path}")
		}

	## The run stopped, saying why; the text is never used.
	stop! : Str => Str
	stop! = |why| {
		Shale.fail!(why)
		""
	}
}

## The rewrite of a list of files, each judged on its root: their trees read,
## their candidates found, the types read of each root and judging them, then
## the confirm and the trials in the files' order, as far as the run's
## programs allow (rewrite_module_calls_as_method_calls.shale's `run_all`).
import pf.Shale
import Candidates
import Compiler
import Judging
import Peers
import Source
import Tally
import Trial
import Types
import lib.RocTypes
import lib.Text

Run := [].{

	## What to rewrite: the files, the roots (`a.roc,b.roc`), the file to
	## start at (counted from 1), whether the types are read, the Roc
	## roc-types' pin must match, the Roc that judges, and the modes:
	## `count_only` (`trials=0`), `types_from` and `undo` (`peers=undo`).
	Asked : {
		paths : List(Str),
		root : Str,
		from : U64,
		use_types : Bool,
		types_roc : Str,
		judge : Trial.Judge,
		count_only : Bool,
		types_from : Str,
		undo : Bool,
	}

	## What the run did: its counts, a line for each call that stays (and
	## each peer method call, and under `trials=0` each call the types left
	## unsettled), the file to go on from when the programs ran out (else 0),
	## its sheet, and the files with a call to try that no root reaches.
	Total : {
		files : U64,
		candidates : U64,
		rewritten : U64,
		left : U64,
		unsettled : U64,
		peers : U64,
		lines : List(Str),
		next : U64,
		tally : Tally.Sheet,
		unreached : List(Str),
	}

	## The programs a run may start, Shale's own budget unset.
	max_programs : U64
	max_programs = 1000

	## Rewrites the files of `asked.paths` from the `asked.from`th on.
	all! : Run.Asked => Run.Total
	all! = |asked| {
		if asked.use_types and asked.root == "" {
			Shale.fail!("rewrite_module_calls_as_method_calls: the types need root=, an app importing every file of the glob: roc-types reads an app, and a type module only through one. Give root=, or types=0 to run without them, every candidate tried")
		}
		if !(asked.use_types) and asked.root.contains(",") {
			Shale.fail!("rewrite_module_calls_as_method_calls: root=${asked.root}: several roots need the types, which say which root reaches a file; give one root, or run with the types")
		}
		roots = if asked.root == "" {
			[]
		} else {
			match RocTypes.root_apps("rewrite_module_calls_as_method_calls", asked.root) {
				Ok(apps) => apps
				Err(why) => {
					Shale.fail!(why)
					[]
				}
			}
		}
		if asked.types_from != "" and roots.len() != 1 {
			Shale.fail!("rewrite_module_calls_as_method_calls: types_from= holds one root's types, and root= names ${roots.len().to_str()}")
		}
		files = parsed!(asked.paths)
		aliases = if asked.use_types {
			Peers.declared_aliases(files.values())
		} else {
			Dict.empty()
		}
		symmetric = match Peers.symmetric_table(Compiler.read_text!(symmetric_functions)) {
			Ok(names) => names
			Err(why) => {
				Shale.fail!(why)
				Set.empty()
			}
		}
		var $judged = []
		for path in asked.paths.drop_first(asked.from - 1) {
			file = match files.get(path) {
				Ok(found) => found
				Err(_) => {
					Shale.fail!("rewrite_module_calls_as_method_calls: roc-ast gave no tree for ${path}")
					Source.of(path, "", [])
				}
			}
			cands = Candidates.of_file(file, module_of_path(path))
			method_calls = if asked.use_types {
				Peers.method_calls(file, symmetric)
			} else {
				[]
			}
			open = cands.any(|cand| cand.why_left == "") or !(method_calls.is_empty())
			$judged = $judged.append({ path, file, real: Shale.real!(path), root: roots.first() ?? path, cands, counts: [], open, unreached: Bool.False, method_calls, peers: [] })
		}
		# The types judge only open candidates, so with none, roc-types never runs.
		read_types = asked.use_types and $judged.any(|one| one.open)
		typing = { by_region: Dict.empty(), aliases, symmetric }
		typed = if read_types {
			with_types!(asked, roots, typing, $judged)
		} else {
			{ tally: Tally.ran(Tally.empty), judged: $judged }
		}
		unreached = typed.judged.keep_if(|one| one.unreached).map(|one| one.path)
		in_mode = typed.judged.map(|one| as_undo(asked.undo, one))
		ready = in_mode.map(with_open_count)
		total = {
			files: 0,
			candidates: 0,
			rewritten: 0,
			left: 0,
			unsettled: 0,
			peers: 0,
			lines: [],
			next: 0,
			tally: typed.tally,
			unreached,
		}
		if asked.count_only {
			counted_by_types(total, ready)
		} else {
			tried!(asked, total, ready)
		}
	}
}

## A file as the run judges it: its tree, real path and root, its
## candidates, the kinds of type the types found, whether the types must
## judge it (`open`: a module call to try, or with the types a symmetric
## method call), whether no root reaches it, its symmetric method calls and
## its peer method calls; then how many calls are open and whether all are
## proven.
Judged : {
	path : Str,
	file : Source.File,
	real : Str,
	root : Str,
	cands : List(Candidates.Candidate),
	counts : Judging.KindCounts,
	open : Bool,
	unreached : Bool,
	method_calls : List(Peers.MethodCall),
	peers : List(Candidates.Candidate),
}

Ready : {
	path : Str,
	file : Source.File,
	root : Str,
	cands : List(Candidates.Candidate),
	counts : Judging.KindCounts,
	unreached : Bool,
	peers : List(Candidates.Candidate),

## The module a .roc file holds, its name without `.roc`.
module_of_path : Str -> Str
module_of_path = |path| Text.basename(path).drop_suffix(".roc")

## Every file's tree from Roc's parser and its text, by path: one program.
parsed! : List(Str) => Dict(Str, Source.File)
parsed! = |paths| {
	var $files = Dict.empty()
	if !(paths.is_empty()) {
		for tree in Shale.roc_tree!(paths) {
			if tree.err != "" {
				Shale.fail!("rewrite_module_calls_as_method_calls: roc-ast: ${tree.path}: ${tree.err}")
			}
			$files = $files.insert(tree.path, Source.of(tree.path, Compiler.read_text!(tree.path), tree.nodes))
		}
	}
	$files
}

## The types read of each root, one at a time, each file's root the first
## whose app reaches it, its open candidates judged by that root's types on
## the way. A file with a call to try that no root reaches is `unreached`,
## and the run goes on without it.
with_types! : Run.Asked, List(Str), Judging.Typing, List(Judged) => { tally : Tally.Sheet, judged : List(Judged) }
with_types! = |asked, roots, typing, judged| {
	only = if asked.types_from == "" {
		Set.empty()
	} else {
		var $reals = Set.empty()
		for path in asked.paths {
			$reals = $reals.insert(Shale.real!(path))
		}
		$reals
	}
	var $tally = Tally.ran(Tally.empty)
	var $judged = judged
	var $found = Set.empty()
	for root in roots {
		if asked.types_from == "" {
			$tally = Tally.ran($tally)
		}
		typed = Types.of_root!({ root, roc: asked.types_roc, types_from: asked.types_from, only })
		if !($tally.typed_roots.contains(root)) {
			$tally = { ..$tally, typed_roots: $tally.typed_roots.append(root) }
		}
		types_here = { ..typing, by_region: typed.by_region }
		var $judged_here = []
		for one in $judged {
			if one.open and !($found.contains(one.path)) and typed.modules.contains(one.real) {
				$found = $found.insert(one.path)
				by_types = Judging.by_types(one.counts, one.file, one.cands, one.real, types_here)
				peers = one.peers.concat(Judging.method_peers(one.file, one.method_calls, one.real, types_here))
				$judged_here = $judged_here.append({ ..one, root, counts: by_types.counts, cands: by_types.cands, peers })
			} else {
				$judged_here = $judged_here.append(one)
			}
		}
		$judged = $judged_here
	}
	var $marked = []
	for one in $judged {
		$marked = $marked.append({ ..one, unreached: one.open and !($found.contains(one.path)) })
	}
	{ tally: $tally, judged: $marked }
}

## Under `peers=undo`, a file's candidates are its peer method calls alone.
as_undo : Bool, Judged -> Judged
as_undo = |undo, one|
	if undo {
		{ ..one, cands: one.peers, peers: [], counts: [] }
	} else {
		one
	}

with_open_count : Judged -> Ready
with_open_count = |one| {
	open_cands = one.cands.keep_if(|cand| cand.why_left == "")
	all_proven = !(open_cands.is_empty()) and !(one.unreached) and open_cands.all(|cand| cand.proven)
	{
		path: one.path,
		file: one.file,
		root: one.root,
		cands: one.cands,
		counts: one.counts,
		unreached: one.unreached,
		peers: one.peers,
		open_count: open_cands.len(),
		all_proven,
	}
}

## The file's peer method calls added to the findings, a `path:line:
## peer_receiver: why` each, and to their count.
with_peer_findings : Run.Total, Ready -> Run.Total
with_peer_findings = |total, one| {
	said = one.peers.map(
		|cand| {
			why = "`.${cand.fn}` is symmetric and its receiver and first argument are both ${cand.module_name}s, so neither is the subject"

## programs it may take are left: `Trial.max_rounds` checks, a test and,
## under `apply`, two `roc fmt`s, and the first file on a root also the
## baseline's check and test.

		$place = $place + 1

			[]
		} else {
			one.cands
		}

			# root has none yet, the confirm's checks and test, and this file's
			# own trial, should the confirm set it aside. Under `apply` each file
			# in the confirm also takes two `roc fmt`s, so only as many join as
			# fit.
			baseline_programs = if $baselines.contains(one.root) {
				0
			} else {
				2
			}
			taken = $total.tally.programs_used + baseline_programs + Trial.max_rounds + 1 + per_file
			var $files = []
			for other in ready.drop_first($place - 1) {
				fits = !(judge.apply) or 2 * ($files.len() + 1) + taken <= Run.max_programs
				if other.all_proven and other.root == one.root and !($total.tally.confirmed.contains(other.path)) and fits {
					$files = $files.append({ path: other.path, file: other.file, cands: other.cands })
				}
			}
			if taken > Run.max_programs or $files.is_empty() {
				return { ..$total, next: asked.from + $place - 1 }
			}
			based = baseline!(judge, $total.tally, $baselines, one.root)
			$baselines = based.baselines
			$total = { ..$total, tally: Trial.confirm!(judge, based.tally, one.root, $files, based.summary) }
		}
		kept_by_confirm = $total.tally.confirmed.get(one.path) ?? 0
		if !(one.unreached) and one.open_count > 0 and kept_by_confirm == 0 and $total.tally.programs_used + per_file > Run.max_programs {
			return { ..$total, next: asked.from + $place - 1 }
		}
		if !(one.unreached) {
			$total = { ..$total, files: $total.files + 1 }
		}
		$total = with_counts($total, one)
		$total = with_peer_findings($total, one)
		if !(cands.is_empty()) {
			var $cands = Candidates.without_lines(cands)
			$kept : U64
			var $kept = kept_by_confirm
			if $kept == 0 and one.open_count > 0 {
				based = baseline!(judge, $total.tally, $baselines, one.root)
				$baselines = based.baselines
				done = Trial.try_file!(judge, based.tally, one.file, $cands, one.root, based.summary)
				$total = { ..$total, tally: done.tally }
				$cands = done.cands
				$kept = done.turned
			}
			left = $cands.keep_if(|cand| cand.why_left != "")
			$total = {
				..$total,
				candidates: $total.candidates + $cands.len(),
				rewritten: $total.rewritten + $kept,
				left: $total.left + left.len(),
				lines: $total.lines.concat(left.map(|cand| left_line(one.path, cand))),
			}
		}
	}
	$total
}

## A callee `Module.fn` split, when it is one: a capital and letters and
## digits, a `.`, then a small letter, letters, digits and `_`, and an
## optional last `!`.
module_call : Str -> Try({ module_name : Str, fn : Str }, [NoModuleCall])
module_call = |raw| {
	parts = raw.split_first(".") ?? { before: "", after: "" }
	module_name = parts.before.to_utf8()
	fn = parts.after.to_utf8()
	fn_body = if (fn.last() ?? 0) == '!' {
		fn.drop_last(1)
	} else {
		fn
	}
	is_module = match module_name.first() {
		Ok(first) => is_capital(first) and module_name.all(is_alphanumeric)
		Err(_) => Bool.False
	}
	is_fn = match fn_body.first() {
		Ok(first) => first >= 'a' and first <= 'z' and fn_body.all(|byte| is_alphanumeric(byte) or byte == '_')
		Err(_) => Bool.False
	}
	if is_module and is_fn {
		Ok({ module_name: parts.before, fn: parts.after })
	} else {
		Err(NoModuleCall)
	}
}

## them by region: a call whose first argument's type is the module's own is
## proven, one of a concrete other type stays, a symmetric call of peers
## stays a module call, and the rest are left to the trial. With
## `peers=undo`, the peer method calls are found here too.

	## kind was first counted.

	## the aliases the files declare, and the symmetric functions.

## A type's head: a type variable (`a`, `_a`, `a where [...]`), or its name
## (`List` of `List(U8)`, `Access.Viewer` of `Access.Viewer(Topic.At)`), a
## function's (an arrow outside every bracket) `function`, `{...}`'s
## `record`, `[...]`'s `tag union` and `(a, b)`'s `tuple`.
Head : [Variable, Named(Str)]

type_head : Str -> Head
type_head = |type_text| {
	bytes = type_text.to_utf8()
	first = bytes.first() ?? 0
	if Peers.has_top_arrow(bytes) {
		Named("function")
	} else if first == '{' {
		Named("record")
	} else if first == '[' {
		Named("tag union")
	} else if first == '(' {
		Named("tuple")
	} else if first < 'A' or first > 'Z' {
		Variable
	} else {
		length = bytes.find_first_index(|byte| !(is_head_byte(byte))) ?? bytes.len()
		Named(Str.from_utf8_lossy(bytes.take_first(length)))
	}
}

is_head_byte = |byte|
	(byte >= 'A' and byte <= 'Z') or (byte >= 'a' and byte <= 'z') or (byte >= '0' and byte <= '9') or byte == '_' or byte == '.'

## `word` after its indefinite article: "an" before a vowel sound. A U
## usually sounds like "you" (U8, User), so it takes "an" only in `Un...`
## (Unvalidated).
with_article : Str -> Str
with_article = |word| {
	first = word.to_utf8().first() ?? 0
	vowel = first == 'A' or first == 'E' or first == 'I' or first == 'O' or first == 'a' or first == 'e' or first == 'i' or first == 'o'

		"an ${word}"
	} else {
		"a ${word}"
	}

## with what the types say. A first argument whose type has the module's own
## name, bare, proves the call: roc-types writes a type by the name it is
## declared with, so a qualified head never does, even one whose last
## segment names the module (`Type.Game` is Type's, not Game's).

## Which calls are peer calls (peer_calls.shale). A method call names a
## subject, so a symmetric function, one where no argument is the subject, is
## called through its module: `U64.max(a, b)`, `Str.concat(a, b)`,
## `List.map2(xs, ys, f)`. The symmetric functions are a list of names,
## cookbook/roc/symmetric_functions.txt, and a call of one is a peer call
## when its first two arguments' types have one head, as roc-types writes
## them at the call with each alias read through to what it names (a `Hue :
## U32` is a `U32`). A function off the list (`starts_with`, a unit's `plus`)
## keeps its method call, whatever its arguments.
import Source
import lib.Text

Peers := [].{

	## Whether two arguments are peers, values of one type: `type` is the
	## first's type with its aliases read through, `module` that type's head.
	Verdict : { peers : Bool, type : Str, module_name : Str }

	## A symmetric method call that passes an argument, `x.max(y)`: its node,
	## its receiver's node, its first argument's, the function, and whether
	## every argument has a span, so its text can be moved.
	MethodCall : { node : U64, receiver : U64, first : U64, fn : Str, spanned : Bool }

	## Every alias the files declare (`Hue : U32`, never `:=` or `::`), by the
	## name roc-types gives it, to the text of the type it names. An alias
	## with parameters is left out: reading it through would need its
	## arguments bound. Where two files declare one name, the later file by
	## path wins.
	declared_aliases : List(Source.File) -> Dict(Str, Str)
	declared_aliases = |files| {
		var $out = Dict.empty()
		for file in files.sort_with(|first, second| Text.byte_order(first.path, second.path)) {
			for (node, place) in file.nodes.map_with_index(|node, index| (node, index + 1)) {
				if node.kind == "s-type-decl" and node.kids.len() == 2 {
					header = Source.kid_node(file, node, 1)
					target = Source.kid_node(file, node, 2)
					if is_bare_header(file, header) and header.has_span and target.has_span {
						between = Text.trim(Source.text_between(file, header.to, target.from))
						if between == ":" {
							$out = $out.insert(declared_name(file, place), Source.text_of(file, target))
						}
					}
				}
			}
		}
		$out
	}

	## Whether two arguments, of the types `first` and `second` as roc-types
	## gives them (`""` where it gives none), are peers: both types, their
	## aliases read through, have one head to call through (`List(U64)` and
	## `List(Str)` are peers to `map2`).
	verdict : Str, Str, Dict(Str, Str) -> Peers.Verdict
	verdict = |first, second, aliases|
		if first == "" or second == "" {
			{ peers: Bool.False, type: "", module_name: "" }
		} else {
			resolved = resolved_type(first, aliases, 0)
			module_name = Peers.module_of(resolved)
			peers = module_name != "" and module_name == Peers.module_of(resolved_type(second, aliases, 0))
			{ peers, type: resolved, module_name }
		}

	## The module a value of the type `type_text` takes its methods from, the
	## type's head (`List` for `List(U8)`, `Shelf.Book`), or `""` for a type
	## with none to call through: a type variable, a record, a tuple, a tag
	## union or a function.
	module_of : Str -> Str
	module_of = |type_text| {
		bytes = type_text.to_utf8()
		if Peers.has_top_arrow(bytes) {
			""
		} else {
			length = type_name_length(bytes, 0)
			Str.from_utf8_lossy(bytes.take_first(length))
		}
	}

	## Whether an arrow, `->` or `=>`, stands outside every bracket.
	has_top_arrow : List(U8) -> Bool
	has_top_arrow = |bytes| {
		$depth : I64
		var $depth = 0
		var $place = 0
		while $place + 1 < bytes.len() {
			byte = bytes.get($place) ?? 0
			if byte == '(' or byte == '[' or byte == '{' {
				$depth = $depth + 1
			}
			if byte == ')' or byte == ']' or byte == '}' {
				$depth = $depth - 1
			}
			next = bytes.get($place + 1) ?? 0
			if $depth == 0 and next == '>' and (byte == '-' or byte == '=') {
				return Bool.True
			}
			$place = $place + 1
		}
		Bool.False
	}

	## The symmetric functions in `text`, symmetric_functions.txt: a set of
	## names, one a line, `#` lines its comments; or why a line is no name.
	symmetric_table : Str -> Try(Set(Str), Str)
	symmetric_table = |text| {
		var $names = Set.empty()
		for text_line in Text.lines(text) {
			word = Text.trim(text_line)
			if word != "" and !(word.starts_with("#")) {
				if !(is_function_name(word.to_utf8())) {
					return Err("peer_calls: symmetric_functions.txt: `${word}` is not a function's name")
				}
				$names = $names.insert(word)
			}
		}
		Ok($names)
	}

	## argument.
	method_calls : Source.File, Set(Str) -> List(Peers.MethodCall)
	method_calls = |file, symmetric| {
		var $calls = []
		for (node, place) in file.nodes.map_with_index(|node, index| (node, index + 1)) {
			if node.kind == "e-method-call" and node.kids.len() == 2 {
				receiver = Source.kid_node(file, node, 1)
				if receiver.kind == "receiver" and receiver.kids.len() == 1 {

					fn = if node.method.starts_with(".") {
						Str.from_utf8_lossy(node.method.to_utf8().drop_first(1))
					} else {
						node.method
					}
					if args.kind == "args" and !(args.kids.is_empty()) and symmetric.contains(fn) {
						value = Source.kid(receiver, 1)
						spanned = Source.node_at(file, value).has_span and args.kids.all(|arg| Source.node_at(file, arg).has_span)
						call = { node: place, receiver: value, first: args.kids.first() ?? 0, fn, spanned }
						$calls = $calls.append(call)
					}
				}
			}
		}
		$calls
	}

## The type `type_text` with every alias in it read through, its runs of
## space made one, up to 8 aliases deep.
resolved_type : Str, Dict(Str, Str), U64 -> Str
resolved_type = |type_text, aliases, depth| {
	bytes = type_text.to_utf8()
	var $out = []
	var $changed = Bool.False
	var $place = 0
	while $place < bytes.len() {
		length = type_name_length(bytes, $place)
		if length == 0 {
			$out = $out.append(bytes.get($place) ?? 0)
			$place = $place + 1
		} else {
			name = Str.from_utf8_lossy(bytes.sublist({ start: $place, len: length }))
			match aliases.get(name) {
				Ok(named) => {
					$out = $out.concat(named.to_utf8())
					$changed = Bool.True
				}
				Err(_) => {
					$out = $out.concat(name.to_utf8())
				}
			}
			$place = $place + length
		}
	}
	spaced = Str.from_utf8_lossy(one_space(Text.trim(Str.from_utf8_lossy($out)).to_utf8()))
	if $changed and depth < 8 {
		resolved_type(spaced, aliases, depth + 1)
	} else {
		spaced
	}
}

## The bytes with each run of white space made one space.
one_space : List(U8) -> List(U8)
one_space = |bytes| {
	var $out = []
	var $in_space = Bool.False
	for byte in bytes {
		if is_space(byte) {
			if !($in_space) {
				$out = $out.append(' ')
			}
			$in_space = Bool.True
		} else {
			$out = $out.append(byte)
			$in_space = Bool.False
		}
	}
	$out
}

is_space : U8 -> Bool
is_space = |byte| byte == ' ' or byte == '\t' or byte == '\n' or byte == '\r' or byte == 11 or byte == 12

## The length of the type name at byte `start`, as roc-types writes one
## (`U64`, `List`, `Shelf.Book`): a capital, then letters, digits and `_`,
## and again after each `.` that a capital follows; 0 where none starts.
type_name_length : List(U8), U64 -> U64
type_name_length = |bytes, start| {
	if !(Text.is_capital(bytes.get(start) ?? 0)) {
		return 0
	}
	var $end = word_end(bytes, start + 1)
	while (bytes.get($end) ?? 0) == '.' and Text.is_capital(bytes.get($end + 1) ?? 0) {
		$end = word_end(bytes, $end + 2)
	}
	$end - start
}

## The place past the letters, digits and `_` from `start` on.
word_end : List(U8), U64 -> U64
word_end = |bytes, start| {
	var $end = start
	while is_word_byte(bytes.get($end) ?? 0) {
		$end = $end + 1
	}
	$end
}

is_word_byte : U8 -> Bool
is_word_byte = |byte| Text.is_capital(byte) or Text.is_small_letter(byte) or Text.is_digit(byte) or byte == '_'

## Whether the bytes are a function's name: a small letter, then small
## letters, digits and `_`, and an optional last `!`.
is_function_name : List(U8) -> Bool
is_function_name = |bytes| {
	body = if (bytes.last() ?? 0) == '!' {
		bytes.drop_last(1)
	} else {
		bytes
	}
	match body.first() {
		Ok(first) => Text.is_small_letter(first) and body.all(|byte| Text.is_small_letter(byte) or Text.is_digit(byte) or byte == '_')
		Err(_) => Bool.False
	}
}

import lib.Tree

Source := [].{

	## A node of the file's tree: its kind and the attributes the rewrite
	## reads (`""` where the tree gives none), its span as byte offsets
	## (`from` its first byte, counted from 1, `to` one past its last, both 0
	## and `has_span` false for a node with no position), its kids and
	## parent as places counted from 1 (the root's parent 0), and its region
	## as the compiler writes it, `line:col:end_line:end_col`, by which its
	## type is found.
	Node : {
		kind : Str,
		raw : Str,
		name : Str,
		field : Str,
		method : Str,
		from : U64,
		to : U64,
		has_span : Bool,
		kids : List(U64),
		parent : U64,
		region : Str,
	}

	## A file: its path, its text (and its bytes), its nodes, and the byte at
	## which each line starts, counted from 1, with one more past the end.
	File : {
		path : Str,
		text : Str,
		bytes : List(U8),
		nodes : List(Source.Node),
		starts : List(U64),
	}

	## The file at `path`, of text `text`, as Roc's tree of it reads.
	of : Str, Str, List(Tree.Node) -> Source.File
	of = |path, text, tree_nodes| {
		bytes = text.to_utf8()
		starts = line_starts(bytes)
		nodes = tree_nodes.map(|tree_node| node_of(tree_node, starts))
		{ path, text, bytes, nodes, starts }
	}

	## The node at a place counted from 1.
	node_at : Source.File, U64 -> Source.Node
	node_at = |file, place|
		match file.nodes.get(place - 1) {
			Ok(node) => node
			Err(_) => crash "Source.node_at: ${file.path} has no node ${place.to_str()}"
		}

	## The place of a node's kid, the kids counted from 1.
	kid : Source.Node, U64 -> U64
	kid = |node, number|
		match node.kids.get(number - 1) {
			Ok(place) => place
			Err(_) => crash "Source.kid: a ${node.kind} with no kid ${number.to_str()}"
		}

	## The node that is a node's kid, the kids counted from 1.
	kid_node : Source.File, Source.Node, U64 -> Source.Node
	kid_node = |file, node, number| Source.node_at(file, Source.kid(node, number))

	## The line the byte at `offset` is on: the last line starting at or
	## before it, and line 1 for an offset before the first.
	line_of_offset : Source.File, U64 -> U64
	line_of_offset = |file, offset| {
		var $low = 0
		var $high = file.starts.len()
		while $low < $high {
			middle = ($low + $high) // 2
			if (file.starts.get(middle) ?? 0) <= offset {
				$low = middle + 1
			} else {
				$high = middle
			}
		}
		if $low == 0 {
			1
		} else {
			$low
		}
	}

	## The bytes from offset `from` up to, not including, `to`; none when
	## `to` is not past `from`.
	bytes_between : Source.File, U64, U64 -> List(U8)
	bytes_between = |file, from, to|
		if to > from and from > 0 {
			file.bytes.sublist({ start: from - 1, len: to - from })
		} else {
			[]
		}

	## The text from offset `from` up to, not including, `to`.
	text_between : Source.File, U64, U64 -> Str
	text_between = |file, from, to| Str.from_utf8_lossy(Source.bytes_between(file, from, to))

	## A node's text.
	text_of : Source.File, Source.Node -> Str
	text_of = |file, node| Source.text_between(file, node.from, node.to)

## The byte at which each line starts, counted from 1, and one more past the

	var $starts = [1]
	$at : U64
	var $at = 0
	for byte in bytes {
		$at = $at + 1
		if byte == '\n' {
			$starts = $starts.append($at + 1)
		}
	}

		}
	has_span = tree_node.line != 0
	from = if has_span {
		start_of(tree_node.line) + tree_node.column - 1
	} else {
		0
	}
	to = if has_span {
		start_of(tree_node.end_line) + tree_node.end_column - 1
	} else {
		0
	}
	starts_at = "${tree_node.line.to_str()}:${tree_node.column.to_str()}"
	ends_at = "${tree_node.end_line.to_str()}:${tree_node.end_column.to_str()}"

		kind: tree_node.kind,
		raw: Tree.attribute(tree_node, "raw"),
		name: Tree.attribute(tree_node, "name"),
		field: Tree.attribute(tree_node, "field"),
		method: Tree.attribute(tree_node, "method"),
		from,
		to,
		has_span,
		kids: tree_node.kids,
		parent: tree_node.parent,

	}
}

## The compiler's judgement of a rewrite: one file's open calls tried round
## by round until `roc check` passes (the trial), and the files whose calls
## the types all prove written at once and checked together (the confirm).
## A file ends as it began unless `apply` and its rewrite passed.
import pf.Shale
import Candidates
import Compiler
import Rewritten
import Source
import Tally
import lib.Text

Trial := [].{

	## The Roc that judges, and whether a rewrite that passes is kept.
	Judge : { roc : Str, apply : Bool }

	## A file the confirm writes: its path, its tree and its candidates.
	Proven : { path : Str, file : Source.File, cands : List(Candidates.Candidate) }

	## Tries one file's open candidates, round by round, until `roc check` on
	## `root` passes or nothing is left to try, then runs the test, which
	## must say `baseline`. Answers how many calls were turned, and the
	## candidates, each that stays with its `why_left`.
	try_file! : Trial.Judge, Tally.Sheet, Source.File, List(Candidates.Candidate), Str, Str => { tally : Tally.Sheet, cands : List(Candidates.Candidate), turned : U64 }
	try_file! = |judge, tally, file, cands, root, baseline| {
		var $tally = tally
		fmt_was_clean = if judge.apply {
			$tally = Tally.ran($tally)
			Compiler.is_fmt_clean!(judge.roc, file.path)
		} else {
			Bool.False
		}
		var $cands = cands
		var $round = 0
		while $round < max_rounds {
			$round = $round + 1
			open = Candidates.open_count($cands)
			if open == 0 {
				Shale.write!(file.path, file.text)
				return { tally: $tally, cands: $cands, turned: 0 }
			}
			written = Rewritten.written(file, Candidates.turning($cands))
			$cands = Candidates.with_lines($cands, written)
			Shale.write!(file.path, written.text)
			$tally = Tally.ran({ ..$tally, rounds: $tally.rounds + 1 })
			checked = Compiler.check!(judge.roc, root)
			if checked.ok {
				if fmt_was_clean {
					$tally = Tally.ran($tally)
					Compiler.fmt!(judge.roc, file.path)
				}
				$tally = Tally.ran($tally)
				summary = Compiler.test_summary!(judge.roc, root)
				if summary != baseline {
					why = "roc test changed: ${summary}, was ${baseline}; the file is put back"
					Shale.write!(file.path, file.text)
					return { tally: $tally, cands: Candidates.leave_open($cands, why), turned: 0 }
				}
				$tally = Tally.noted($tally, file, $cands)
				if !(judge.apply) {
					Shale.write!(file.path, file.text)
				}
				return { tally: $tally, cands: $cands, turned: open }
			}
			judged = judge_errors(file, written.line_of_node, checked.errors, $cands)
			$cands = judged.cands
			if !(judged.refused) and (judged.stray != NoStray or checked.errors.is_empty()) {
				why = match judged.stray {
					Stray(stray) => "roc check: ${stray.title} at ${stray.file}:${stray.line.to_str()}, on no rewritten call"
					NoStray => "roc check failed, naming no line: ${Compiler.first_line_said(checked.out)}"
				}
				Shale.write!(file.path, file.text)
				return { tally: $tally, cands: Candidates.leave_open($cands, "${why}; the file is put back"), turned: 0 }
			}
			$cands = Candidates.without_lines($cands)
		}
		Shale.write!(file.path, file.text)
		why = "still failing after ${max_rounds.to_str()} rounds; the file is put back"
		{ tally: $tally, cands: Candidates.leave_open($cands, why), turned: 0 }
	}

	## Confirms the files whose open calls are all proven: written with all
	## their calls turned, at once, and judged together by one `roc check` on
	## `root` and one `roc test`. When the check fails, the files its errors
	## name are set aside and the rest written again, up to `max_rounds`
	## checks; an error in no file of theirs (a caller's), a failure naming no
	## line, or a changed test sets them all aside. `confirmed` holds each
	## kept file's count of turned calls, and 0 for a file set aside, which
	## the trial then judges alone.
	confirm! : Trial.Judge, Tally.Sheet, Str, List(Trial.Proven), Str => Tally.Sheet
	confirm! = |judge, tally, root, files, baseline| {
		var $tally = tally
		var $remaining = []
		for proven in files {
			$tally = { ..$tally, confirmed: $tally.confirmed.insert(proven.path, 0) }
			fmt_was_clean = if judge.apply {
				$tally = Tally.ran($tally)
				Compiler.is_fmt_clean!(judge.roc, proven.path)
			} else {
				Bool.False
			}
			$remaining = $remaining.append({ proven, fmt_was_clean, turned: Candidates.open_count(proven.cands) })
		}
		var $round = 0
		while $round < max_rounds {
			$round = $round + 1
			if $remaining.is_empty() {
				return $tally
			}
			for one in $remaining {
				Shale.write!(one.proven.path, Rewritten.written(one.proven.file, Candidates.turning(one.proven.cands)).text)
			}
			$tally = Tally.ran({ ..$tally, confirms: $tally.confirms + 1 })
			checked = Compiler.check!(judge.roc, root)
			if checked.ok {
				for one in $remaining {
					if one.fmt_was_clean {
						$tally = Tally.ran($tally)
						Compiler.fmt!(judge.roc, one.proven.path)
					}
				}
				$tally = Tally.ran($tally)
				passed = Compiler.test_summary!(judge.roc, root) == baseline
				for one in $remaining {
					if passed {
						$tally = Tally.noted($tally, one.proven.file, one.proven.cands)
					}
					if !(passed) or !(judge.apply) {
						Shale.write!(one.proven.path, one.proven.file.text)
					}
					if passed {
						$tally = { ..$tally, confirmed: $tally.confirmed.insert(one.proven.path, one.turned) }
					}
				}
				return $tally
			}
			for one in $remaining {
				Shale.write!(one.proven.path, one.proven.file.text)
			}
			named = |one| checked.errors.any(|error| Text.basename(one.proven.path) == error.file)
			names_another = |error| !($remaining.any(|one| Text.basename(one.proven.path) == error.file))
			if checked.errors.is_empty() or checked.errors.any(names_another) {
				return $tally
			}
			$remaining = $remaining.drop_if(named)
		}
		$tally
	}

	## How many checks a file may take: each round leaves out the calls the
	## compiler refused, and the next may find more that their errors hid.
	max_rounds : U64
	max_rounds = 6
}

## A failed round's errors laid on the file's turned calls. An error on no
## turned call is a stray, and counts only in a round that refused no call:
## the compiler may complain where a refused call's value is used, and the
## next round, without that call, shows whether the stray was its.
judge_errors : Source.File, Dict(U64, U64), List(Candidates.CheckError), List(Candidates.Candidate) -> { cands : List(Candidates.Candidate), refused : Bool, stray : [Stray(Candidates.CheckError), NoStray] }
judge_errors = |file, line_of_node, errors, cands| {
	var $cands = cands
	var $refused = Bool.False
	var $stray = NoStray
	own = Text.basename(file.path)
	for error in errors {
		if error.file == own {
			blamed = Candidates.blame(file, line_of_node, error, $cands)
			$cands = blamed.cands
			if blamed.hit {
				$refused = Bool.True
			} else if $stray == NoStray {
				$stray = Stray(error)
			}
		} else if $stray == NoStray {
			$stray = Stray(error)
		}
	}
	{ cands: $cands, refused: $refused, stray: $stray }
}

## The binding of the plain name `name`, used at node `at`, that gives it
## its type. The binding that counts is the nearest one up the tree: in a
## block, the last definition before the statement holding the use; at the
## top level, any definition. A `var`'s type is its start's, so for a
## `$name` only its `var` counts, never a reassignment.

binding_of = |file, at, name| {
	is_var = name.starts_with("$")
	var $child = at
	var $up = Source.node_at(file, at).parent
	while $up > 0 {
		scope = Source.node_at(file, $up)
		if scope.kind == "e-lambda" and has_parameter_named(file, scope, name) {
			return Unbound
		}
		if scope.kind == "s-for" and scope.kids.len() == 3 and $child == Source.kid(scope, 3) {
			if is_parameter_named(file, Source.kid(scope, 1), name) {
				return Loop(Source.kid(scope, 2))
			}
		}
		in_block = scope.kind == "statements" and scope.parent > 0 and Source.node_at(file, scope.parent).kind == "e-block"
		var $definition = 0
		var $annotated = Bool.False
		var $passed = Bool.False
		for kid in scope.kids {
			if in_block and kid == $child {
				$passed = Bool.True
			}
			if !($passed) {
				statement = Source.node_at(file, kid)
				if is_var {
					if statement.kind == "s-var" and statement.name == name and statement.kids.len() == 1 {

					$definition = kid

					$annotated = Bool.True
				}
			}

			if $annotated {
				return Unbound
			}

			return Value(definition.kids.last() ?? 0)
		}

-- The reason the call `module.fn(...)` at node `i` must stay, whatever the
-- types say, or "" when it may be tried.
function reason_call_stays(t, i, module, fn)
  local n = t.nodes[i]
  local parent = n.parent > 0 and t.nodes[n.parent] or nil
  if parent ~= nil and parent.kind == "e-arrow-call" and parent.kids[2] == i then
    return "the right of a `|>`, whose first argument the pipe gives"
  end
  -- An argument with no span (the compiler's `e-match`) begins before any
  -- node it holds, at `match`, so its text can't be found, and writing it
  -- from no span would copy the whole file.
  for a in drop(n.kids, 1) do
    if not t.nodes[a].has_span then return "an argument with no span in the tree (" .. t.nodes[a].kind .. "), whose text cannot be moved" end
  end
  local first = n.kids[2]
  local first_node = t.nodes[first]

    if first_node.kind ~= "e-tuple" then text = "(" .. text .. ")" end

function baseline_test_of(ctx, baselines, root)
  if not has_key(baselines, root) then
    if not roc_check(ctx, root).ok then fail("rewrite_method_calls: `roc check " .. root .. "` fails before any rewrite: fix it first") end
    baselines[root] = roc_test_summary(ctx, root)
  end
  return baselines[root]
end

-- ── the types (read unless `types=0`) ────────────────────────────────────

-- Every expression's type in the app `root`, from `roc_types` (one program;
-- the library stops the run on a root the checker refuses, whose types
-- would be guesses). Returns `{by_region, modules}`: each type keyed by
-- `<real path>|line:col:end_line:end_col`, and the set of the modules' real
-- paths. Where several expressions share a region, the first that roc-types
-- gives, the outermost, is kept as the value written there. Regions are
-- shared where the compiler desugars (`x?` is a `match` over `x`, `Ok(x)` a
-- nominal over a tag, `a and b` an `if`), and the inner expressions are
-- parts of the desugaring, not the argument's value.
function types_by_region(ctx, root)
  local got = nil
  if ctx.types_from ~= "" then
    got = roc_types_from_file(ctx.types_from, root, ctx.roc, ctx.only)
  else
    ctx.programs_used = ctx.programs_used + 1
    got = roc_types(root, ctx.roc)
  end
  local by_region = {}
  for path in got.paths do
    for rec in got.modules[path] do
      if rec.kind == "expr" then
        local key = path .. "|" .. rec.line .. ":" .. rec.column .. ":" .. rec.end_line .. ":" .. rec.end_column
        if not has_key(by_region, key) then by_region[key] = rec.type end
      end
    end
  end
  for l in got.warnings do print("rewrite_method_calls: roc-types " .. root .. ": " .. l) end
  return {by_region = by_region, modules = set_of(got.paths)}
end

-- A type's head, as `{head}`, or `{var = true}` for a type variable (`a`,
-- `_a`, `a where [...]`). `List(U8)`'s head is `List` and
-- `Access.Viewer(Topic.At)`'s is `Access.Viewer`. A function's (an arrow
-- outside every bracket) is `function`, `{...}`'s is `record`, `[...]`'s
-- is `tag union`, and `(a, b)`'s is `tuple`.
function type_head(ty)
  if find(ty, "->") ~= nil or find(ty, "=>") ~= nil then
    local depth = 0
    for i = 1, #ty - 1 do
      local ch = sub(ty, i, i)
      if ch == "(" or ch == "[" or ch == "{" then depth = depth + 1 end
      if ch == ")" or ch == "]" or ch == "}" then depth = depth - 1 end
      local two = sub(ty, i, i + 1)
      if depth == 0 and (two == "->" or two == "=>") then return {head = "function"} end
    end
  end
  local first = sub(ty, 1, 1)
  if first == "{" then return {head = "record"} end
  if first == "[" then return {head = "tag union"} end
  if first == "(" then return {head = "tuple"} end
  if re_find(first, "^[A-Z]$") == nil then return {var = true} end
  local m = re_find(ty, "^([A-Za-z0-9_.]+)")
  return {head = m.groups[1]}
end

-- `word` after its indefinite article: "an" before a vowel letter, "a"
-- otherwise (`an Admin`, `a List`).
-- "an" before a vowel sound. A U usually sounds like "you" (U8, User),
-- so it takes "an" only in `Un...` (Unvalidated).
function with_article(word)
  if re_match(word, "^([AEIOaeio]|[Uu]n)") then return "an " .. word end
  return "a " .. word
end

-- Whether a first argument whose type has the head `head` proves a call of
-- module `module`, that is, whether `x.fn(...)` finds the module's `fn`. A
-- method is looked up in the block of the type's own declaration, so the
-- head must be the module's own type. roc-types writes a type by the name
-- it is declared with (no import's alias, and never a package's
-- shorthand): a type module's own type, or a builtin, bare (`Str`, `Point`,
-- `Meter` from a package imported as `pk.Meter`), and a type declared
-- inside a module's block qualified by that module (`Shapes.Shape`). So
-- the one form that proves is the bare module name, a package's type
-- module included. A qualified head never does, even one whose last
-- segment names the module: `Type.Game` is Type's, not Game's, and
-- `History.History` is a type in History's block, not History itself.
function proves_module_call(head, module)
  return head == module
end

-- Judges each open candidate of the file at `p` by its first argument's
-- type in `types` (from the root that reaches the file). A call of a
-- function in `symmetric` whose first two arguments are peers
-- (`peer_verdict`, with the trees' `aliases`) stays a module call; a concrete other type leaves the call, with that reason; a
-- type that proves the call (`proves_module_call`) marks it `proven`; the
-- rest stay open for the trial. Each kind is counted in `kind_counts`.
function judge_by_types(kind_counts, t, cands, p, types, aliases, symmetric)
  local real_path = real(p)
  for c in cands do
    if c.why_left == "" then
      local call = t.nodes[c.node]
      local key = real_path .. "|" .. t.nodes[call.kids[2]].region
      local v = {peers = false}
      if #call.kids >= 3 and has_key(symmetric, c.fn) then v = peer_verdict(get(types.by_region, key, ""), get(types.by_region, real_path .. "|" .. t.nodes[call.kids[3]].region, ""), aliases) end
      if v.peers then
        incr(kind_counts, "peers")
        c.why_left = "peers: `" .. c.fn .. "` is symmetric and both arguments are " .. v.module .. "s, so neither is the subject"
      elseif not has_key(types.by_region, key) then
        incr(kind_counts, "no type")
        c.kind = "no type"
      else
        local h = type_head(types.by_region[key])
        if has_key(h, "var") then
          incr(kind_counts, "a type variable")
          c.kind = "a type variable"
        else
          if proves_module_call(h.head, c.module) then
            incr(kind_counts, "the module's type")
            c.proven = true
          else
            incr(kind_counts, "another type")
            c.why_left = "the first argument is " .. with_article(h.head) .. ", not " .. with_article(c.module)
          end
        end
      end
    end
  end
  return nil
end

-- The peer method calls of the file at `p` (`j.method_calls`, judged by
-- `types`): each symmetric method call whose receiver and first argument are peers,
-- with a module to call through, goes in `j.peers` as a candidate that
-- turns it back (`undo`), `a.max(b)` to `U64.max(a, b)`. The types prove
-- the module's function, as the receiver's type is where the method was
-- found. A call with an argument that has no span stays, as a module call's
-- does.
function judge_method_peers(j, t, p, types, aliases)
  local real_path = real(p)
  for m in j.method_calls do
    local v = peer_verdict(get(types.by_region, real_path .. "|" .. t.nodes[m.recv].region, ""), get(types.by_region, real_path .. "|" .. t.nodes[m.first].region, ""), aliases)
    if v.peers then
      local why = m.spanned and "" or "an argument with no span in the tree, whose text cannot be moved"
      push(j.peers, {node = m.node, module = v.module, fn = m.fn, line = line_of_offset(t, t.nodes[m.node].span.from), why_left = why, proven = true, kind = "", from_line = 0, to_line = 0, undo = true, type = v.type})
    end
  end
  return nil
end

-- ── what a rewrite leaves for the sweep ─────────────────────────────────

-- The width past which a line is long, as check_stacking.shale's `long`.
const LONG_LINE = 100

-- Records, for `made_by_rewrites`, that the file at `p`, whose tree is `t`,
-- had the calls in `turning` turned: by a passing trial or confirm, or
-- under `trials=0` the proven calls. The text after is the rewrite's, before
-- any `roc fmt`, and `joined` holds its lines where a turned call that
-- spanned several lines now sits on one, as a stacked call's arguments do.
function note_rewrite(ctx, p, t, turning)
  local after = rewritten_text(t, turning)
  local joined = {}
  for k in keys(turning) do
    local c = turning[k]
    local s = t.nodes[c.node].span
    if c.from_line == c.to_line and line_of_offset(t, s.from) < line_of_offset(t, s.to - 1) then push(joined, c.from_line) end
  end
  if after ~= t.text then ctx.rewrites[p] = {before = t.text, after = after, joined = joined} end
  return nil
end

-- A line's width, a tab counting four columns.
function columns_of(l)
  return #l + 3 * (#split(l, "\t") - 1)
end

-- What the rewrites in `ctx.rewrites` left for a sweep to fix in the same
-- pass, which the recipe reports and does not fix: each import that the
-- file used before its rewrite and no longer uses after it, by roc-ast's
-- trees of both texts (imports.shale's `unused_imports`), and each line
-- wider than LONG_LINE columns where a turned call joined its lines (a turn
-- alone only shortens a line). A line is named by its number in the
-- rewritten text, before any `roc fmt`. Returns `{lines, unused, long, programs}`, `lines` a
-- `path:line: made_unused_import: why` or `made_long_line` each.
function made_by_rewrites(ctx)
  local out = {lines = {}, unused = 0, long = 0, programs = 0}
  local paths = sort(keys(ctx.rewrites))
  if #paths == 0 then return out end
  -- roc_tree reads files, so both texts are written to a scratch folder.
  local dir = fresh_dir("scratch:", "made_by_rewrites")
  local files = {}
  for k = 1, #paths do
    local r = ctx.rewrites[paths[k]]
    replace(dir .. "/" .. k .. "/before.roc", r.before)
    replace(dir .. "/" .. k .. "/after.roc", r.after)
    push(files, dir .. "/" .. k .. "/before.roc")
    push(files, dir .. "/" .. k .. "/after.roc")
  end
  out.programs = 1
  local trees = {}
  for f in roc_tree(files) do
    if f.err ~= "" then fail("rewrite_method_calls: roc-ast refused a rewritten text (" .. f.path .. "): " .. f.err) end
    trees[f.path] = f.nodes
  end
  for k = 1, #paths do
    local p = paths[k]
    local r = ctx.rewrites[p]
    local before_lines = split(r.before, "\n")
    local after_lines = split(r.after, "\n")
    local was_unused = {}
    for imp in unused_imports(trees[dir .. "/" .. k .. "/before.roc"], before_lines) do was_unused[imp.label] = true end
    for imp in unused_imports(trees[dir .. "/" .. k .. "/after.roc"], after_lines) do
      if not has_key(was_unused, imp.label) then
        out.unused = out.unused + 1
        push(out.lines, p .. ":" .. imp.line .. ": made_unused_import: " .. imp.label .. " was used only by the calls the rewrite turned: delete the line")
      end
    end
    local seen = {}
    for i in sort(r.joined) do
      local l = after_lines[i]
      if columns_of(l) > LONG_LINE and not has_key(seen, "" .. i) then
        seen["" .. i] = true
        out.long = out.long + 1
        push(out.lines, p .. ":" .. i .. ": made_long_line: " .. columns_of(l) .. " columns after the rewrite joined a call's lines: stack it (wrap_long_lines.shale): " .. trim(sub(l, 1, 120)))
      end
    end
  end
  remove_tree(dir)
  return out
end

-- ── the trial and the confirm ────────────────────────────────────────────

-- Tries one file's open candidates, round by round, until `roc check`
-- passes or nothing is left to try, then runs the test. Returns how many
-- calls were turned; each call that stays gets its `why_left`. The file
-- ends as it began unless `APPLY` and some call was turned.
function try_file(ctx, p, t, cands, root, baseline_test)
  local original = t.text
  local fmt_was_clean = APPLY and is_fmt_clean(ctx, p)
  for round = 1, MAX_ROUNDS do
    local open = open_calls(cands)
    if open.count == 0 then
      replace(p, original)
      return 0
    end
    replace(p, rewritten_text(t, open.turning))
    ctx.rounds = ctx.rounds + 1
    local r = roc_check(ctx, root)
    if r.ok then
      if fmt_was_clean then roc_fmt(ctx, p) end
      local test_summary = roc_test_summary(ctx, root)
      if test_summary ~= baseline_test then
        leave_open_calls(cands, "roc test changed: " .. test_summary .. ", was " .. baseline_test .. "; the file is put back")
        replace(p, original)
        return 0
      end
      note_rewrite(ctx, p, t, open.turning)
      if not APPLY then replace(p, original) end
      return open.count
    end
    -- Each error on a turned call's lines leaves that call. An error on no
    -- turned call is a stray, and counts only in a round that refused no
    -- call: the compiler may complain where a refused call's value is used,
    -- and the next round, without that call, shows whether the stray was
    -- its.
    local stray = nil
    local refused = false
    for e in r.errors do
      if e.file == basename(p) then
        local hit = false
        -- One error may fall on two calls, so every call is checked.
        for c in cands do
          if c.from_line > 0 and e.line >= c.from_line and e.line <= c.to_line then
            if c.why_left == "" then c.why_left = "roc check: " .. e.title end
            hit = true
          end
        end
        if not hit then hit = blame_calls_on_definition(t, e, cands) end
        if hit then refused = true end
        if not hit and stray == nil then stray = e end
      elseif stray == nil then
        stray = e
      end
    end
    if not refused and (stray ~= nil or #r.errors == 0) then
      local why = stray and ("roc check: " .. stray.title .. " at " .. stray.file .. ":" .. stray.line .. ", on no rewritten call") or ("roc check failed, naming no line: " .. first_line_said(r.out))
      leave_open_calls(cands, why .. "; the file is put back")
      replace(p, original)
      return 0
    end
    for c in cands do c.from_line = 0 end
  end
  leave_open_calls(cands, "still failing after " .. MAX_ROUNDS .. " rounds; the file is put back")
  replace(p, original)
  return 0
end

-- Confirms the files whose open calls are all proven: `files` (`{p, t,
-- cands}`) are written with all their calls turned, at once, and one
-- `roc check` on `root` and one `roc test` judge them together. When the
-- check fails, the files its errors name are set aside and the rest written
-- again, up to `MAX_ROUNDS` checks. An error in no file of theirs (a
-- caller's), a failure naming no line, or a changed test sets them all
-- aside. `ctx.confirmed[p]` is each kept file's count of turned calls, and
-- 0 for a file set aside, which `try_file` then judges alone. A file ends
-- as it began unless `APPLY` and it was kept.
function confirm_proven_files(ctx, root, files, baseline_test)
  local remaining = {}
  for f in files do
    ctx.confirmed[f.p] = 0
    local open = open_calls(f.cands)
    f.turning = open.turning
    f.count = open.count
    f.fmt_was_clean = APPLY and is_fmt_clean(ctx, f.p)
    push(remaining, f)
  end
  for round = 1, MAX_ROUNDS do
    if #remaining == 0 then return nil end
    for f in remaining do replace(f.p, rewritten_text(f.t, f.turning)) end
    ctx.confirms = ctx.confirms + 1
    local r = roc_check(ctx, root)
    if r.ok then
      for f in remaining do
        if f.fmt_was_clean then roc_fmt(ctx, f.p) end
      end
      local test_summary = roc_test_summary(ctx, root)
      if test_summary == baseline_test then for f in remaining do note_rewrite(ctx, f.p, f.t, f.turning) end end
      if test_summary ~= baseline_test or not APPLY then for f in remaining do replace(f.p, f.t.text) end end
      if test_summary == baseline_test then for f in remaining do ctx.confirmed[f.p] = f.count end end
      return nil
    end
    for f in remaining do replace(f.p, f.t.text) end
    local named = {}
    local error_elsewhere = #r.errors == 0
    for e in r.errors do
      local hit = false
      for f in remaining do
        if basename(f.p) == e.file then
          named[f.p] = true
          hit = true
        end
      end
      if not hit then error_elsewhere = true end
    end
    if error_elsewhere then return nil end
    local rest = {}
    for f in remaining do if not has_key(named, f.p) then push(rest, f) end end
    remaining = rest
  end
  return nil
end

-- The files of `globs`, several separated by commas, in the order the globs
-- give them, each once.
function paths_of_globs(globs)
  local seen = {}
  local paths = {}
  for g in split(globs, ",") do
    for q in glob(trim(g)) do
      if not has_key(seen, q) then
        seen[q] = true
        push(paths, q)
      end
    end
  end
  return paths
end

-- The root each file is judged on, by path, with the types judging each
-- file's open candidates on the way. With the types read, a file's root is
-- the first of `roots` whose app reaches it; otherwise the one root, or the
-- file itself when there is none. The roots' types are read and used one at
-- a time, but nothing is collected between them: a `yield()` there keeps
-- every file's tree, which on Safari passes the kept budget's ceiling, so
-- the arena holds each root's types until the run ends. A file with a call to try (in the set `open`) that no root
-- reaches is in `unreached`, and the run goes on without it: the types
-- can't judge it, and a path spelled two ways (through a link) would
-- quietly find no type for any of its candidates. A file with none has
-- nothing for the types to judge, so it needs no root.
function roots_of_files(ctx, paths, from, roots, read_types, open, trees, judged)
  local out = {}
  local unreached = {}
  for k = from, #paths do
    out[paths[k]] = #roots > 0 and roots[1] or paths[k]
  end
  if not read_types then return {root_of = out, unreached = unreached} end
  local found = {}
  for r in roots do
    local types = types_by_region(ctx, r)
    ctx.types_by_root[r] = true
    for k = from, #paths do
      local p = paths[k]
      if has_key(open, p) and not has_key(found, p) and has_key(types.modules, real(p)) then
        found[p] = true
        out[p] = r
        judge_by_types(judged[p].kind_counts, trees[p], judged[p].cands, p, types, ctx.aliases, ctx.symmetric)
        judge_method_peers(judged[p], trees[p], p, types, ctx.aliases)
      end
    end
    types = nil
  end
  for k = from, #paths do
    local p = paths[k]
    if has_key(open, p) and not has_key(found, p) then
      push(unreached, p)
      judged[p].unreached = true
    end
  end
  return {root_of = out, unreached = unreached}
end

-- Adds the peer method calls of the file `p` (`j.peers`) to `total`'s
-- findings, a `path:line: peer_receiver: why` each, and to its count.
function peer_findings(total, p, j)
  for c in j.peers do
    total.peers = total.peers + 1
    push(total.lines, p .. ":" .. c.line .. ": peer_receiver: `." .. c.fn .. "` is symmetric and its receiver and first argument are both " .. c.module .. "s, so neither is the subject: " .. c.module .. "." .. c.fn .. "(...)")
  end
  return nil
end

-- `trials=0`'s answer, the files of `paths` from the `from`th on as the
-- types judged them (`judged`, by path): a proven call counts as
-- rewritten, an open one the types didn't settle as unsettled, with a line
-- naming the kind of type they gave, and a call that stays as left, with
-- its reason. Nothing is written or compiled.
function counted_by_types(total, paths, from, judged, trees)
  for k = from, #paths do
    local p = paths[k]
    local j = judged[p]
    local cands = j.unreached and {} or j.cands
    local proven = {}
    for c in cands do
      if c.why_left == "" and c.proven then proven["" .. c.node] = c end
    end
    if #keys(proven) > 0 then note_rewrite(total.ctx, p, trees[p], proven) end
    if not j.unreached then total.files = total.files + 1 end
    for kind in keys(j.kind_counts) do incr(total.ctx.type_kind_counts, kind, j.kind_counts[kind]) end
    total.candidates = total.candidates + #cands
    peer_findings(total, p, j)
    for c in cands do
      if c.why_left ~= "" then
        total.left = total.left + 1
        push(total.lines, p .. ":" .. c.line .. ": " .. c.module .. "." .. c.fn .. ": " .. c.why_left)
      elseif c.proven then
        total.rewritten = total.rewritten + 1
      else
        total.unsettled = total.unsettled + 1
        push(total.lines, p .. ":" .. c.line .. ": " .. c.module .. "." .. c.fn .. ": unsettled: the types gave " .. c.kind .. ", and only a trial would tell")
      end
    end
  end
  return total
end

-- Rewrites the files of `paths` from the `from`th on, each judged on its
-- root (`root` may name several, `a.roc,b.roc`, when `use_types`). Returns
-- `{candidates, rewritten, left, lines, next, ctx}`: `lines` has a
-- `path:line: Module.fn: why` for each call that stays, and `next` is the
-- file to go on from when the program limit stopped the run, else 0.
-- `trees` is the files' parse trees, or nil to read them here (one
-- program). `roc` is the OS path of the tree's Roc, which roc-types' pin
-- must match. `opts` may hold `count_only` (`trials=0`: the types decide,
-- nothing is tried, and `unsettled` counts the calls they leave open) and
-- `types_from` (the one root's types read from that file).
function run_all(paths, root, from, trees, use_types, roc, opts?)
  local count_only = opts ~= nil and get(opts, "count_only", false)
  local types_from = opts ~= nil and get(opts, "types_from", "") or ""
  local undo = opts ~= nil and get(opts, "undo", false)
  if use_types and root == "" then fail("rewrite_method_calls: the types need root=, an app importing every file of the glob: roc-types reads an app, and a type module only through one. Give root=, or types=0 to run without them, every candidate tried") end
  if not use_types and find(root, ",") ~= nil then fail("rewrite_method_calls: root=" .. root .. ": several roots need the types, which say which root reaches a file; give one root, or run with the types") end
  local roots = root == "" and {} or roc_types_root_apps("rewrite_method_calls", root)
  if types_from ~= "" and #roots ~= 1 then fail("rewrite_method_calls: types_from= holds one root's types, and root= names " .. #roots) end
  -- A root's types read from a file are parsed for the glob's files alone.
  local only = {}
  if types_from ~= "" then
    for p in paths do only[real(p)] = true end
  end
  local ctx = {programs_used = 0, type_kind_counts = {}, types_by_root = {}, roc = roc, types_from = types_from, only = only, rounds = 0, confirms = 0, confirmed = {}, rewrites = {}}
  if trees == nil then
    trees = parse_trees(paths)
    ctx.programs_used = 1
  end
  ctx.aliases = use_types and peer_aliases(trees) or {}
  ctx.symmetric = peer_symmetric_table(read(SYMMETRIC_FUNCTIONS))
  local total = {candidates = 0, rewritten = 0, left = 0, unsettled = 0, peers = 0, lines = {}, files = 0, next = 0, ctx = ctx, unreached = {}}
  -- Each file's candidates, `{cands, kind_counts, open_count, all_proven,
  -- method_calls, peers}`. A file the types must judge is in `open`: one
  -- with a module call to try, or, with the types, a symmetric method
  -- call that passes an argument, which may be a peer call.
  local judged = {}
  local any_open = false
  local open = {}
  for k = from, #paths do
    local p = paths[k]
    if not has_key(trees, p) then fail("rewrite_method_calls: roc-ast gave no tree for " .. p) end
    local cands = candidates_in(trees[p], strip_suffix(basename(p), ".roc"))
    for c in cands do
      if c.why_left == "" then
        any_open = true
        open[p] = true
      end
    end
    local method_calls = use_types and peer_method_calls(trees[p], ctx.symmetric) or {}
    if #method_calls > 0 then
      any_open = true
      open[p] = true
    end
    judged[p] = {cands = cands, kind_counts = {}, open_count = 0, all_proven = false, unreached = false, method_calls = method_calls, peers = {}}
  end
  -- The types judge only open candidates, so with none, roc-types never runs.
  local read_types = use_types and any_open
  local reach = roots_of_files(ctx, paths, from, roots, read_types, open, trees, judged)
  local root_of = reach.root_of
  total.unreached = reach.unreached
  -- `peers=undo`: a file's candidates are its peer method calls alone.
  if undo then
    for k = from, #paths do
      local j = judged[paths[k]]
      j.cands = j.peers
      j.peers = {}
      j.kind_counts = {}
    end
  end
  for k = from, #paths do
    local p = paths[k]
    local j = judged[p]
    j.all_proven = true
    for c in j.cands do
      if c.why_left == "" then
        j.open_count = j.open_count + 1
        if not c.proven then j.all_proven = false end
      end
    end
    if j.open_count == 0 or j.unreached then j.all_proven = false end
  end
  if count_only then return counted_by_types(total, paths, from, judged, trees) end
  local baselines = {}
  for k = from, #paths do
    local p = paths[k]
    local j = judged[p]
    local cands = j.unreached and {} or j.cands
    local r = root_of[p]
    if j.unreached then
      -- Not begun: listed after the findings.
    elseif j.all_proven and not has_key(ctx.confirmed, p) then
      -- The programs left after this root's baseline (two, if it has none
      -- yet), the confirm's checks and test, and this file's own trial,
      -- should the confirm set it aside. Under `apply=1` each file in the
      -- confirm also takes two `roc fmt`s, so only as many join as fit.
      local room = MAX_PROGRAMS - ctx.programs_used - (has_key(baselines, r) and 0 or 2) - MAX_ROUNDS - 1 - PROGRAMS_PER_FILE
      local files = {}
      for m = k, #paths do
        local q = paths[m]
        local jq = judged[q]
        if jq.all_proven and root_of[q] == r and not has_key(ctx.confirmed, q) and (not APPLY or 2 * (#files + 1) <= room) then push(files, {p = q, t = trees[q], cands = jq.cands}) end
      end
      if room < 0 or #files == 0 then
        total.next = k
        return total
      end
      confirm_proven_files(ctx, r, files, baseline_test_of(ctx, baselines, r))
      for c in cands do c.from_line = 0 end
    end
    local kept_by_confirm = get(ctx.confirmed, p, 0)
    if not j.unreached and j.open_count > 0 and kept_by_confirm == 0 and ctx.programs_used + PROGRAMS_PER_FILE > MAX_PROGRAMS then
      total.next = k
      return total
    end
    -- A file's type counts are added once it is begun, so a run that stops
    -- before it leaves them to the next run.
    if not j.unreached then total.files = total.files + 1 end
    for kind in keys(j.kind_counts) do incr(ctx.type_kind_counts, kind, j.kind_counts[kind]) end
    peer_findings(total, p, j)
    if #cands > 0 then
      local kept = kept_by_confirm
      if kept == 0 and j.open_count > 0 then kept = try_file(ctx, p, trees[p], cands, r, baseline_test_of(ctx, baselines, r)) end
      total.candidates = total.candidates + #cands
      total.rewritten = total.rewritten + kept
      for c in cands do
        if c.why_left ~= "" then
          total.left = total.left + 1
          push(total.lines, p .. ":" .. c.line .. ": " .. c.module .. "." .. c.fn .. ": " .. c.why_left)
        end
      end
    end
  end
  return total
end

function report_check(failures, passed, what)
  if passed then
    print("ok    " .. what)
  else
    print("FAIL  " .. what)
    failures.count = failures.count + 1
  end
  return nil
end

-- roc-types reads an app, and an app names its platform.

-- The child run `plant=<name>`: copies the plants to a fresh scratch folder
-- and runs the one plant there with `apply=1`, then checks the result
-- against its `EXPECT` entry.
function self_test_plant(name)
  if not has_key(EXPECT, name) then fail("rewrite_method_calls: self-test: no plant " .. name .. " in the table") end
  local want = EXPECT[name]
  local scratch = fresh_dir("scratch:", "methods")
  for f in ls(PLANTS_DIR) do replace(scratch .. "/" .. f, read(PLANTS_DIR .. "/" .. f)) end
  local failures = {count = 0}
  local file = get(want, "file", name)
  local p = scratch .. "/" .. file .. ".roc"
  local paths = {p}
  if has_key(want, "with") then push(paths, scratch .. "/" .. want.with .. ".roc") end
  local with_types = has_key(want, "types")
  -- A plant with `tree_pin` runs as though the tree pinned that Roc.
  local roc = has_key(want, "tree_pin") and real("tools:roc-" .. want.tree_pin .. "/roc") or roc_bin("rewrite_method_calls", SELF_TEST)
  local roots = {}
  for r in split(get(want, "root", ""), ",") do
    if r ~= "" then push(roots, scratch .. "/" .. r .. ".roc") end
  end
  -- A plant with `types_from` reads the types from a file, as a gate keeps
  -- them: roc-types' output on the root, written here first.
  local types_from = ""
  if has_key(want, "types_from") then
    types_from = scratch .. "/root.types"
    replace(types_from, must(run({real(roc_types_exe(roc)), real(roots[1])}, {timeout_ms = 300000, whole = 64000000})).out)
  end
  local count_only = has_key(want, "count")
  local got = run_all(paths, join(roots, ","), 1, nil, with_types, roc, {count_only = count_only, types_from = types_from, undo = has_key(want, "undo")})
  if has_key(want, "stop") then fail("rewrite_method_calls: self-test: " .. name .. " ran to its end, and was to stop: " .. want.stop) end
  if count_only then
    report_check(failures, got.rewritten == want.proven, name .. ": " .. got.rewritten .. " proven, want " .. want.proven)
    report_check(failures, got.unsettled == want.unsettled, name .. ": " .. got.unsettled .. " unsettled, want " .. want.unsettled)
    report_check(failures, read(p) == read(PLANTS_DIR .. "/" .. file .. ".roc"), name .. ": counted, and left as it was")
  end
  if has_key(want, "peers") then report_check(failures, got.peers == want.peers, name .. ": " .. got.peers .. " peer_receiver, want " .. want.peers) end
  if not has_key(want, "stop") then
    local made = made_by_rewrites(got.ctx)
    local want_unused = get(want, "made_unused", 0)
    local want_long = get(want, "made_long", 0)
    report_check(failures, made.unused == want_unused and made.long == want_long, name .. ": " .. made.unused .. " made_unused_import and " .. made.long .. " made_long_line, want " .. want_unused .. " and " .. want_long .. (#made.lines > 0 and (" (" .. join(made.lines, "; ") .. ")") or ""))
    if has_key(want, "made_says") then report_check(failures, contains(join(made.lines, "\n"), want.made_says), name .. ": reports " .. want.made_says) end
  end
  if types_from ~= "" then report_check(failures, got.ctx.programs_used == 1, name .. ": the types read from a file, so roc-ast the one program (" .. got.ctx.programs_used .. ")") end
  local want_unreached = get(want, "unreached", 0)
  report_check(failures, #got.unreached == want_unreached, name .. ": " .. #got.unreached .. " not reached by a root, want " .. want_unreached .. " (" .. join(got.unreached, ", ") .. ")")
  local left = got.lines
  report_check(failures, #left == want.left, name .. ": " .. #left .. " left, want " .. want.left .. (#left > 0 and (" (" .. join(left, "; ") .. ")") or ""))
  if has_key(want, "why") then report_check(failures, #left > 0 and contains(join(left, "\n"), want.why), name .. ": left because " .. want.why) end
  local want_path = PLANTS_DIR .. "/" .. file .. ".want"
  if exists(want_path) and not count_only then report_check(failures, read(p) == read(want_path), name .. ": comes out as " .. file .. ".want") end
  -- In Typed, the types turn one call and leave the other, so the trial
  -- decides neither.
  local kinds = got.ctx.type_kind_counts
  if name == "Typed" then report_check(failures, #keys(kinds) == 2 and get(kinds, "another type", 0) == 1 and get(kinds, "the module's type", 0) == 1, name .. ": by type, " .. json_write(kinds)) end
  local confirms = got.ctx.confirms
  local rounds = got.ctx.rounds
  local want_confirms = get(want, "confirms", 0)
  report_check(failures, confirms == want_confirms, name .. ": " .. confirms .. " confirm(s), want " .. want_confirms)
  if has_key(want, "rounds") then report_check(failures, rounds == want.rounds, name .. ": " .. rounds .. " round(s), want " .. want.rounds) end
  if not with_types then report_check(failures, #keys(kinds) == 0 and #keys(got.ctx.types_by_root) == 0, name .. ": under types=0, no roc-types run") end
  -- With the types, roc-types runs once per root when a candidate is open,
  -- and not at all on a file with none (`no_types_run`).
  if with_types then
    local runs = #keys(got.ctx.types_by_root)
    local want_runs = has_key(want, "no_types_run") and 0 or #roots
    report_check(failures, runs == want_runs, name .. ": " .. runs .. " roc-types run(s), want " .. want_runs)
  end
  if want.left > 0 and not exists(want_path) then report_check(failures, read(p) == read(PLANTS_DIR .. "/" .. file .. ".roc"), name .. ": put back as it was") end
  if has_key(want, "with") then report_check(failures, read(paths[2]) == read(PLANTS_DIR .. "/" .. want.with .. ".roc"), name .. ": " .. want.with .. " left as it was") end
  if failures.count > 0 then fail("rewrite_method_calls: self-test: " .. name .. ": " .. failures.count .. " failed") end
  remove_tree(scratch)
  return nil
end

-- Runs every plant in a child run of its own; a failed plant is named, and
-- the rest still run.
function self_test()
  if not exists(ROC_TYPES_EXE) then fail("rewrite_method_calls: self-test: no " .. ROC_TYPES_EXE .. ", which the plants marked `types` are judged by: " .. ROC_TYPES_INSTALL) end
  local failed = {}
  local globbed = paths_of_globs(PLANTS_DIR .. "/Proven.roc, " .. PLANTS_DIR .. "/NoC*.roc," .. PLANTS_DIR .. "/Proven.roc")
  local globs_held = #globbed == 2 and basename(globbed[1]) == "Proven.roc" and basename(globbed[2]) == "NoCalls.roc"
  print((globs_held and "ok    " or "FAIL  ") .. "glob=: two globs and a repeat give Proven.roc, then NoCalls.roc, each once (" .. join(globbed, ", ") .. ")")
  if not globs_held then push(failed, "glob=") end
  for file in ls(PLANTS_DIR) do
    local name = strip_suffix(file, ".roc")
    if ends_with(file, ".roc") and not has_key(EXPECT, name) and not has_key(ROOT_PLANTS, name) then
      print("FAIL  " .. file .. ": a plant the table does not judge")
      push(failed, file)
    end
  end
  for name in sort(keys(EXPECT)) do
    local r = run({"shale", "--root", ".", THIS_SCRIPT, "self_test=1", "plant=" .. name}, {timeout_ms = 3600000, tail = 200})
    local stop = get(EXPECT[name], "stop", "")
    if stop ~= "" then
      local held = r.ended == "exit" and r.code ~= 0 and contains(r.out, stop)
      print((held and "ok    " or "FAIL  ") .. name .. ": stops, saying " .. stop .. (held and "" or (": " .. trim(r.out))))
      if not held then push(failed, name) end
    else
      for l in lines(r.out) do
        if not starts_with(l, "── budgets") then print(l) end
      end
      if r.ended ~= "exit" or r.code ~= 0 then push(failed, name) end
    end
  end
  if #failed > 0 then fail("rewrite_method_calls: self-test: " .. #failed .. " plant(s) failed: " .. join(failed, ", ")) end
  print("rewrite_method_calls: self-test: every plant as it says")
end

if PLANT ~= "" and not SELF_TEST then
  fail("rewrite_method_calls: plant= is for the self-test's child runs, and needs self_test=1")
elseif SELF_TEST and PLANT ~= "" then
  self_test_plant(PLANT)
elseif SELF_TEST then
  self_test()
else
  local paths = paths_of_globs(GLOB)
  local total = run_all(paths, ROOT, FROM, nil, USE_TYPES, roc_bin("rewrite_method_calls", SELF_TEST), {count_only = COUNT_ONLY, types_from = TYPES_FROM, undo = UNDO})
  local summary = (UNDO and "peers=undo, " or "") .. (APPLY and "" or "dry run: ") .. "files " .. total.files .. ", candidates " .. total.candidates .. ", rewritten " .. total.rewritten .. ", left " .. total.left .. ", peer_receiver " .. total.peers .. ", programs " .. total.ctx.programs_used .. " (checks: " .. total.ctx.confirms .. " confirming proven calls, " .. total.ctx.rounds .. " in trial rounds)"
  if COUNT_ONLY then summary = "count: files " .. total.files .. ", candidates " .. total.candidates .. ", proven " .. total.rewritten .. ", unsettled " .. total.unsettled .. ", left " .. total.left .. ", peer_receiver " .. total.peers .. ", programs " .. total.ctx.programs_used .. " (no trials: the types alone decide)" end
  if total.next > 0 then summary = summary .. ", in files " .. FROM .. " to " .. (total.next - 1) .. " of " .. #paths .. ": a run holds " .. MAX_PROGRAMS .. " programs; go on with from=" .. total.next end
  local made = made_by_rewrites(total.ctx)
  summary = summary .. ", made_unused_import " .. made.unused .. ", made_long_line " .. made.long
  local finding_lines = total.lines
  for l in made.lines do push(finding_lines, l) end
  if USE_TYPES then
    local kinds = {}
    for kind in sort(keys(total.ctx.type_kind_counts)) do push(kinds, kind .. " " .. total.ctx.type_kind_counts[kind]) end
    push(finding_lines, "types: the first arguments of the candidates the types judged: " .. join(kinds, ", "))
  end
  print_findings(summary, finding_lines)
  for u in total.unreached do print("not reached by " .. ROOT .. ": " .. u) end
  if #total.unreached > 0 then fail("rewrite_method_calls: no root reaches " .. #total.unreached .. " file(s) with a call to try, listed above: their calls were left untried") end
end
