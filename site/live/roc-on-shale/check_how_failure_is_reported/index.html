## check_how_failure_is_reported, in Roc on Shale's platform (roc_platform/README.md): how Roc
## code reports failure, against the rules in roc-apps docs/roc-style.md
## ("Errors and failure"). Its launcher, cookbook/roc/check_how_failure_is_reported.shale, is
## how it is run, and its header says the five checks, `error_alias`,
## `optional_tag`, `set_fallback`, `placeholder` and `number_fallback`.
## check_goldens.shale, beside this, holds its output on the plants to its
## goldens, line for line.
##
##   cd <repo> && shale cookbook:roc/check_how_failure_is_reported.shale glob='app/**/*.roc'
##   shale cookbook:roc/check_how_failure_is_reported.shale self_test=1
##
## The arguments: `glob` (`**/*.roc`), `types` (`1` asks Roc's type
## checker), `root` (the app or apps the types are read from) and
## `self_test` (`self_test=1` checks the plants, the typed ones with the
## types).
##
## This root reads the optionals' words (`Shale.read!`), globs the files
## (`Shale.glob!`), gathers the types (Types, through `Shale.run!`), has
## Roc's parser read the files (`Shale.roc_tree!`) and prints the verdict
## (`Shale.print!`, `Shale.fail!`). Its modules are beside it: a module a
## check or two (ErrorAlias, OptionalTag, Fallbacks), Types and SelfTest;
## Tree, Forest (the files' trees joined), Text, Finding and RocTypes are the
## cookbook's Roc library, cookbook/roc/lib/.
## budget: steps=400000000 arena_bytes=3000000000 output_bytes=4000000 programs=100
app [main!] { lib: "../lib/main.roc", pf: platform "../../../roc_platform/platform/main.roc" }

import pf.Shale
import ErrorAlias
import Fallbacks
import lib.Finding
import lib.Forest
import OptionalTag
import lib.RocTypes
import SelfTest
import lib.Text
import Types

main! : {} => {}
main! = |{}| {
	self_test = Shale.arg!("self_test", "") == "1"
	plants = SelfTest.plants
	glob_asked = Shale.arg!("glob", "**/*.roc")
	glob = if self_test {
		"${plants}*.roc"
	} else {
		glob_asked
	}
	types_asked = Shale.arg!("types", "0")
	if types_asked != "0" and types_asked != "1" {
		Shale.fail!("check_how_failure_is_reported: types=${types_asked} must be 0 (the default) or 1")
	}
	# The self-test checks its typed plants with the types, and the rest
	# without.
	use_types = self_test or types_asked == "1"
	root_asked = Shale.arg!("root", "")
	root = if self_test {
		"${plants}typed/QuietApp.roc"
	} else {
		root_asked
	}
	optionals = read_word_set!("optionals.txt")
	paths = if self_test {
		Shale.glob!(glob).concat(Shale.glob!("${plants}typed/*.roc"))
	} else {
		Shale.glob!(glob)
	}
	types = if use_types {
		roc = Types.roc_bin!("check_how_failure_is_reported", self_test)
		# The self-test's untyped plants are exempt from being reached by its
		# root.
		exempt = if self_test {
			Set.from_list(Shale.glob!("${plants}*.roc"))
		} else {
			Set.empty()
		}
		Types.gather!({ who: "check_how_failure_is_reported", kind: Expressions, root, roc, paths, exempt })
	} else {
		{ typing: RocTypes.no_types, unreached: [] }
	}
	var $parsed = []
	var $refused = []
	for file in Shale.roc_tree!(paths) {
		if file.err == "" {
			$parsed = $parsed.append({ path: file.path, nodes: file.nodes })
		} else if file.err.starts_with(file.path) {
			$refused = $refused.append(file.err)
		} else {
			$refused = $refused.append("${file.path}: ${file.err}")
		}
	}
	forest = Forest.join($parsed)
	aliases = ErrorAlias.judge(forest)
	optional_found = OptionalTag.judge(forest, optionals)
	findings = aliases.findings
		.concat(optional_found.map(|found| found.finding))
		.concat(Fallbacks.judge!(forest, types.typing))
	unjudged = Text.sort_bytewise(aliases.unjudged.keys())
	counted = unjudged.map(|name| "${name} (${(aliases.unjudged.get(name) ?? 0).to_str()})")
	unjudged_said = if counted.is_empty() {
		""
	} else {
		"; error_alias not judged, declared outside the files hunted: ${Str.join_with(counted, ", ")}"
	}
	summary = "check_how_failure_is_reported (Roc): ${paths.len().to_str()} files; ${Finding.counts_of(findings, checks)}${unjudged_said}"
	print_findings!(summary, findings.map(line_of))
	fail_if_refused!($refused, types.unreached)
	if self_test {
		judged = { paths, findings, optionals: optional_found, unjudged }
		SelfTest.run!(judged, checks)
	}
}

## The checks, in the order the summary counts them.
checks : List(Str)
checks = ["error_alias", "optional_tag", "set_fallback", "placeholder", "number_fallback"]

## A finding as its line, `path:line: check: what`, its line `?` when no node
## from the finding's on has one.
line_of : Finding -> Str
line_of = |finding|
	if finding.line == 0 {
		"${finding.path}:?: ${finding.check}: ${finding.what}"
	} else {
		Finding.line_of(finding)
	}

## The words listed in cookbook/words/<file>, one a line, `--` starting a
## note. A line of two words, or a word listed twice, stops the run.
read_word_set! : Str => Set(Str)
read_word_set! = |file| {
	text = match Shale.read!("cookbook:words/${file}") {
		Found(found) => found
		Missing => {
			Shale.fail!("check_how_failure_is_reported: no cookbook/words/${file}")
			""
		}
	}
	var $words = Set.empty()
	for line in Text.lines(text) {
		word = Text.trim(Text.before_first(line, "--"))
		if word.contains(" ") or $words.contains(word) {
			Shale.fail!("check_how_failure_is_reported: cookbook/words/${file}: `${word}` is not a single word, or is listed twice")
		}
		if word != "" {
			$words = $words.insert(word)
		}
	}
	$words
}

## The summary, each finding, then the summary again (cookbook/findings.shale):
## the first copy survives a cut output, and a gate reads the last line.
print_findings! : Str, List(Str) => {}
print_findings! = |summary, finding_lines| {
	Shale.print!(summary)
	if !(finding_lines.is_empty()) {
		for line in finding_lines {
			Shale.print!(line)
		}
		Shale.print!(summary)
	}
}

## After the findings: the files that didn't parse, and those no root of the
## types reaches (judged without them), named, then a stop
## (cookbook/roc/parsed.shale's `fail_if_refused`).
fail_if_refused! : List(Str), List(Str) => {}
fail_if_refused! = |refused, unreached| {
	if !(refused.is_empty()) or !(unreached.is_empty()) {
		for refusal in refused {
			Shale.print!("refused: ${refusal}")
		}
		for missed in unreached {
			Shale.print!(missed)
		}
		var $why = []
		if !(refused.is_empty()) {
			$why = $why.append("roc-ast refused ${refused.len().to_str()} files (named above), which have no tree to judge")
		}
		if !(unreached.is_empty()) {
			$why = $why.append("${unreached.len().to_str()} files (named above) are reached by no root of the types, so were judged without them")
		}
		Shale.fail!("check_how_failure_is_reported: ${Str.join_with($why, "; ")}; every other file was judged")
	}
}

## The `error_alias` check: a function's result is a `Try` whose error is an
## alias of tags (`ReadError : [Missing, Failed]`, or a module's
## `Accounts.Unreadable`) instead of the tags written in place. The alias is
## looked up in the files checked, through up to 8 aliases of aliases: first
## among the file's own declarations, then by its qualified name, which a
## file's top-level types also have by the file's module name (`Access.Answer`
## for `Answer` in Access.roc). Only top-level and module-member annotations
## are read. An error declared outside the files checked can't be judged, and
## is counted by name for the summary.
import lib.Finding
import lib.Forest
import lib.Tree

ErrorAlias := [].{

	## The findings, in the order of the annotations and of the errors inside
	## each result, and how often each error declared outside the files was
	## named.
	judge : Forest -> { findings : List(Finding), unjudged : Dict(Str, U64) }
	judge = |forest| {
		declared = declarations_of(forest)
		var $findings = []
		var $unjudged = Dict.empty()
		$at : U64
		var $at = 0
		while $at < Forest.count(forest) {
			$at = $at + 1
			annotation = $at
			signature = Forest.kid(forest, annotation, 1)
			is_judged = Forest.kind(forest, annotation) == "s-type-anno" and is_top_level(forest, annotation)
			if is_judged and Forest.kind_or_none(forest, signature) == "ty-fn" {
				result = Forest.kids(forest, signature).last() ?? 0
				last = Forest.last_descendant(forest, result)
				$inside : U64
				var $inside = result
				while $inside <= last {
					error = try_error(forest, $inside)
					error_name = if error > 0 {
						Forest.name(forest, error)
					} else {
						""
					}
					if error_name != "" and Forest.kind(forest, error) == "ty" {
						path = Forest.path(forest, annotation)
						match error_kind(forest, declared, { path, name: error_name, hops: 0 }) {
							Union => {
								$findings = $findings.append(alias_finding(forest, annotation, error_name))
							}
							Unknown => {
								counted = ($unjudged.get(error_name) ?? 0) + 1
								$unjudged = $unjudged.insert(error_name, counted)
							}
							Other => {}
						}
					}
					$inside = $inside + 1
				}
			}
		}
		{ findings: $findings, unjudged: $unjudged }
	}
}

## Every type declared in the files checked, by its qualified name (and a
## file's top-level ones by the module's name too), and by its file's path
## and bare name: the place of what it is declared as.
Declared : { qualified : Dict(Str, U64), in_file : Dict(Str, U64) }

declarations_of : Forest -> Declared
declarations_of = |forest| {
	var $qualified = Dict.empty()
	var $in_file = Dict.empty()
	$at : U64
	var $at = 0
	while $at < Forest.count(forest) {
		$at = $at + 1
		name = declared_name(forest, $at)
		if name != "" {
			definition = Forest.kid(forest, $at, 2)
			qualified = nested_name(forest, $at)
			path = Forest.path(forest, $at)
			$qualified = $qualified.insert(qualified, definition)
			if qualified == name {
				$qualified = $qualified.insert("${Tree.module_name(path)}.${name}", definition)
			}
			$in_file = $in_file.insert("${path}|${name}", definition)
		}
	}
	{ qualified: $qualified, in_file: $in_file }
}

## The kinds of node above a top-level or member annotation.
top_level_kinds : Set(Str)
top_level_kinds = Set.from_list(["file", "statements", "s-type-decl", "associated"])

## Whether every node above the one at `at` is a file, its statements, a type
## declaration, or a type's associated block.
is_top_level : Forest, U64 -> Bool
is_top_level = |forest, at| {
	var $above = Forest.parent(forest, at)
	while $above > 0 {
		if !(top_level_kinds.contains(Forest.kind(forest, $above))) {
			return Bool.False
		}
		$above = Forest.parent(forest, $above)
	}
	Bool.True
}

## The name the type declaration at `at` declares, `""` if it is not one.
declared_name : Forest, U64 -> Str
declared_name = |forest, at|
	if Forest.kind(forest, at) != "s-type-decl" or Forest.kids(forest, at).len() < 2 {
		""
	} else {
		Forest.name(forest, Forest.kid(forest, at, 1))
	}

## The declaration's name qualified by the declarations around it
## (`Stored.Problem`).
nested_name : Forest, U64 -> Str
nested_name = |forest, at| {
	var $names = [declared_name(forest, at)]
	var $above = Forest.parent(forest, at)
	while $above > 0 {
		around = declared_name(forest, $above)
		if around != "" {
			$names = List.prepend($names, around)
		}
		$above = Forest.parent(forest, $above)
	}
	Str.join_with($names, ".")
}

## What an error type is: an alias of tags, a type declared outside the
## files checked, or anything else (a record, a builtin, an empty union).
ErrorKind : [Union, Unknown, Other]

## The kind of the error type `name` named in `path`, followed through `hops`
## aliases so far, at most 8.
error_kind : Forest, Declared, { path : Str, name : Str, hops : U64 } -> ErrorKind
error_kind = |forest, declared, named| {
	found = match declared.in_file.get("${named.path}|${named.name}") {
		Ok(at) => Ok(at)
		Err(_) => declared.qualified.get(named.name)
	}
	if builtin_types.contains(named.name) {
		Other
	} else {
		match found {
			Err(_) => Unknown
			Ok(at) => {
				kind = Forest.kind(forest, at)
				aliased = Forest.name(forest, at)
				if kind == "ty-tag-union" {
					if Forest.union_tags(forest, at).is_empty() {
						Other
					} else {
						Union
					}
				} else if kind == "ty" and aliased != "" and named.hops < 8 {
					error_kind(forest, declared, { ..named, name: aliased, hops: named.hops + 1 })
				} else {
					Other
				}
			}
		}
	}
}

## The error of a `Try(ok, error)` type at `at`, 0 for any other node.
try_error : Forest, U64 -> U64
try_error = |forest, at|
	if Forest.kind(forest, at) != "ty-apply" or Forest.kids(forest, at).len() != 3 {
		0
	} else if Forest.name(forest, Forest.kid(forest, at, 1)) == "Try" {
		Forest.kid(forest, at, 3)
	} else {
		0
	}

alias_finding : Forest, U64, Str -> Finding
alias_finding = |forest, annotation, error_name| {
	fixes = "list the tags it can return in place, or leave the error to inference (`_`)"
	{
		path: Forest.path(forest, annotation),
		line: Forest.line_number(forest, annotation),
		check: "error_alias",
		what: "`${Forest.name(forest, annotation)}` fails with `${error_name}`, a shared alias of tags: ${fixes}",
	}
}

## The checks of what a `??` falls back to:
##
## - `set_fallback`: `List.set(l, i, v) ?? l`, or `l.set(i, v) ?? l`, where
##   the fallback is the list being stored into: a placeholder, and on a
##   list of strings a copy on every store in the dev build.
## - `placeholder`: a fallback that is a string literal with nothing
##   interpolated (`?? "bogus"`, `?? ""`).
## - `number_fallback`: a fallback that is a number after a lookup by index
##   (`List.get(l, i) ?? 0`, `l.get(i) ?? 0`); with the types, after any
##   lookup that fails only with `[OutOfBounds]`.
##
## With the types, a finding names the fallback's type and the tags the
## lookup fails with. Each `??` is judged by the three in that order.
import pf.Shale
import lib.Finding
import lib.Forest
import lib.RocTypes

Fallbacks := [].{

	## The findings, in the order of the `??`s, with the expressions' types
	## when the run has them.
	judge! : Forest, RocTypes.Typing => List(Finding)
	judge! = |forest, typing| {
		var $findings = []
		$at : U64
		var $at = 0
		while $at < Forest.count(forest) {
			$at = $at + 1
			at = $at
			is_fallback = Forest.kind(forest, at) == "e-binop" and Forest.attribute(forest, at, "op") == "??"
			if is_fallback and Forest.kids(forest, at).len() == 2 {
				pair = {
					lookup: Forest.kid(forest, at, 1),
					fallback: Forest.kid(forest, at, 2),
					path: Forest.path(forest, at),
				}
				$findings = $findings.concat(judge_pair!(forest, typing, pair))
			}
		}
		$findings
	}
}

## A `??`'s two sides and its file.
Pair : { lookup : U64, fallback : U64, path : Str }

## What one `??` is: its set_fallback, its placeholder, its number_fallback.
judge_pair! : Forest, RocTypes.Typing, Pair => List(Finding)
judge_pair! = |forest, typing, pair| {
	has_types = typing.real.contains(pair.path)
	lookup_error = if has_types {
		lookup_error_of!(forest, typing, pair)
	} else {
		""
	}
	lookup_said = if lookup_error != "" {
		", for a lookup that fails with `${lookup_error}`"
	} else {
		""
	}
	line = Forest.line_number(forest, pair.fallback)
	at_fallback = |check, what| { path: pair.path, line, check, what }
	var $findings = []
	target = list_stored_into(forest, pair.lookup)
	if target > 0 and is_same(forest, target, pair.fallback) {
		why = "the fallback names `l`, an answer no store means (and on a list of strings a copy on every store in the dev build); write `?? crash \"...\"` instead"
		cited = "(roc-apps CLAUDE.md, \"Performance: know what copies\")"
		$findings = $findings.append(at_fallback("set_fallback", "`List.set(l, i, v) ?? l`: ${why} ${cited}"))
	}
	style = "(roc-apps docs/roc-style.md, \"Errors and failure\")"
	if is_literal_text(forest, pair.fallback) {
		text = Str.join_with(Forest.kids(forest, pair.fallback).map(|part| Forest.raw(forest, part)), "")
		typed = if has_types {
			", a Str${lookup_said}"
		} else {
			""
		}
		why = "if the lookup can't fail, `crash` with a reason, \"never a silent `?? default`\"; if it can, keep that case as a tag"
		$findings = $findings.append(at_fallback("placeholder", "falls back to the literal `\"${text}\"`${typed}: ${why} ${style}"))
	}
	is_judged_lookup = lookup_error == "[OutOfBounds]" or (!(has_types) and is_index_lookup(forest, pair.lookup))
	if is_number(forest, pair.fallback) and is_judged_lookup {
		number = Forest.raw(forest, pair.fallback)
		typed = if has_types {
			", a ${type_at(forest, typing, { path: pair.path, at: pair.fallback }) ?? "?"}"
		} else {
			""
		}
		why = "an index past the end is a bug, not a ${number}; `crash` with a reason, or keep the case as a tag"
		$findings = $findings.append(at_fallback("number_fallback", "falls back to `${number}`${typed}${lookup_said}: ${why} ${style}"))
	}
	$findings
}

## The tag union the lookup's `Try` fails with, such as `[OutOfBounds]`, or
## `""` when the checker gave it no such type.
lookup_error_of! : Forest, RocTypes.Typing, Pair => Str
lookup_error_of! = |forest, typing, pair|
	match type_at(forest, typing, { path: pair.path, at: pair.lookup }) {
		Ok(lookup_type) =>
			match Shale.re_find!(lookup_type, "^Try\\((.*), (\\[[^\\[\\]]*\\])\\)$") {
				Found(matched) => matched.groups.get(1) ?? ""
				NoMatch => ""
			}
		Err(_) => ""
	}

## The checker's type of the expression at a place in a file it reached.
type_at : Forest, RocTypes.Typing, { path : Str, at : U64 } -> Try(Str, [NoType])
type_at = |forest, typing, spot| RocTypes.type_at(typing, spot.path, Forest.node(forest, spot.at))

## Whether the expressions at two places are the same value written twice
## (`board`, `r.b`): the same kinds, names and fields, all the way down.
is_same : Forest, U64, U64 -> Bool
is_same = |forest, first, second| {
	is_alike = |attribute| Forest.attribute(forest, first, attribute) == Forest.attribute(forest, second, attribute)
	first_kids = Forest.kids(forest, first)
	second_kids = Forest.kids(forest, second)
	if Forest.kind(forest, first) != Forest.kind(forest, second) or !(is_alike("raw")) or !(is_alike("field")) {
		Bool.False
	} else if first_kids.len() != second_kids.len() {
		Bool.False
	} else {
		List.map2(first_kids, second_kids, |a, b| (a, b)).all(|(a, b)| is_same(forest, a, b))
	}
}

## The list that the `List.set(l, i, v)` or `l.set(i, v)` call at a place
## stores into, 0 for any other expression.
list_stored_into : Forest, U64 -> U64
list_stored_into = |forest, at| {
	kind = Forest.kind(forest, at)
	kids = Forest.kids(forest, at)
	callee = Forest.kid(forest, at, 1)
	if kind == "e-apply" and kids.len() == 4 and Forest.kind_or_none(forest, callee) == "e-ident" and Forest.raw(forest, callee) == "List.set" {
		Forest.kid(forest, at, 2)
	} else if kind == "e-method-call" and Forest.attribute(forest, at, "method") == ".set" {
		is_receiver = |kid| Forest.kind(forest, kid) == "receiver" and Forest.kids(forest, kid).len() == 1
		match kids.find_first(is_receiver) {
			Ok(receiver) => Forest.kid(forest, receiver, 1)
			Err(_) => 0
		}
	} else {
		0
	}
}

## Whether the expression at a place is a string literal with nothing
## interpolated.
is_literal_text : Forest, U64 -> Bool
is_literal_text = |forest, at| {
	is_part = |kid| Forest.kind(forest, kid) == "e-string-part"
	Forest.kind(forest, at) == "e-string" and Forest.kids(forest, at).all(is_part)
}

## Whether the expression at a place is a lookup by index: `List.get(l, i)`
## or `l.get(i)`.
is_index_lookup : Forest, U64 -> Bool
is_index_lookup = |forest, at| {
	kind = Forest.kind(forest, at)
	callee = Forest.kid(forest, at, 1)
	if kind == "e-apply" and Forest.kind_or_none(forest, callee) == "e-ident" {
		Forest.raw(forest, callee) == "List.get"
	} else {
		kind == "e-method-call" and Forest.attribute(forest, at, "method") == ".get"
	}
}

is_number : Forest, U64 -> Bool
is_number = |forest, at| {
	kind = Forest.kind(forest, at)
	kind == "e-int" or kind == "e-frac"
}

## The `optional_tag` check: a tag named like an optional (`Some`, `None`,
## `Nothing`; cookbook/words/optionals.txt lists them), matched by its whole
## name. A type that names one beside a tag with a payload is a finding at
## the tag's line; with no payload there, the union is a mode with two ways
## (`[Words, None]`). A tag in an expression or a pattern is a finding once a
## file, at its first use, only when no type in the files checked names it.
import lib.Finding
import lib.Forest
import lib.Text

OptionalTag := [].{

	## A finding with the tag it names, which the self-test holds to the
	## finding's line.
	Found : { finding : Finding, tag : Str }

-- roc-apps docs/roc-style.md ("Errors and failure"). It reads each .roc
-- file's parse tree from Roc's own parser, so text in a comment or a string
-- never matches. A finding is a question for the reader, not a verdict.
-- There are five checks:
--
--   error_alias   A function's result is a `Try` whose error is an alias of
--                 tags (`ReadError : [Missing, Failed]`, or a module's
--                 `Accounts.Unreadable`) instead of the tags written in
--                 place. A shared alias makes every caller handle tags its
--                 function can't return. The alias is looked up in the files
--                 checked, through an alias of an alias. An error that is not
--                 a tag union (a record, a `Str`), an alias in an argument,
--                 an inferred error (`_`) and a type variable are all fine.
--                 An error declared outside the files checked can't be
--                 judged, so the summary line names it.
--   optional_tag  A tag named like an optional (`Some`, `None`, `Nothing`;
--                 the full list is cookbook/words/optionals.txt). A value
--                 that may be absent should be tagged by what it is, and its
--                 absence by why (`[Latest(Message), NoMessage]`). A type
--                 that names such a tag beside a tag with a payload is a
--                 finding at the tag's line. With no payload there, the union
--                 is a mode with two ways (`[Words, None]`), and `None` is one
--                 of them. A tag in an expression or a pattern is a finding,

  if kind(at) == "ty-apply" and #ks == 3 and node_name(ks[1]) == "Try" then return ks[3] end
  return 0
end

-- How many times each error declared outside the files checked is named.
const unjudged_errors = {}

for i = 1, #NODES do
  if kind(i) == "s-type-anno" and is_top_level(i) then
    local ks = kids(i)
    if #ks > 0 and kind(ks[1]) == "ty-fn" then
      local parts = kids(ks[1])
      local result = parts[#parts]
      for j = result, last_descendant(result) do
        local err = try_error_node(j)
        if err > 0 and node_name(err) ~= "" and kind(err) == "ty" then
          local name = node_name(err)
          local ek = error_kind(NODE_PATH[i], name, 0)
          if ek == "union" then record_finding(NODE_PATH[i], node_line(i), "error_alias", "`" .. node_name(i) .. "` fails with `" .. name .. "`, a shared alias of tags: list the tags it can return in place, or leave the error to inference (`_`)") end
          if ek == "unknown" then unjudged_errors[name] = get(unjudged_errors, name, 0) + 1 end
        end
      end
    end
  end
end

-- ── optional_tag ─────────────────────────────────────────────────────────

-- The optional tags some type names, and the first use of each optional
-- tag in an expression or pattern, keyed by path and tag.
const optionals_in_types = {}
const first_optional_uses = {}
-- Each `optional_tag` finding as `{path, line, tag}`, for the self-test.
const optional_findings = {}
for i = 1, #NODES do
  local k = kind(i)
  if k == "ty-tag-union" then
    local has_payload = false
    for g in union_tag_nodes(i) do
      if kind(g) == "ty-apply" then has_payload = true end
    end
    for g in union_tag_nodes(i) do
      local t = tag_name(g)
      if has_key(OPTIONALS, t) then
        optionals_in_types[t] = true
        if has_payload then
          push(optional_findings, {path = NODE_PATH[i], line = node_line(g), tag = t})
          record_finding(NODE_PATH[i], node_line(g), "optional_tag", "`" .. t .. "` names an optional: tag what the value is, and its absence by why (`[Latest(Message), NoMessage]`)")
        end
      end
    end
  elseif k == "e-tag" or k == "p-tag" then
    local t = node_raw(i)
    local key = NODE_PATH[i] .. "|" .. t
    if has_key(OPTIONALS, t) and not has_key(first_optional_uses, key) then first_optional_uses[key] = {path = NODE_PATH[i], line = node_line(i), tag = t} end
  end
end
for key in sort(keys(first_optional_uses)) do
  local first_use = first_optional_uses[key]
  if not has_key(optionals_in_types, first_use.tag) then
    push(optional_findings, first_use)
    record_finding(first_use.path, first_use.line, "optional_tag", "`" .. first_use.tag .. "` names an optional, and no type in the files read says what it is: tag the value by what it is, and its absence by why")
  end
end

-- ── set_fallback and placeholder: what a `??` falls back to ──────────────

-- Whether the expressions at `a` and `b` are the same value written twice
-- (`board`, `r.b`): the same kinds, names and fields, all the way down.
function same_expr(a, b)
  if kind(a) ~= kind(b) or node_raw(a) ~= node_raw(b) or get(NODES[a], "field", "") ~= get(NODES[b], "field", "") then return false end
  local ka = kids(a)
  local kb = kids(b)
  if #ka ~= #kb then return false end
  for k = 1, #ka do
    if not same_expr(ka[k], kb[k]) then return false end
  end
  return true
end

-- The list that the `List.set(l, i, v)` or `l.set(i, v)` call at `at`
-- stores into, or 0 for any other expression.
function set_target(at)
  local ks = kids(at)
  if kind(at) == "e-apply" and #ks == 4 and kind(ks[1]) == "e-ident" and node_raw(ks[1]) == "List.set" then return ks[2] end
  if kind(at) == "e-method-call" and get(NODES[at], "method", "") == ".set" then
    for k in ks do
      if kind(k) == "receiver" and #kids(k) == 1 then return kids(k)[1] end
    end
  end
  return 0
end

-- Whether `at` is a string literal with nothing interpolated.
function is_literal_str(at)
  if kind(at) ~= "e-string" then return false end
  for k in kids(at) do
    if kind(k) ~= "e-string-part" then return false end
  end
  return true
end

-- Whether `at` is a lookup by index: `List.get(l, i)` or `l.get(i)`.
function is_index_lookup(at)
  local ks = kids(at)
  if kind(at) == "e-apply" and kind(ks[1]) == "e-ident" then return node_raw(ks[1]) == "List.get" end
  return kind(at) == "e-method-call" and get(NODES[at], "method", "") == ".get"
end

function is_number(at)
  return kind(at) == "e-int" or kind(at) == "e-frac"
end

for i = 1, #NODES do
  local ks = kids(i)
  if kind(i) == "e-binop" and get(NODES[i], "op", "") == "??" and #ks == 2 then
    local lookup = ks[1]
    local fallback = ks[2]
    local path = NODE_PATH[i]
    local target = set_target(lookup)
    if target > 0 and same_expr(target, fallback) then record_finding(path, node_line(fallback), "set_fallback", "`List.set(l, i, v) ?? l`: the fallback names `l`, so every store copies; write `?? crash \"...\"` instead (roc-apps CLAUDE.md, \"Six shapes copy\")") end
    -- With types, the tag union the lookup's `Try` fails with, such as
    -- `[OutOfBounds]`, or "" when it is unknown.
    local has_types = USE_TYPES and has_key(TYPES.real, path)
    local lookup_error = ""
    if has_types then
      local lookup_type = roc_type_at(TYPES, "expr", path, NODES[lookup])
      local m = lookup_type ~= nil and re_find(lookup_type, "^Try\\((.*), (\\[[^\\[\\]]*\\])\\)$") or nil
      if m ~= nil then lookup_error = m.groups[2] end
    end
    local lookup_said = lookup_error ~= "" and (", for a lookup that fails with `" .. lookup_error .. "`") or ""
    if is_literal_str(fallback) then
      local text = ""
      for k in kids(fallback) do text = text .. node_raw(k) end
      record_finding(path, node_line(fallback), "placeholder", "falls back to the literal `\"" .. text .. "\"`" .. (has_types and (", a Str" .. lookup_said) or "") .. ": if the lookup can't fail, `crash` with a reason, \"never a silent `?? default`\"; if it can, keep that case as a tag (roc-apps docs/roc-style.md, \"Errors and failure\")")
    end
    if is_number(fallback) and (lookup_error == "[OutOfBounds]" or (not has_types and is_index_lookup(lookup))) then
      local fallback_type = has_types and (", a " .. (roc_type_at(TYPES, "expr", path, NODES[fallback]) or "?")) or ""
      record_finding(path, node_line(fallback), "number_fallback", "falls back to `" .. node_raw(fallback) .. "`" .. fallback_type .. lookup_said .. ": an index past the end is a bug, not a " .. node_raw(fallback) .. "; `crash` with a reason, or keep the case as a tag (roc-apps docs/roc-style.md, \"Errors and failure\")")
    end
  end
end

-- A plant's name relative to PLANTS. A typed plant's module name is read
-- in snake case: `typed/PlaceholderIndex.roc` is `placeholder_index.roc`.
function plant_name(path)
  local name = sub(path, #PLANTS + 1)
  if starts_with(name, "typed/") then return snake_case_plant_name(sub(name, 7)) end
  return name
end

function is_plant_for(name, check)
  return name == check .. ".roc" or starts_with(name, check .. "_")
end

if SELF_TEST then
  local failures = {}
  for c in CHECKS do
    local planted = 0
    for p in paths do
      local mine = is_plant_for(plant_name(p), c)
      if mine then planted = planted + 1 end
      local did_fire = has_key(fired, p .. "|" .. c)
      if mine and not did_fire then push(failures, "`" .. c .. "` does not fire on its plant " .. p) end
      if did_fire and not mine then push(failures, "`" .. c .. "` fires on " .. p .. ", another check's plant") end
    end
    if planted == 0 then push(failures, "`" .. c .. "` has no plant") end
  end
  -- No check would ever fail on a plant named for none of them.
  for p in paths do
    local name = plant_name(p)
    local named = starts_with(name, "quiet_")
    for c in CHECKS do
      if is_plant_for(name, c) then named = true end
    end
    if not named then push(failures, p .. ": a plant the table does not judge") end
  end
  -- Every error a plant names is declared in a plant; otherwise a quiet
  -- plant could pass without `error_alias` judging it.
  for n in sort(keys(unjudged_errors)) do push(failures, "`error_alias` cannot judge `" .. n .. "`, declared in no plant") end
  for f in optional_findings do
    local source_lines = split(read(f.path), "\n")
    if f.line == "?" or not contains(source_lines[f.line], f.tag) then push(failures, "`optional_tag` says " .. f.path .. ":" .. f.line .. ", a line that does not name `" .. f.tag .. "`") end
  end
  for w in failures do print("FAIL " .. w) end
  if #failures > 0 then fail("check_errors self_test: " .. #failures .. " plants wrong") end
  print("check_errors (Roc) self_test: PASS (each check fires on its plants alone, the quiet_ plants on none, each optional at its tag's line)")
end
