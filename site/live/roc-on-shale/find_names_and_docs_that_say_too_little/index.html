## find_names_and_docs_that_say_too_little, in Roc on Shale's platform (roc_platform/README.md):
## names and top comments that say less than they should. Its launcher,
## cookbook/roc/find_names_and_docs_that_say_too_little.shale, is how it is run, and its header
## says the rules, `pure_mutator`, `abbreviation` and `module_doc`.
## check_goldens.shale, beside this, holds its output on the plants to its
## goldens, line for line.
##
##   cd <repo> && shale cookbook:roc/find_names_and_docs_that_say_too_little.shale glob=app/**/*.roc
##   shale cookbook:roc/find_names_and_docs_that_say_too_little.shale self_test=1
##
## The arguments: `glob` (`**/*.roc`), `abbreviations`, `words`, `accepted`,
## `terms`, `house` (`house=0` leaves out the house style, `abbreviation`) and
## `self_test` (`self_test=1` runs the plants, each through the launcher).
##
## This root reads the word lists, runs each check on each file and prints
## the verdict. Its modules are beside it: one a check (PureMutator,
## Abbreviation, ModuleDoc) and SelfTest. What they read (Tree, Declarations,
## Text) and Finding are the cookbook's Roc library, cookbook/roc/lib/.
## budget: steps=800000000 arena_bytes=2000000000 output_bytes=4000000 roc_cpu_ms=600000 programs=100
app [main!] { lib: "../lib/main.roc", pf: platform "../../../roc_platform/platform/main.roc" }

import pf.Shale
import lib.Declarations
import lib.Finding
import lib.Text
import Abbreviation
import ModuleDoc
import PureMutator
import SelfTest

main! : {} => {}
main! = |{}| {
	if Shale.arg!("self_test", "") == "1" {
		SelfTest.run!(checks)
	} else {
		verdict = run_check!({})
		if !(List.is_empty(verdict.found)) {
			Shale.fail!(verdict.summary)
		}
	}
}

read_word_list! : Str => Str
read_word_list! = |path|
	match Shale.read!(path) {
		Found(text) => text
		Missing => {
			Shale.fail!("no word list ${path}")
			""
		}
	}

## A list of one word per line, any note after `--` cut off, as a set. A
## word listed twice, or a line of two words, stops the run.
word_set! : Str => Set(Str)
word_set! = |path| {
	var $set = Set.empty()
	for l in Text.lines(read_word_list!(path)) {
		word = Str.trim(Text.before_first(l, "--"))
		if Str.contains(word, " ") or Set.contains($set, word) {
			Shale.fail!("${path}: `${word}` is no single new word")
		}
		if word != "" {
			$set = Set.insert($set, word)
		}
	}
	$set
}

## The accepted terms file: a term a line, each with its reason after `--`.
## A line without a reason stops the run.
accepted_terms! : Str => Set(Str)
accepted_terms! = |path| {
	for l in Text.lines(read_word_list!(path)) {
		parts = Str.split_on(l, "--")
		term = Str.trim(List.first(parts) ?? "")
		reason = Str.trim(List.get(parts, 1) ?? "")
		if term != "" and reason == "" {
			Shale.fail!("${path}: `${term}` has no reason after `--`")
		}
	}
	word_set!(path)
}

## Whether `house=0` leaves the house style out (cookbook/roc/house.shale).
house_is_off! : {} => Bool
house_is_off! = |{}| {
	house = Shale.arg!("house", "1")
	if house != "0" and house != "1" {
		Shale.fail!("house must be 1 (the default) or 0, not ${house}")
	}
	house == "0"
}

## The checks, in the order the summary counts them.
checks : List(Str)
checks = ["pure_mutator", "abbreviation", "module_doc"]

## The summary and the finding lines, sorted.
verdict_of : List(Finding), U64 -> { summary : Str, found : List(Str) }
verdict_of = |report, file_count| {
	sorted = List.sort_with(report, Finding.order)
	found = sorted.map(Finding.line_of)
	{ summary: "find_names_and_docs_that_say_too_little (Roc): ${U64.to_str(file_count)} files; ${Finding.counts_of(sorted, checks)}", found }
}

run_check! : {} => { summary : Str, found : List(Str) }
run_check! = |{}| {
	abbreviations = word_set!(Shale.arg!("abbreviations", "cookbook:words/abbreviations.txt"))
	words = word_set!(Shale.arg!("words", "cookbook:words/short_words.txt"))
	accepted_here = accepted_terms!("cookbook:words/accepted_terms.txt")
	project_accepted = Shale.arg!("accepted", "")
	accepted = if project_accepted == "" {
		accepted_here
	} else {
		Set.union(accepted_here, accepted_terms!(project_accepted))
	}
	lists = { abbreviations, words, accepted }
	terms_file = Shale.arg!("terms", "")
	terms = if terms_file == "" {
		Set.empty()
	} else {
		word_set!(terms_file)
	}
	paths = Shale.glob!(Shale.arg!("glob", "**/*.roc"))
	var $report = []
	var $refused = []
	for f in Shale.roc_tree!(paths) {
		if f.err != "" {
			$refused = $refused.append(
				if Str.starts_with(f.err, f.path) {
					f.err
				} else {
					"${f.path}: ${f.err}"
				},
			)
		} else {
			src = match Shale.read!(f.path) {
				Found(text) => Str.split_on(text, "\n")
				Missing => []
			}
			declared = Declarations.from_file({ path: f.path, nodes: f.nodes, source_lines: src })
			aliases = List.fold(
				declared.types,
				Dict.empty(),
				|m, d| if !(d.is_nominal) {
					Dict.insert(m, d.name, d.underlying)
				} else {
					m
				},
			)
			for d in declared.values {
				$report = List.concat($report, PureMutator.judge(f.path, d, aliases, terms))
			}
			if !(house_is_off!({})) {
				$report = List.concat($report, Abbreviation.judge(f.path, f.nodes, declared, lists))
			}
			$report = List.concat($report, ModuleDoc.judge!(f.path, f.nodes, declared, src))
		}
	}
	verdict = verdict_of($report, List.len(paths))
	summary = if house_is_off!({}) {
		"${verdict.summary}; left out as house style: abbreviation"
	} else {
		verdict.summary
	}
	print_findings!(summary, verdict.found)
	if !(List.is_empty($refused)) {
		for r in $refused {
			Shale.print!("refused: ${r}")
		}
		Shale.fail!("find_names_and_docs_that_say_too_little: roc-ast refused ${U64.to_str(List.len($refused))} files (named above), which have no tree to judge; every other file was judged")
	}
	{ summary, found: verdict.found }
}

## The summary, each finding, then the summary again (cookbook/findings.shale).
print_findings! : Str, List(Str) => {}
print_findings! = |summary, found| {
	Shale.print!(summary)
	if !(List.is_empty(found)) {
		for l in found {
			Shale.print!(l)
		}
		Shale.print!(summary)
	}
}

## The `abbreviation` check: a name with a word short for another, read from
## every name a file gives or says (the launcher's header says the rule).
import lib.Declarations
import lib.Finding
import lib.Text
import lib.Tree

Abbreviation := [].{

	## The word lists the check reads: the abbreviations, the English words
	## of two to four letters, and the accepted terms.
	Lists : { abbreviations : Set(Str), words : Set(Str), accepted : Set(Str) }

	## The findings of the file at `path`, a name once a file.
	judge : Str, List(Tree.Node), Declarations, Abbreviation.Lists -> List(Finding)
	judge = |path, tree, declared, lists| {
		var $seen = Set.empty()
		var $report = []
		for given in names_given(path, tree, declared) {
			if !(Set.contains($seen, given.name)) {
				$seen = Set.insert($seen, given.name)
				found = Text.name_words(given.name).keep_if(|w| is_abbreviated(w, lists))
				if !(List.is_empty(found)) {
					$report = $report.append({ path, line: given.line, check: "abbreviation", what: "`${given.name}` abbreviates: ${Str.join_with(found, ", ")}" })
				}
			}
		}
		$report
	}
}

## Whether a short word stands as it is: an English word, an accepted term,
## or a letter and an axis (`dx`, `vy`).
is_plain_short_word : Str, Abbreviation.Lists -> Bool
is_plain_short_word = |w, lists| {
	if Set.contains(lists.words, w) or Set.contains(lists.accepted, w) {
		Bool.True
	} else {
		bytes = Str.to_utf8(w)
		second = List.get(bytes, 1) ?? 0
		List.len(bytes) == 2 and (second == 120 or second == 121 or second == 122)
	}
}

## Whether a word of two to four letters is short for another.
is_short_abbreviation : Str, Abbreviation.Lists -> Bool
is_short_abbreviation = |w, lists| {
	bytes = Str.to_utf8(w)
	n = List.len(bytes)
	if n < 2 or n > 4 or !(List.all(bytes, Text.is_small_letter)) {
		Bool.False
	} else {
		!(is_plain_short_word(w, lists))
	}
}

## The stems a word may be the plural of (`shapes`: `shape`, `shap`; `tries`:
## `trie`, `tri`, `try`), none for a word ending as no plural does.
plural_stems : Str -> List(Str)
plural_stems = |w| {
	n = w.count_utf8_bytes()
	no_plural = ["ss", "us", "is", "as", "os"].any(|e| Str.ends_with(w, e))
	if n < 3 or !(Str.ends_with(w, "s")) or no_plural {
		return []
	}
	one = [Text.bytes_of(w, { start: 0, count: n - 1 })]
	two = if Str.ends_with(w, "es") and n > 3 {
		one.append(Text.bytes_of(w, { start: 0, count: n - 2 }))
	} else {
		one
	}
	if Str.ends_with(w, "ies") and n > 4 {
		two.append("${Text.bytes_of(w, { start: 0, count: n - 3 })}y")
	} else {
		two
	}
}

## Whether a stem is a whole word, so its plural is plain.
is_whole_stem : Str, Abbreviation.Lists -> Bool
is_whole_stem = |stem, lists|
	if Set.contains(lists.accepted, stem) {
		Bool.True
	} else if stem.count_utf8_bytes() <= 4 {
		is_plain_short_word(stem, lists)
	} else {
		!(Set.contains(lists.abbreviations, stem))
	}

## Whether a word of a name is an abbreviation (the launcher's header says
## the rule).
is_abbreviated : Str, Abbreviation.Lists -> Bool
is_abbreviated = |word, lists| {
	letters = Str.to_utf8(word).keep_if(|b| Text.is_small_letter(b) or Text.is_digit(b))
	whole = Str.from_utf8_lossy(letters)
	if Set.contains(lists.accepted, whole) {
		return Bool.False
	}
	if Set.contains(lists.abbreviations, whole) {
		return Bool.True
	}
	var $end = List.len(letters)
	while $end > 0 and Text.is_digit(List.get(letters, $end - 1) ?? 0) {
		$end = $end - 1
	}
	w = Str.from_utf8_lossy(List.take_first(letters, $end))
	if Set.contains(lists.accepted, w) {
		return Bool.False
	}
	if Set.contains(lists.abbreviations, w) {
		return Bool.True
	}
	if w.count_utf8_bytes() <= 4 and is_plain_short_word(w, lists) {
		return Bool.False
	}
	stems = plural_stems(w)
	if stems.any(|stem| is_whole_stem(stem, lists)) {
		Bool.False
	} else if stems.any(|stem| Set.contains(lists.abbreviations, stem) or is_short_abbreviation(stem, lists)) {
		Bool.True
	} else {
		is_short_abbreviation(w, lists)
	}
}

## The tag a node names, `""` for none: a tag in a union, a pattern or an
## expression, a module's qualified tag by its last part.
tag_named : List(Tree.Node), U64 -> Str
tag_named = |tree, at| {
	n = Tree.node_at(tree, at)
	if n.kind == "p-tag" or n.kind == "e-tag" {
		return Tree.last_part(Tree.attribute(n, "raw"))
	}
	if n.kind != "ty" or n.parent == 0 {
		return ""
	}
	holder = Tree.node_at(tree, n.parent)
	if holder.kind == "tags" {
		Tree.attribute(n, "name")
	} else if holder.kind == "ty-apply" and List.first(holder.kids) == Ok(at) and Tree.node_at(tree, holder.parent).kind == "tags" {
		Tree.attribute(n, "name")
	} else {
		""
	}
}

## The names a file gives or says, with the line each appears on: the
## module's, what it declares in file order, then its parameters, locals and
## tags.
names_given : Str, List(Tree.Node), Declarations -> List({ name : Str, line : U64 })
names_given = |path, tree, declared| {
	var $out = if Tree.is_app(tree) {
		[]
	} else {
		[{ name: Tree.module_name(path), line: 1 }]
	}
	for d in declared_in_file_order(declared) {
		$out = $out.append({ name: d.name, line: d.line })
	}
	var $i = 0
	for n in tree {
		$i = $i + 1
		if n.kind == "p-ident" {
			$out = $out.append({ name: Tree.attribute(n, "raw"), line: n.line })
		}
		tag = tag_named(tree, $i)
		if tag != "" and tag != "Ok" and tag != "Err" {
			$out = $out.append({ name: tag, line: n.line })
		}
	}
	$out
}

## What the file declares, its types, fields and values together, in file
## order: by the place of each one's node.
declared_in_file_order : Declarations -> List({ place : U64, name : Str, line : U64 })
declared_in_file_order = |declared| {
	types = declared.types.map(|d| { place: d.place, name: d.name, line: d.line })
	fields = declared.fields.map(|d| { place: d.place, name: d.name, line: d.line })
	values = declared.values.map(|d| { place: d.place, name: d.name, line: d.line })
	types.concat(fields).concat(values).sort_with(
		|first, second|
			if first.place < second.place {
				Before
			} else if first.place > second.place {
				After
			} else {
				Same
			},
	)
}

## The `module_doc` check: a module with no `##` doc, or a top comment or a
## doc above its type that tells history (the launcher's header says the
## rule). A package's or platform's main.roc says which modules it keeps to
## itself, so it reads files through the platform.
import pf.Shale
import lib.Declarations
import lib.Finding
import lib.Headers
import lib.Text
import lib.Tree

ModuleDoc := [].{

	## The findings of the file at `path`, none for an app.
	judge! : Str, List(Tree.Node), Declarations, List(Str) => List(Finding)
	judge! = |path, tree, declared, src| {
		if Tree.is_app(tree) {
			return []
		}
		name = Tree.module_name(path)
		type_line = List.fold(
			declared.types,
			0,
			|at, d| if d.name == name and d.owner == "" {
				d.line
			} else {
				at
			},
		)
		if type_line == 0 {
			return []
		}
		top = opening_comment(src)
		above_type = if type_line > 1 {
			Str.trim(Text.line_at(src, type_line - 1))
		} else {
			""
		}
		opens_with_doc = Str.starts_with(List.first(top) ?? "", "##")
		has_doc = opens_with_doc or (type_line > 1 and Str.starts_with(above_type, "##"))
		kept_to_itself = (
			!(List.is_empty(top)) or Str.starts_with(
				Str.trim(
					Text.line_at(
						src,
						if type_line > 1 {
							type_line - 1
						} else {
							1
						},
					),
				),
				"#",
			)
		) and is_private_module!(path)
		var $report = []
		if !(has_doc) and !(kept_to_itself) {
			$report = $report.append({ path, line: 1, check: "module_doc", what: "`${name}` has no `##` doc saying what it is for" })
		}
		history = history_word!(top)
		if history != "" {
			$report = $report.append({ path, line: 1, check: "module_doc", what: "`${name}`'s top comment says `${history}`: say what the module does, and leave history to git" })
		}
		doc = comment_above(src, type_line)
		doc_history = history_word!(doc.lines)
		if doc.first > 1 and doc_history != "" {
			$report = $report.append({ path, line: doc.first, check: "module_doc", what: "`${name}`'s doc says `${doc_history}`: say what the module does, and leave history to git" })
		}
		$report
	}
}

## The history word in the comment lines, or `""`. Each line's leading `#`s
## go before the lines are joined, so a phrase wrapped across two lines reads
## as one.
history_word! : List(Str) => Str
history_word! = |comment| {
	var $words = []
	for l in comment {
		$words = $words.append(Str.trim(Shale.re_replace!(l, "^#+", "")))
	}
	match Shale.re_find!(Str.with_ascii_lowercased(Str.join_with($words, " ")), "\\b(originally|used to|emitted|ported)\\b") {
		Found(m) => List.first(m.groups) ?? ""
		NoMatch => ""
	}
}

## The comment lines just above line `at`, trimmed, and the line they start
## on (`at` when there are none).
comment_above : List(Str), U64 -> { lines : List(Str), first : U64 }
comment_above = |src, at| {
	var $first = at
	while $first > 1 and Str.starts_with(Str.trim(Text.line_at(src, $first - 1)), "#") {
		$first = $first - 1
	}
	{ lines: List.sublist(src, { start: $first - 1, len: at - $first }).map(Str.trim), first: $first }
}

## The comment lines that open a file, trimmed.
opening_comment : List(Str) -> List(Str)
opening_comment = |src| {
	var $out = []
	var $going = Bool.True
	for l in src {
		if $going and Str.starts_with(Str.trim(l), "#") {
			$out = $out.append(Str.trim(l))
		} else {
			$going = Bool.False
		}
	}
	$out
}

## What a folder's `main.roc` exposes, when it is a package's or a
## platform's (cookbook/roc/lib/'s `Headers.exposed_by`).
exposed_by_folder! : Str => { is_package : Bool, modules : List(Str) }
exposed_by_folder! = |dir| {
	main = if dir == "" or dir == "." {
		"main.roc"
	} else {
		"${dir}/main.roc"
	}
	match Shale.read!(main) {
		Missing => { is_package: Bool.False, modules: [] }
		Found(text) => Headers.exposed_by(text)
	}
}

## Whether a file is a module its package keeps to itself.
is_private_module! : Str => Bool
is_private_module! = |path| {
	exposed = exposed_by_folder!(Text.dirname(path))
	exposed.is_package and !(List.contains(exposed.modules, Tree.module_name(path))) and Text.basename(path) != "main.roc"
}

## The `pure_mutator` check: an annotated pure function whose first word
## says it changes something though nothing is being added to (the
## launcher's header says the rule).
import lib.Declarations
import lib.Finding
import lib.Text
import lib.Tree

PureMutator := [].{

	## The finding for value `d` of the file at `path`, if it is one.
	## `aliases` are the file's structural type aliases, by name, and `terms`
	## the project's fixed terms that only look like a verb.
	judge : Str, Declarations.ValueDeclared, Dict(Str, Str), Set(Str) -> List(Finding)
	judge = |path, d, aliases, terms| {
		if !(d.is_function) or !(d.annotated) or Str.ends_with(d.name, "!") or opens_with_term(d.name, terms) {
			return []
		}
		first_word = Text.before_first(d.name, "_")
		is_mutation = is_mutation_word(first_word) and !(gives_first_back(d, aliases))
		if first_word == "get" or is_mutation {
			[{ path, line: d.line, check: "pure_mutator", what: "`${d.name}` is pure and gives a new `${d.result_type}`: name it for what it gives" }]
		} else {
			[]
		}
	}
}

## The value of a `Try(value, errors)`, or `result` when it is no `Try`.
value_of_try : Str -> Str
value_of_try = |result| {
	if !(Str.starts_with(result, "Try(")) {
		return result
	}
	bytes = Str.to_utf8(result)
	var $depth = 0
	var $k = 4
	while $k < List.len(bytes) {
		b = List.get(bytes, $k) ?? 0
		if b == 40 or b == 123 or b == 91 {
			$depth = $depth + 1
		} else if b == 41 or b == 125 or b == 93 {
			if $depth == 0 {
				return result
			}
			$depth = $depth - 1
		} else if b == 44 and $depth == 0 {
			return Str.trim(Text.bytes_of(result, { start: 4, count: $k - 4 }))
		}
		$k = $k + 1
	}
	result
}

record_holds : Str, Str -> Bool
record_holds = |record, type_name|
	Str.starts_with(record, "{") and (Str.contains(record, " : ${type_name},") or Str.contains(record, " : ${type_name} }"))

## Whether a function's result is what its first argument was added to.
gives_first_back : Declarations.ValueDeclared, Dict(Str, Str) -> Bool
gives_first_back = |d, aliases| {
	value = value_of_try(d.result_type)
	if value == d.first_type or record_holds(d.first_type, value) {
		Bool.True
	} else {
		record_holds(Dict.get(aliases, Tree.last_part(value)) ?? value, d.first_type)
	}
}

opens_with_term : Str, Set(Str) -> Bool
opens_with_term = |name, terms| Set.to_list(terms).any(|term| name == term or Str.starts_with(name, "${term}_"))

is_mutation_word : Str -> Bool
is_mutation_word = |w| w == "push" or w == "fill" or w == "set" or w == "append" or w == "add"

## The recipe's self-test: each plant under
## cookbook/roc/plants/find_names_and_docs_that_say_too_little/ run through the launcher, held to
## its findings, its exit and what it says.
import pf.Shale
import lib.Text

SelfTest := [].{

	## Each row run as a child of its own, through `launcher`; a check of
	## `checks` no row fires fails too, since it could be blind.
	run! : List(Str) => {}
	run! = |checks| {
		var $wrong = 0.U64
		var $fired = Set.empty()
		for row in rows {

function value_of_try(result)
  if not starts_with(result, "Try(") then return result end
  local depth = 0
  for i = 5, #result do
    local c = sub(result, i, i)
    if c == "(" or c == "{" or c == "[" then
      depth = depth + 1
    elseif c == ")" or c == "}" or c == "]" then
      depth = depth - 1
    elseif c == "," and depth == 0 then

    end
  end
  return result
end

-- The stems a word may be the plural of, each way English makes one:
-- `shapes` gives `shape` and `shap`, `boxes` `boxe` and `box`, and `tries`

function plural_stems(w)

  if ends_with(w, "es") and #w > 3 then push(stems, sub(w, 1, #w - 2)) end

-- project accepts never is, and one the abbreviations file lists always is,
-- whatever its length (`delim`, `params`). A plural is judged by every
-- stem it could have (`plural_stems`): it is plain when any stem is a whole
-- word (`ids`, `boxes`, `shapes`, `tries`), and else an abbreviation when a
-- stem is one (`attrs`, `delims`). Past those, a word of two to four
-- letters that is no plain word is one. Its marks (`!`, `$`) and trailing
-- digits (`buf0`) go first; a letter alone is `short_name`'s, in hunt_names.

  if has_key(lists.abbreviations, w) then return true end
  w = re_replace(w, "[0-9]+$", "")

function run_check()
  local lists = {abbreviations = word_set(ABBREVIATIONS_FILE), words = word_set(SHORT_WORDS_FILE), accepted = accepted_terms(ACCEPTED_FILE)}
  if PROJECT_ACCEPTED_FILE ~= "" then
    for term in keys(accepted_terms(PROJECT_ACCEPTED_FILE)) do lists.accepted[term] = true end
  end
  local terms = TERMS_FILE == "" and {} or word_set(TERMS_FILE)
  local paths = glob(GLOB)
  local parsed = parse_roc_files(paths)
  local report = {}
  for f in parsed.files do
    local declared = declarations_of(f)
    local aliases = {}
    for d in declared do
      if d.kind == "type" and not d.is_nominal then aliases[d.name] = d.underlying end
    end
    for d in declared do
      if d.kind == "value" then judge_mutator(d, aliases, terms, report) end
    end
    if not house_leaves_out("find_unclear_names", "abbreviation") then judge_abbreviations(f, declared, lists, report) end
    judge_module_doc(f, declared, report)
  end
  local verdict = verdict_of("find_unclear_names", CHECKS, report, #paths)
  verdict.summary = verdict.summary .. house_note("find_unclear_names")
  print_findings(verdict.summary, verdict.found)
  fail_if_refused("find_unclear_names", parsed.refused)
  return verdict
end

-- This `fail` stands at the top level so the summary is the last line a
-- caller reads: a stop inside a function prints the calls after it.
if SELF_TEST then
  self_test()
else
  local verdict = run_check()
  if #verdict.found > 0 then fail(verdict.summary) end
end
