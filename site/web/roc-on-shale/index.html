Roc on Shale

Shale is a small scripting language for the work an agent does on a machine. Every run has budgets for its steps, its reads and its output, reads only beneath the folder it is rooted at, and runs a program only when its rules allow it.

Shale can also run a Roc app as a script. The app is built on a Roc platform whose host is Shale's own runtime, so each effect it makes (reading a file, listing a folder, running a program, printing) is a call into Shale, held to the same rules and budgets as a Shale script. Everything else is plain Roc. The platform's README explains how it works.

A recipe that is kept for a long time is worth writing in Roc: its types catch mistakes before it runs, its tests sit beside its code, and its modules can be shared. A one-off stays in Shale. Each example below is a recipe from Shale's cookbook in Roc, followed by the Shale script it was ported from, each file in full and colored by roc-syntax-color. The links at each example's top and end skip to the one before or after it.

## find_unclear_names, in Roc on Shale's platform (roc_platform/README.md):
## names and top comments that say less than they should. Its launcher,
## cookbook/roc/find_unclear_names.shale, is how it is run, and its header
## says the rules, `pure_mutator`, `abbreviation` and `module_doc`.
## check_goldens.shale, beside this, holds its output on the plants to its
## goldens, line for line.
##
##   cd <repo> && shale cookbook:roc/find_unclear_names.shale glob=app/**/*.roc
##   shale cookbook:roc/find_unclear_names.shale self_test=1
##
## The arguments: `glob` (`**/*.roc`), `abbreviations`, `words`, `accepted`,
## `terms`, `house` (`house=0` leaves out the house style, `abbreviation`) and
## `self_test` (`self_test=1` runs the plants, each through the launcher).
##
## This root reads the word lists, runs each check on each file and prints
## the verdict. Its modules are beside it: one a check (PureMutator,
## Abbreviation, ModuleDoc), what they read (Tree, Declarations, Text),
## Finding, and SelfTest.
## budget: steps=800000000 arena_bytes=2000000000 output_bytes=4000000 roc_cpu_ms=600000 programs=100
app [main!] { pf: platform "../../../roc_platform/platform/main.roc" }

import pf.Shale
import Abbreviation
import Declarations
import Finding
import ModuleDoc
import PureMutator
import SelfTest
import Text

main! : {} => {}
main! = |{}| {
	if Shale.arg!("self_test", "") == "1" {
		SelfTest.run!({})
	} else {
		verdict = run_check!({})
		if !(List.is_empty(verdict.found)) {
			Shale.fail!(verdict.summary)
		}
	}
}

read_word_list! : Str => Str
read_word_list! = |path|
	match Shale.read!(path) {
		Found(text) => text
		Missing => {
			Shale.fail!("no word list ${path}")
			""
		}
	}

## A list of one word per line, any note after `--` cut off, as a set. A
## word listed twice, or a line of two words, stops the run.
word_set! : Str => Set(Str)
word_set! = |path| {
	var $set = Set.empty()
	for l in Text.lines(read_word_list!(path)) {
		word = Str.trim(Text.before_first(l, "--"))
		if Str.contains(word, " ") or Set.contains($set, word) {
			Shale.fail!("${path}: `${word}` is no single new word")
		}
		if word != "" {
			$set = Set.insert($set, word)
		}
	}
	$set
}

## The accepted terms file: a term a line, each with its reason after `--`.
## A line without a reason stops the run.
accepted_terms! : Str => Set(Str)
accepted_terms! = |path| {
	for l in Text.lines(read_word_list!(path)) {
		parts = Str.split_on(l, "--")
		term = Str.trim(List.first(parts) ?? "")
		reason = Str.trim(List.get(parts, 1) ?? "")
		if term != "" and reason == "" {
			Shale.fail!("${path}: `${term}` has no reason after `--`")
		}
	}
	word_set!(path)
}

## Whether `house=0` leaves the house style out (cookbook/roc/house.shale).
house_is_off! : {} => Bool
house_is_off! = |{}| {
	house = Shale.arg!("house", "1")
	if house != "0" and house != "1" {
		Shale.fail!("house must be 1 (the default) or 0, not ${house}")
	}
	house == "0"
}

## The summary and the finding lines, sorted (cookbook/roc/gate_recipe.shale's
## `verdict_of`).
verdict_of : List(Finding), U64 -> { summary : Str, found : List(Str) }
verdict_of = |report, file_count| {
	sorted = List.sort_with(report, Finding.order)
	counts = Finding.checks.map(|c| "${c} ${U64.to_str(List.count_if(sorted, |r| r.check == c))}")
	found = sorted.map(|r| "${r.path}:${U64.to_str(r.line)}: ${r.check}: ${r.what}")
	{ summary: "find_unclear_names (Roc): ${U64.to_str(file_count)} files; ${Str.join_with(counts, ", ")}", found }
}

run_check! : {} => { summary : Str, found : List(Str) }
run_check! = |{}| {
	abbreviations = word_set!(Shale.arg!("abbreviations", "cookbook:words/abbreviations.txt"))
	words = word_set!(Shale.arg!("words", "cookbook:words/short_words.txt"))
	accepted_here = accepted_terms!("cookbook:words/accepted_terms.txt")
	project_accepted = Shale.arg!("accepted", "")
	accepted = if project_accepted == "" {
		accepted_here
	} else {
		Set.union(accepted_here, accepted_terms!(project_accepted))
	}
	lists = { abbreviations, words, accepted }
	terms_file = Shale.arg!("terms", "")
	terms = if terms_file == "" {
		Set.empty()
	} else {
		word_set!(terms_file)
	}
	paths = Shale.glob!(Shale.arg!("glob", "**/*.roc"))
	var $report = []
	var $refused = []
	for f in Shale.roc_tree!(paths) {
		if f.err != "" {
			$refused = $refused.append(
				if Str.starts_with(f.err, f.path) {
					f.err
				} else {
					"${f.path}: ${f.err}"
				},
			)
		} else {
			src = match Shale.read!(f.path) {
				Found(text) => Str.split_on(text, "\n")
				Missing => []
			}
			declared = Declarations.of(f.nodes, src)
			aliases = List.fold(
				declared,
				Dict.empty(),
				|m, d| if d.kind == Type and !(d.is_nominal) {
					Dict.insert(m, d.name, d.underlying)
				} else {
					m
				},
			)
			for d in declared {
				if d.kind == Value {
					$report = List.concat($report, PureMutator.judge(f.path, d, aliases, terms))
				}
			}
			if !(house_is_off!({})) {
				$report = List.concat($report, Abbreviation.judge(f.path, f.nodes, declared, lists))
			}
			$report = List.concat($report, ModuleDoc.judge!(f.path, f.nodes, declared, src))
		}
	}
	verdict = verdict_of($report, List.len(paths))
	summary = if house_is_off!({}) {
		"${verdict.summary}; left out as house style: abbreviation"
	} else {
		verdict.summary
	}
	print_findings!(summary, verdict.found)
	if !(List.is_empty($refused)) {
		for r in $refused {
			Shale.print!("refused: ${r}")
		}
		Shale.fail!("find_unclear_names: roc-ast refused ${U64.to_str(List.len($refused))} files (named above), which have no tree to judge; every other file was judged")
	}
	{ summary, found: verdict.found }
}

## The summary, each finding, then the summary again (cookbook/findings.shale).
print_findings! : Str, List(Str) => {}
print_findings! = |summary, found| {
	Shale.print!(summary)
	if !(List.is_empty(found)) {
		for l in found {
			Shale.print!(l)
		}
		Shale.print!(summary)
	}
}

## The `abbreviation` check: a name with a word short for another, read from
## every name a file gives or says (the launcher's header says the rule).
import Declarations
import Finding
import Text
import Tree

Abbreviation := [].{

	## The word lists the check reads: the abbreviations, the English words
	## of two to four letters, and the accepted terms.
	Lists : { abbreviations : Set(Str), words : Set(Str), accepted : Set(Str) }

	## The findings of the file at `path`, a name once a file.
	judge : Str, List(Tree.Node), List(Declarations.Declared), Abbreviation.Lists -> List(Finding)
	judge = |path, tree, declared, lists| {
		var $seen = Set.empty()
		var $report = []
		for given in names_given(path, tree, declared) {
			if !(Set.contains($seen, given.name)) {
				$seen = Set.insert($seen, given.name)
				found = name_words(given.name).keep_if(|w| is_abbreviated(w, lists))
				if !(List.is_empty(found)) {
					$report = $report.append({ path, line: given.line, check: "abbreviation", what: "`${given.name}` abbreviates: ${Str.join_with(found, ", ")}" })
				}
			}
		}
		$report
	}
}

## A name's words, lowercased: a type's split at its capitals (`NodeInfo`
## gives `node`, `info`), a field's at its underscores.
name_words : Str -> List(Str)
name_words = |name| {
	var $out = []
	var $current = []
	for b in Str.to_utf8(name) {
		if b == 95 {
			if !(List.is_empty($current)) {
				$out = $out.append(Text.lower_bytes($current))
			}
			$current = []
		} else if Text.is_upper(b) and !(List.is_empty($current)) {
			$out = $out.append(Text.lower_bytes($current))
			$current = [b]
		} else {
			$current = $current.append(b)
		}
	}
	if List.is_empty($current) {
		$out
	} else {
		$out.append(Text.lower_bytes($current))
	}
}

## Whether a short word stands as it is: an English word, an accepted term,
## or a letter and an axis (`dx`, `vy`).
is_plain_short_word : Str, Abbreviation.Lists -> Bool
is_plain_short_word = |w, lists| {
	if Set.contains(lists.words, w) or Set.contains(lists.accepted, w) {
		Bool.True
	} else {
		bytes = Str.to_utf8(w)
		second = List.get(bytes, 1) ?? 0
		List.len(bytes) == 2 and (second == 120 or second == 121 or second == 122)
	}
}

## Whether a word of two to four letters is short for another.
is_short_abbreviation : Str, Abbreviation.Lists -> Bool
is_short_abbreviation = |w, lists| {
	bytes = Str.to_utf8(w)
	n = List.len(bytes)
	if n < 2 or n > 4 or !(List.all(bytes, Text.is_lower)) {
		Bool.False
	} else {
		!(is_plain_short_word(w, lists))
	}
}

## The stems a word may be the plural of (`shapes`: `shape`, `shap`; `tries`:
## `trie`, `tri`, `try`), none for a word ending as no plural does.
plural_stems : Str -> List(Str)
plural_stems = |w| {
	n = Text.byte_count(w)
	no_plural = ["ss", "us", "is", "as", "os"].any(|e| Str.ends_with(w, e))
	if n < 3 or !(Str.ends_with(w, "s")) or no_plural {
		return []
	}
	one = [Text.bytes_of(w, 0, n - 1)]
	two = if Str.ends_with(w, "es") and n > 3 {
		one.append(Text.bytes_of(w, 0, n - 2))
	} else {
		one
	}
	if Str.ends_with(w, "ies") and n > 4 {
		two.append("${Text.bytes_of(w, 0, n - 3)}y")
	} else {
		two
	}
}

## Whether a stem is a whole word, so its plural is plain.
is_whole_stem : Str, Abbreviation.Lists -> Bool
is_whole_stem = |stem, lists|
	if Set.contains(lists.accepted, stem) {
		Bool.True
	} else if Text.byte_count(stem) <= 4 {
		is_plain_short_word(stem, lists)
	} else {
		!(Set.contains(lists.abbreviations, stem))
	}

## Whether a word of a name is an abbreviation (the launcher's header says
## the rule).
is_abbreviated : Str, Abbreviation.Lists -> Bool
is_abbreviated = |word, lists| {
	letters = Str.to_utf8(word).keep_if(|b| Text.is_lower(b) or Text.is_digit(b))
	whole = Str.from_utf8_lossy(letters)
	if Set.contains(lists.accepted, whole) {
		return Bool.False
	}
	if Set.contains(lists.abbreviations, whole) {
		return Bool.True
	}
	var $end = List.len(letters)
	while $end > 0 and Text.is_digit(List.get(letters, $end - 1) ?? 0) {
		$end = $end - 1
	}
	w = Str.from_utf8_lossy(List.take_first(letters, $end))
	if Set.contains(lists.accepted, w) {
		return Bool.False
	}
	if Set.contains(lists.abbreviations, w) {
		return Bool.True
	}
	if Text.byte_count(w) <= 4 and is_plain_short_word(w, lists) {
		return Bool.False
	}
	stems = plural_stems(w)
	if stems.any(|stem| is_whole_stem(stem, lists)) {
		Bool.False
	} else if stems.any(|stem| Set.contains(lists.abbreviations, stem) or is_short_abbreviation(stem, lists)) {
		Bool.True
	} else {
		is_short_abbreviation(w, lists)
	}
}

## The tag a node names, `""` for none: a tag in a union, a pattern or an
## expression, a module's qualified tag by its last part.
tag_named : List(Tree.Node), U64 -> Str
tag_named = |tree, at| {
	n = Tree.node_at(tree, at)
	if n.kind == "p-tag" or n.kind == "e-tag" {
		return Tree.last_part(Tree.attribute(n, "raw"))
	}
	if n.kind != "ty" or n.parent == 0 {
		return ""
	}
	holder = Tree.node_at(tree, n.parent)
	if holder.kind == "tags" {
		Tree.attribute(n, "name")
	} else if holder.kind == "ty-apply" and List.first(holder.kids) == Ok(at) and Tree.node_at(tree, holder.parent).kind == "tags" {
		Tree.attribute(n, "name")
	} else {
		""
	}
}

## The names a file gives or says, with the line each appears on.
names_given : Str, List(Tree.Node), List(Declarations.Declared) -> List({ name : Str, line : U64 })
names_given = |path, tree, declared| {
	var $out = if Tree.is_app(tree) {
		[]
	} else {
		[{ name: Tree.module_name(path), line: 1 }]
	}
	for d in declared {
		$out = $out.append({ name: d.name, line: d.line })
	}
	var $i = 0
	for n in tree {
		$i = $i + 1
		if n.kind == "p-ident" {
			$out = $out.append({ name: Tree.attribute(n, "raw"), line: n.line })
		}
		tag = tag_named(tree, $i)
		if tag != "" and tag != "Ok" and tag != "Err" {
			$out = $out.append({ name: tag, line: n.line })
		}
	}
	$out
}

## What a file declares at its top level or as a module type's member, read
## from its tree in file order (cookbook/roc/declared.shale's
## `declarations_of`): what the checks judge.
import Text
import Tree

Declarations := [].{

	## A type, a value (an annotation, or a definition with none above it) or a
	## record field of a type or an annotation.
	Declared : {
		line : U64,
		kind : [Type, Value, Field],
		name : Str,
		owner : Str,
		annotated : Bool,
		fn : Bool,
		result : Str,
		first : Str,
		is_nominal : Bool,
		underlying : Str,
	}

	## Everything the file declares at its top level or as a module type's
	## member, in file order.
	of : List(Tree.Node), List(Str) -> List(Declarations.Declared)
	of = |tree, src| {
		var $outer = []
		var $declaration_lines = []
		var $annotated = Set.empty()
		var $out = []
		var $i = 0
		for n in tree {
			$i = $i + 1
			i = $i
			parent_outer = n.parent != 0 and (List.get($outer, n.parent - 1) ?? Bool.False) and is_outer_kind(Tree.node_at(tree, n.parent).kind)
			is_outer = n.parent == 0 or parent_outer
			$outer = $outer.append(is_outer)
			kind = n.kind
			$declaration_lines = $declaration_lines.append(
				if is_outer and (kind == "s-type-anno" or kind == "s-type-decl") {
					n.line
				} else {
					0
				},
			)
			if is_outer and kind == "s-type-decl" {
				$out = List.concat($out, declared_type(tree, src, i))
			}
			if kind == "anno-record-field" and field_declaration_line(tree, $declaration_lines, i) != 0 {
				$out = $out.append(plain(n.line, Field, Tree.attribute(n, "name"), owner_of(tree, i)))
			}
			if is_outer and kind == "s-type-anno" {
				for d in annotated_value(tree, i) {
					$out = $out.append(d)
					$annotated = Set.insert($annotated, "${d.owner}.${d.name}")
				}
			}
			if is_outer and kind == "s-decl" and Tree.kid_kind(tree, n, 0) == "p-ident" {
				name = Tree.attribute(Tree.node_at(tree, Tree.first_kid(n)), "raw")
				owner = owner_of(tree, i)
				if !(Set.contains($annotated, "${owner}.${name}")) {
					$out = $out.append({ ..plain(n.line, Value, name, owner), fn: Tree.kid_kind(tree, n, 1) == "e-lambda" })
				}
			}
		}
		$out
	}
}

## The kinds a definition may sit in and still be top-level or a module
## type's member.
is_outer_kind : Str -> Bool
is_outer_kind = |kind| kind == "file" or kind == "statements" or kind == "s-type-decl" or kind == "associated"

## The name of the type declaration around node `at`, or "" at the top level.
owner_of : List(Tree.Node), U64 -> Str
owner_of = |tree, at| {
	var $k = Tree.node_at(tree, at).parent
	while $k != 0 {
		n = Tree.node_at(tree, $k)
		if n.kind == "s-type-decl" {
			return Tree.attribute(Tree.node_at(tree, Tree.first_kid(n)), "name")
		}
		$k = n.parent
	}
	""
}

## The line of the declaration a record field belongs to, as
## `declaration_lines` holds it, or 0 when an expression, a pattern or a
## definition lies between them (the field is then a local's).
field_declaration_line : List(Tree.Node), List(U64), U64 -> U64
field_declaration_line = |tree, declaration_lines, at| {
	var $k = Tree.node_at(tree, at).parent
	while $k != 0 {
		n = Tree.node_at(tree, $k)
		if Str.starts_with(n.kind, "e-") or Str.starts_with(n.kind, "p-") or n.kind == "s-decl" {
			return 0
		}
		if n.kind == "s-type-anno" or n.kind == "s-type-decl" {
			return List.get(declaration_lines, $k - 1) ?? 0
		}
		$k = n.parent
	}
	0
}

plain : U64, [Type, Value, Field], Str, Str -> Declarations.Declared
plain = |line, kind, name, owner| { line, kind, name, owner, annotated: Bool.False, fn: Bool.False, result: "", first: "", is_nominal: Bool.False, underlying: "" }

## The annotation at `i`, unless it is only a `where` clause.
annotated_value : List(Tree.Node), U64 -> List(Declarations.Declared)
annotated_value = |tree, i| {
	n = Tree.node_at(tree, i)
	match List.first(n.kids) {
		Err(_) => []
		Ok(type_at) => {
			type_node = Tree.node_at(tree, type_at)
			if type_node.kind == "where" {
				return []
			}
			base = { ..plain(n.line, Value, Tree.attribute(n, "name"), owner_of(tree, i)), annotated: Bool.True, result: Tree.render_type(tree, type_at) }
			if type_node.kind == "ty-fn" {
				first = if List.len(type_node.kids) > 1 {
					Tree.render_type(tree, Tree.first_kid(type_node))
				} else {
					""
				}
				[{ ..base, fn: Bool.True, result: Tree.render_type(tree, Tree.last_kid(type_node)), first }]
			} else {
				[base]
			}
		}
	}
}

## The type declared at `i`: `is_nominal` says it is declared with `:=` or
## `::`, and `underlying` is the type it is built on.
declared_type : List(Tree.Node), List(Str), U64 -> List(Declarations.Declared)
declared_type = |tree, src, i| {
	n = Tree.node_at(tree, i)
	name_node = Tree.node_at(tree, Tree.first_kid(n))
	name = Tree.attribute(name_node, "name")
	if name == "" {
		return []
	}
	base = plain(n.line, Type, name, owner_of(tree, i))
	match List.get(n.kids, 1) {
		Ok(built_at) =>
			if Tree.node_at(tree, built_at).kind == "associated" {
				[base]
			} else {
				between_text = Text.bytes_of(Text.line_at(src, name_node.end_line), name_node.end_col - 1, 4)
				is_nominal = Str.contains(between_text, ":=") or Str.contains(between_text, "::")
				[{ ..base, is_nominal, underlying: Tree.render_type(tree, built_at) }]
			}
		Err(_) => [base]
	}
}

## A finding of one of the recipe's checks: the file, the line and what is
## unclear there, ordered as the Shale original sorted its report.
import Text

Finding := { path : Str, line : U64, check : Str, what : Str }.{

	## The checks, in the order the summary counts them.
	checks : List(Str)
	checks = ["pure_mutator", "abbreviation", "module_doc"]

	## By path, then line, then what is said, each path and saying bytewise.
	order : Finding, Finding -> [Before, Same, After]
	order = |a, b|
		match Text.byte_order(a.path, b.path) {
			Same =>
				if a.line < b.line {
					Before
				} else if a.line > b.line {
					After
				} else {
					Text.byte_order(a.what, b.what)
				}
			other => other
		}

						$going = Bool.False

					} else {
						$k = $k + 1
					}
				(Err(_), Ok(_)) => {
					$answer = Before
					$going = Bool.False
				}
				(Ok(_), Err(_)) => {
					$answer = After
					$going = Bool.False
				}
				(Err(_), Err(_)) => {

				}
			}
		}

	}

## starts.

byte_order = |first, second| {
	pairs = List.map2(
		first.to_utf8(),
		second.to_utf8(),
		|x, y| (x, y),
	)
	match pairs.find_first(|(x, y)| x != y) {
		Ok((x, y)) =>
			if x < y {
				Before
			} else {
				After
			}
		Err(_) =>
			if first.count_utf8_bytes() < second.count_utf8_bytes() {
				Before
			} else if first.count_utf8_bytes() > second.count_utf8_bytes() {
				After
			} else {
				Same
			}
	}
}

## Bytewise order, as Shale's `sort` puts strings.
byte_order : Str, Str -> [Before, Same, After]
byte_order = |a, b| {
	x = Str.to_utf8(a)
	y = Str.to_utf8(b)
	var $k = 0
	var $answer = Same
	var $going = Bool.True
	while $going {
		match (List.get(x, $k), List.get(y, $k)) {
			(Ok(p), Ok(q)) =>
				if p < q {
					$answer = Before
					$going = Bool.False
				} else if p > q {
					$answer = After
					$going = Bool.False
				} else {
					$k = $k + 1
				}
			(Err(_), Ok(_)) => {
				$answer = Before
				$going = Bool.False
			}
			(Ok(_), Err(_)) => {
				$answer = After
				$going = Bool.False
			}
			(Err(_), Err(_)) => {
				$going = Bool.False
			}
		}
	}
	$answer
}

-- This recipe holds each Roc module to a size: 500 lines is our most, the
-- whole file counted, its expects, blank lines and comments included.
-- Someone composing a program from ours should be able to take a module
-- whole, and a module big enough that they would take half of it is the
-- smell. A small module may keep its expects inside it, one-stop shopping;
-- a module with extensive tests moves them to a `*Spec.roc` beside it.
--
--   cd <repo> && shale <shale>/cookbook/roc/check_module_size.shale glob='app/**/*.roc'
--   cd <repo> && shale <shale>/cookbook/roc/check_module_size.shale tests='*Spec.roc,gates/'
--   cd <repo> && shale <shale>/cookbook/roc/check_module_size.shale exempt=tools/module_size.txt all=1
--   cd <shale> && shale cookbook:roc/check_module_size.shale self_test=1
--
-- The arguments:
--   glob       the files to read, several separated by commas, `**/*.roc` by
--              default; a path holding `zig-out/` or `node_modules/` is left out;
--   tests      the project's test roots, separated by commas, `*Spec.roc,test/`
--              by default: an entry ending in `/` is a folder anywhere in a
--              path, `*X` a file name ending in `X`, anything else a whole
--              file name;
--   exempt     the project's exemptions file, below;
--   limit      the most lines a module may have, 500 by default;
--   all        with `all=1`, each module's count too;
--   self_test  with `self_test=1`, check the plants in
--              `cookbook/roc/plants/check_module_size/` (run from the shale
--              checkout's root).
--
-- Tests have no limit (Steve, 2026-11-19), so a file of a test root is
-- left out of the check by rule, not by an exemption line: a Spec is where a
-- module's extensive tests go once they would crowd it, and what someone
-- takes whole is the module, not its tests. A module's own expects count,
-- since they are part of what is taken.
--
-- The exemptions file names the modules that may be bigger, one `path --
-- reason` a line, the path as this recipe prints it (from the root the run
-- starts at). A blank line or one starting with `#` is skipped. A line with
-- no reason, a path that isn't there, or a path given twice stops the run,
-- and an exempt module now within the limit is a finding. Exemptions are
-- rare: a module that properly shows the workhorse pattern.
--
-- The findings, between two copies of the summary (cookbook/findings.shale):
--   path: module_size: N lines (limit L)
--   path: unneeded_exemption: N lines (limit L) needs no exemption
-- The run fails on any finding.
-- budget: steps=50000000 reads=100000

const SELF_TEST = arg("self_test", "") == "1"

function comma_items(s)
  local out = {}
  for x in split(s, ",") do
    if trim(x) ~= "" then push(out, trim(x)) end
  end
  return out
end

-- Whether `path` is in one of the test roots `roots`, as the header says.
function in_test_root(path, roots)
  local base = basename(path)
  for r in roots do
    if ends_with(r, "/") then
      if starts_with(path, r) or contains(path, "/" .. r) or contains(path, ":" .. r) then return true end
    elseif starts_with(r, "*") then
      if ends_with(base, sub(r, 2)) then return true end
    elseif base == r then
      return true
    end
  end
  return false
end

-- The exemptions by path, each `{path, reason, line}`.
function exemptions_of(file)
  local by_path = {}
  if file == "" then return by_path end
  if not exists(file) then fail("check_module_size: no exemptions file " .. file) end
  local number = 0
  for raw in lines(read(file)) do
    number = number + 1
    local l = trim(raw)
    if l ~= "" and not starts_with(l, "#") then
      local at = find(l, " -- ")
      local where = file .. ":" .. number .. ": "
      if at == nil or trim(sub(l, at + 4)) == "" then fail("check_module_size: " .. where .. "expected `path -- reason`, got `" .. l .. "`") end
      local path = trim(sub(l, 1, at - 1))
      if not exists(path) then fail("check_module_size: " .. where .. "`" .. path .. "` is no file") end
      if has_key(by_path, path) then fail("check_module_size: " .. where .. "`" .. path .. "` is exempted twice") end
      by_path[path] = {path = path, reason = trim(sub(l, at + 4)), line = number}
    end
  end
  return by_path
end

-- The paths `globs` names, each once, sorted, build output left out.
function module_paths(globs)
  local seen = {}
  for g in comma_items(globs) do
    for p in glob(g) do
      if not contains(p, "zig-out/") and not contains(p, "node_modules/") then seen[p] = true end
    end
  end
  local out = keys(seen)
  sort(out)
  return out
end

-- The whole check, as `{summary, found, listing}`.
function module_sizes(opts)
  local exemptions = exemptions_of(opts.exempt)
  local roots = comma_items(opts.tests)
  local limit = opts.limit
  local found = {}
  local listing = {}
  local counts = {modules = 0, tests = 0, over = 0, exempt = 0, unneeded = 0}
  for path in module_paths(opts.glob) do
    local n = count_lines(path)
    if in_test_root(path, roots) then
      counts.tests = counts.tests + 1
      push(listing, path .. ": " .. n .. " lines, a test root's, no limit")
    else
      counts.modules = counts.modules + 1
      local exemption = get(exemptions, path)
      local note = ""
      if exemption ~= nil then
        counts.exempt = counts.exempt + 1
        note = " (exempt: " .. exemption.reason .. ")"
        if n <= limit then
          counts.unneeded = counts.unneeded + 1
          push(found, path .. ": unneeded_exemption: " .. n .. " lines (limit " .. limit .. ") needs no exemption")
        end
      elseif n > limit then
        counts.over = counts.over + 1
        push(found, path .. ": module_size: " .. n .. " lines (limit " .. limit .. ")")
      end
      push(listing, path .. ": " .. n .. " lines" .. note)
    end
  end
  local summary = "check_module_size (Roc): " .. counts.modules .. " modules, " .. counts.tests .. " test files left out; module_size " .. counts.over .. " over " .. limit .. " lines, " .. counts.exempt .. " exempt, unneeded_exemption " .. counts.unneeded
  return {summary = summary, found = found, listing = listing}
end

-- One self-test row: `{name, opts, found, says, stops}`. `found` is how many
-- findings it must give; `says` literals its output must hold, or its stop
-- with `stops`.
function self_test_row(row)
  local opts = {glob = PLANTS .. "*.roc", tests = "*Spec.roc,test/", exempt = "", limit = 20}
  for k in keys(get(row, "opts", {})) do opts[k] = row.opts[k] end
  local r = attempt(module_sizes, opts)
  local faults = {}
  local said = ""
  if get(row, "stops", false) then
    if r.ok then push(faults, "it didn't stop") else said = r.error end
  elseif not r.ok then
    push(faults, "it stopped: " .. r.error)
  else
    said = r.value.summary .. "\n" .. join(r.value.found, "\n") .. "\n" .. join(r.value.listing, "\n")
    if #r.value.found ~= get(row, "found", 0) then push(faults, #r.value.found .. " findings, not " .. get(row, "found", 0)) end
  end
  for literal in get(row, "says", {}) do
    if not contains(said, literal) then push(faults, "it lacks `" .. literal .. "`") end
  end
  print((#faults == 0 and "ok   " or "FAIL ") .. row.name .. (#faults == 0 and "" or ": " .. join(faults, "; ")))
  if #faults > 0 then print(said) end
  return #faults
end

  local at = PLANTS

## from its tree in file order: the types, each with its shape, and the
## values' names. It keeps what check_families reads of
## cookbook/roc/declared.shale's `declarations_of`;
## cookbook/roc/find_unclear_names/Declarations.roc reads more of it for that
## recipe.
import Tree

Declarations := [].{

	## A type declaration: its name, the type it is a member of (`""` at the
	## top level), its line, and the shape of the type it is built on (`""`
	## when it is built on none, as a module's `Cat := [].{ ... }` is not).
	TypeDeclaration : {
		name : Str,
		owner : Str,
		line : U64,
		shape : Str,
	}

	## A value declared by an annotation or a definition, and the type it is a
	## member of.
	ValueName : { name : Str, owner : Str }

	## What one file declares, in file order. A value annotated and defined is
	## there twice.
	FileDeclarations : {
		types : List(Declarations.TypeDeclaration),
		values : List(Declarations.ValueName),
	}

	## Every type and value the file declares at its top level or as a module
	## type's member.
	from_tree : List(Tree.Node) -> Declarations.FileDeclarations
	from_tree = |nodes| {
		var $outer_answers = []
		var $types = []
		var $values = []
		var $at = 0
		for node in nodes {
			$at = $at + 1
			place = { outer_answers: $outer_answers, parent: node.parent }
			is_outer = node.parent == 0 or is_outer_parent(nodes, place)
			$outer_answers = $outer_answers.append(is_outer)
			if is_outer {
				$types = $types.concat(type_declared(nodes, $at))
				$values = $values.concat(value_declared(nodes, $at))
			}
		}

## Whether a node's `parent` lets it be outer: the parent is outer itself, as
## `outer_answers` holds it so far (a parent comes before its kids), and of a
## kind a definition may sit in and still be top-level or a module type's
## member, the type's `associated` block among them. Inside any other kind,
## such as a lambda's block, a definition is local.
is_outer_parent : List(Tree.Node), { outer_answers : List(Bool), parent : U64 } -> Bool
is_outer_parent = |nodes, place| {
	parent_is_outer = match place.outer_answers.get(place.parent - 1) {
		Ok(answer) => answer
		Err(_) => crash "check_families: node ${place.parent.to_str()} comes after its kid"
	}
	kind = Tree.node_at(nodes, place.parent).kind
	parent_is_outer and ["file", "statements", "s-type-decl", "associated"].contains(kind)
}

## The name of the type declaration around node `at`, or "" at the top level.
owner_of : List(Tree.Node), U64 -> Str
owner_of = |nodes, at| {
	var $up = Tree.node_at(nodes, at).parent
	while $up != 0 {
		node = Tree.node_at(nodes, $up)
		if node.kind == "s-type-decl" {
			return Tree.attribute(Tree.node_at(nodes, Tree.first_kid(node)), "name")
		}
		$up = node.parent
	}
	""
}

## The type declared at `at`, when it is a type declaration with a name.
type_declared : List(Tree.Node), U64 -> List(Declarations.TypeDeclaration)
type_declared = |nodes, at| {
	node = Tree.node_at(nodes, at)
	if node.kind != "s-type-decl" {
		return []
	}
	name = Tree.attribute(Tree.node_at(nodes, Tree.first_kid(node)), "name")
	if name == "" {
		return []
	}
	shape = match node.kids.get(1) {
		Ok(built_on) =>
			if Tree.node_at(nodes, built_on).kind == "associated" {
				""
			} else {
				Tree.shape(nodes, built_on)
			}
		Err(_) => ""
	}
	[
		{
			name,
			owner: owner_of(nodes, at),
			line: node.line,
			shape,
		},
	]
}

	first_kind = Tree.first_kid_kind(nodes, node)
	if node.kind == "s-type-anno" and first_kind != "" and first_kind != "where" {
		[{ name: Tree.attribute(node, "name"), owner: owner_of(nodes, at) }]
	} else if node.kind == "s-decl" and first_kind == "p-ident" {
		name = Tree.attribute(Tree.node_at(nodes, Tree.first_kid(node)), "raw")
		[{ name, owner: owner_of(nodes, at) }]
	} else {
		[]
	}

	## level, with the names of that type's members, and the text its doc
	## gives (the file's opening comment and the comment above its type).

-- This recipe counts how much a Zig tree's comments say, and how loudly. For
-- each file it gives its code lines, its doc lines (`///`, `//!`), its plain
-- `//` lines, comment lines per 100 lines of code, words a comment line, and
-- the loud and dated comment lines. Run before and after a commit that
-- rewrites comments, it shows what the commit changed.
--
--   cd <repo> && shale <shale>/cookbook/zig/count_comments.shale
--   shale <shale>/cookbook/zig/count_comments.shale dir=zig:lib/std/zig files=AstGen.zig,Parse.zig,Zir.zig
--   cd <repo> && shale <shale>/cookbook/zig/count_comments.shale top=10
--
-- `files` is comma-separated globs under `dir` (default `.`, the root;
-- a mount works): default `src/**/*.zig,tools/*.zig,build.zig`. A glob's
-- `*` on the command line is the shell's unless quoted.
--
-- A line is code when it holds anything but a comment, a comment line
-- when it is only one (a comment after code counts as code). A loud line
-- holds `**`, a heading of two upper-case words or more (`THE HUNT`), or a
-- `──` banner. A dated line names a QUEUE or
-- FEEDBACK item, a date (`2026-10-20`), or a person (`Steve`, `the box`,
-- `CC`). The table is the loudest first, `top` rows of it (default all),
-- then the whole's totals.
--
-- budget: steps=900000000 reads=100000 read_bytes=400000000 arena_bytes=2000000000

const DIR = arg("dir", ".")
const FILES = arg("files", "src/**/*.zig,tools/*.zig,build.zig")
const TOP = arg("top", 0)

function is_loud(c)
  return contains(c, "**") or contains(c, "──") or re_match(c, "\\b[A-Z][A-Z]+ [A-Z][A-Z]+")
end

function is_dated(c)
  return re_match(c, "QUEUE|FEEDBACK|\\b20\\d\\d-\\d\\d-\\d\\d\\b|\\bSteve\\b|\\bthe box\\b|\\bCC\\b")
end

-- The files `FILES` names under `DIR`, each once, in the order named.
local paths = {}
local seen = {}
for pattern in split(FILES, ",") do
  local p = trim(pattern)
  local full = p
  if DIR ~= "." then full = DIR .. "/" .. p end
  local got = {}
  if contains(p, "*") then got = glob(full) elseif exists(full) then got = {full} end
  for g in got do
    if not get(seen, g, false) then
      seen[g] = true
      push(paths, g)
    end
  end
end
if #paths == 0 then fail("count_comments: no file matches " .. FILES .. " under " .. DIR) end

local rows = {}
local all = {code = 0, doc = 0, plain = 0, words = 0, loud = 0, dated = 0}
for p in paths do
  local r = {code = 0, doc = 0, plain = 0, words = 0, loud = 0, dated = 0}
  for l in lines(read(p)) do
    local t = trim(l)
    if t ~= "" then
      if starts_with(t, "//") then
        if starts_with(t, "///") or starts_with(t, "//!") then r.doc = r.doc + 1 else r.plain = r.plain + 1 end
        r.words = r.words + #words(sub(t, 4))
        if is_loud(t) then r.loud = r.loud + 1 end
        if is_dated(t) then r.dated = r.dated + 1 end
      else
        r.code = r.code + 1
      end
    end
  end
  for k in keys(all) do all[k] = all[k] + r[k] end
  local shown = p
  if DIR ~= "." and starts_with(p, DIR .. "/") then shown = sub(p, #DIR + 2) end
  push(rows, {file = shown, code = r.code, doc = r.doc, plain = r.plain, loud = r.loud, dated = r.dated, r = r})
end

-- A share as `n.d`, of `part` per 100 of `whole`.
function per100(part, whole)
  if whole == 0 then return "-" end
  local tenths = (part * 1000 + whole // 2) // whole
  return (tenths // 10) .. "." .. (tenths % 10)
end

for row in rows do
  local comments = row.r.doc + row.r.plain
  row.per100 = per100(comments, row.r.code)
  row.words = per100(row.r.words, comments * 100)
end
local sorted = sort(rows, "loud", "desc")
if TOP > 0 then sorted = take(sorted, TOP) end
print(table(sorted, {"file", "code", "doc", "plain", "per100", "words", "loud", "dated"}))
local comments = all.doc + all.plain
print(#paths .. " files: " .. all.code .. " code lines, " .. comments .. " comment lines (" .. all.doc .. " doc, " .. all.plain ..
  " plain), " .. per100(comments, all.code) .. " per 100, " .. per100(all.words, comments * 100) .. " words a line, " ..
  all.loud .. " loud, " .. all.dated .. " dated")
