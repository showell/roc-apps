## A line's bytes read as Shale's builtins read a string: a place counted
## from 1, cut as `sub` cuts, trimmed as `trim` trims, and a width that is
## the bytes with a tab counting four. A recipe that scans Roc a byte at a
## time (Groups, Stacking) reads its lines through these, so its places and
## widths are its Shale original's.
Bytes := [].{

	## A line's width: its bytes, a tab counting four.
	width_of : List(U8) -> U64
	width_of = |bytes| bytes.len() + 3 * bytes.count_if(|byte| byte == '\t')

	## The width of a line given as text.
	line_width : Str -> U64
	line_width = |text| Bytes.width_of(text.to_utf8())

	## Line `number` of the lines, counted from 1, as bytes: empty past the end.
	line_of : List(Str), U64 -> List(U8)
	line_of = |source_lines, number|
		if number == 0 {
			[]
		} else {
			(source_lines.get(number - 1) ?? "").to_utf8()
		}

	## The byte at `place`, counted from 1, or 0 past either end.
	byte_at : List(U8), U64 -> U8
	byte_at = |bytes, place|
		if place == 0 {
			0
		} else {
			bytes.get(place - 1) ?? 0
		}

	## Shale's `sub(s, first, last)` of bytes: the places from `first` to
	## `last`, counted from 1, cut to the bytes there are.
	span : List(U8), U64, U64 -> List(U8)
	span = |bytes, first, last| {
		start = if first == 0 {
			1
		} else {
			first
		}
		end = if last < bytes.len() {
			last
		} else {
			bytes.len()
		}
		if start > end {
			[]
		} else {
			bytes.sublist({ start: start - 1, len: end - start + 1 })
		}
	}

	## Shale's `sub(s, first)`: the bytes from place `first` to the end.
	rest_from : List(U8), U64 -> List(U8)
	rest_from = |bytes, first| Bytes.span(bytes, first, bytes.len())

	## The bytes without the spaces, tabs, carriage returns and newlines at
	## their ends, as Shale's `trim` takes them.
	trimmed : List(U8) -> List(U8)
	trimmed = |bytes|
		match bytes.find_first_index(|byte| !(is_blank(byte))) {
			Ok(start) => {
				end = bytes.find_last_index(|byte| !(is_blank(byte))) ?? start
				bytes.sublist({ start, len: end + 1 - start })
			}
			Err(_) => []
		}

	## The bytes without the blanks at their end, as Shale's `trim_end`.
	trimmed_end : List(U8) -> List(U8)
	trimmed_end = |bytes|
		match bytes.find_last_index(|byte| !(is_blank(byte))) {
			Ok(end) => bytes.take_first(end + 1)
			Err(_) => []
		}

	## Whether `byte` is one of `[A-Za-z0-9_]`.
	is_word_byte : U8 -> Bool
	is_word_byte = |byte| Bytes.is_letter(byte) or Bytes.is_digit(byte) or byte == '_'

	## Whether `byte` is an ASCII letter, small or capital.
	is_letter : U8 -> Bool
	is_letter = |byte| (byte >= 'a' and byte <= 'z') or (byte >= 'A' and byte <= 'Z')

	## Whether `byte` is a digit.
	is_digit : U8 -> Bool
	is_digit = |byte| byte >= '0' and byte <= '9'

	## Whether `byte` is a small letter or `_`, which starts a lower name.
	starts_lower_name : U8 -> Bool
	starts_lower_name = |byte| (byte >= 'a' and byte <= 'z') or byte == '_'

	## The last byte before `place` that is not a space, or 0.
	byte_before : List(U8), U64 -> U8
	byte_before = |bytes, place| {
		var $at = place - 1
		while $at > 0 and Bytes.byte_at(bytes, $at) == ' ' {
			$at = $at - 1
		}
		Bytes.byte_at(bytes, $at)
	}

	## The first place at `place` or after it that is no space: the end of ` *`.
	spaces_end : List(U8), U64 -> U64
	spaces_end = |bytes, place| {
		var $end = place
		while Bytes.byte_at(bytes, $end) == ' ' {
			$end = $end + 1
		}
		$end
	}

	## The first place at `place` or after it that is no `[A-Za-z0-9_]`.
	word_end : List(U8), U64 -> U64
	word_end = |bytes, place| {
		var $end = place
		while Bytes.is_word_byte(Bytes.byte_at(bytes, $end)) {
			$end = $end + 1
		}
		$end
	}
}

is_blank = |byte| byte == ' ' or byte == '\t' or byte == '\r' or byte == '\n'

## types and annotations, never a local's. Each carries `place`, its node's

	types : List(Declarations.TypeDeclared),
	values : List(Declarations.ValueDeclared),

	## and `shape` the same as a shape (Tree.shape), both `""` when it is
	## built on none, as a module's `Cat := [].{ ... }` is not; `is_nominal`

	## `result_type` what it gives when it is a function, else the same, and
	## `first_type` its first argument's type, `""` when it takes none.

		path : Str,

		line : U64,
		name : Str,
		owner : Str,
		written_type : Str,

	## place) and its text cut at each newline.

		path : Str,
		nodes : List(Tree.Node),
		source_lines : List(Str),

	## What `earlier` holds, then what `later` does.
	concat : Declarations, Declarations -> Declarations
	concat = |earlier, later| {
		types: earlier.types.concat(later.types),
		values: earlier.values.concat(later.values),
		fields: earlier.fields.concat(later.fields),
	}

	## What the file declares at its top level or as a module type's member,
	## each kind in file order. A definition of an annotated value sets the
	## annotation's `has_text`, and the last definition says.
	from_file : Declarations.FileRead -> Declarations
	from_file = |file| {
		nodes = file.nodes
		var $outer_answers = []
		var $declaration_lines = []
		var $annotation_places = Dict.empty()
		var $declared = Declarations.none_yet
		var $at = 0
		for node in nodes {
			$at = $at + 1
			at = $at
			place = { outer_answers: $outer_answers, parent: node.parent }
			is_outer = node.parent == 0 or is_outer_parent(nodes, place)
			$outer_answers = $outer_answers.append(is_outer)
			kind = node.kind
			is_declaration = kind == "s-type-anno" or kind == "s-type-decl"
			$declaration_lines = $declaration_lines.append(
				if is_outer and is_declaration {
					node.line
				} else {
					0
				},
			)
			if is_outer and kind == "s-type-decl" {
				match type_declared(file, at) {
					Ok(declared) => {
						$declared = { ..$declared, types: $declared.types.append(declared) }
					}
					Err(Unnamed) => {}
				}
			}
			field_place = { declaration_lines: $declaration_lines, at }
			if kind == "anno-record-field" and field_declaration_line(nodes, field_place) != 0 {
				field = {
					path: file.path,
					place: at,
					line: node.line,
					name: Tree.attribute(node, "name"),
					owner: owner_of(nodes, at),
					written_type: Tree.written_type(nodes, Tree.first_kid(node)),
				}
				$declared = { ..$declared, fields: $declared.fields.append(field) }
			}
			if is_outer and kind == "s-type-anno" {
				match annotated_value(file, at) {
					Ok(value) => {
						value_key = "${value.owner}.${value.name}"
						value_place = $declared.values.len()
						$annotation_places = $annotation_places.insert(value_key, value_place)
						$declared = { ..$declared, values: $declared.values.append(value) }
					}
					Err(OnlyWhere) => {}
				}
			}
			if is_outer and kind == "s-decl" and Tree.kid_kind(nodes, node, 1) == "p-ident" {
				name = Tree.attribute(Tree.node_at(nodes, Tree.first_kid(node)), "raw")
				owner = owner_of(nodes, at)
				values = match $annotation_places.get("${owner}.${name}") {
					Ok(annotation_place) => {
						definition = { place: annotation_place, has_text: is_text(nodes, node) }
						with_text($declared.values, definition)
					}
					Err(_) => {
						value = {
							path: file.path,
							place: at,
							line: node.line,
							name,
							owner,
							annotated: Bool.False,
							is_function: Tree.kid_kind(nodes, node, 2) == "e-lambda",
							has_text: Bool.False,
							written_type: "",
							result_type: "",
							first_type: "",
						}
						$declared.values.append(value)
					}
				}
				$declared = { ..$declared, values }
			}
		}
		$declared
	}
}

## Whether a node's `parent` lets it be outer: the parent is outer itself, as
## `outer_answers` holds it so far (a parent comes before its kids), and of a
## kind a definition may sit in and still be top-level or a module type's
## member, the type's `associated` block among them. Inside any other kind,
## such as a lambda's block, a definition is local.
is_outer_parent : List(Tree.Node), { outer_answers : List(Bool), parent : U64 } -> Bool
is_outer_parent = |nodes, place| {
	parent_is_outer = match place.outer_answers.get(place.parent - 1) {
		Ok(answer) => answer
		Err(_) => {
			parent = place.parent.to_str()
			crash "Declarations: node ${parent} comes after its kid"
		}
	}
	kind = Tree.node_at(nodes, place.parent).kind
	parent_is_outer and ["file", "statements", "s-type-decl", "associated"].contains(kind)
}

## The name of the type declaration around node `at`, or "" at the top level.
owner_of : List(Tree.Node), U64 -> Str
owner_of = |nodes, at| {
	var $up = Tree.node_at(nodes, at).parent
	while $up != 0 {
		node = Tree.node_at(nodes, $up)
		if node.kind == "s-type-decl" {
			return Tree.attribute(Tree.node_at(nodes, Tree.first_kid(node)), "name")
		}
		$up = node.parent
	}
	""
}

## The line of the declaration the record field at `at` belongs to, as
## `declaration_lines` holds it so far, or 0 when an expression, a pattern or
## a definition lies between them (the field is then a local's), or the
## declaration is not outer.
field_declaration_line : List(Tree.Node), { declaration_lines : List(U64), at : U64 } -> U64
field_declaration_line = |nodes, field_place| {
	var $up = Tree.node_at(nodes, field_place.at).parent
	while $up != 0 {
		node = Tree.node_at(nodes, $up)
		if node.kind.starts_with("e-") or node.kind.starts_with("p-") or node.kind == "s-decl" {
			return 0
		}
		if node.kind == "s-type-anno" or node.kind == "s-type-decl" {
			return match field_place.declaration_lines.get($up - 1) {
				Ok(line) => line
				Err(_) => crash "Declarations: a field's declaration comes after it"
			}
		}
		$up = node.parent
	}
	0
}

## Whether the definition `node` gives a string with something in it: `""`
## spans two columns.

is_text = |nodes, node|
	match node.kids.get(1) {
		Ok(at) => {
			given = Tree.node_at(nodes, at)
			is_wide = given.end_line > given.line or given.end_column > given.column + 2
			given.kind == "e-string" and is_wide
		}
		Err(_) => Bool.False
	}

## The values with the annotation at the definition's place given its
## `has_text`.
with_text : List(Declarations.ValueDeclared), Definition -> List(Declarations.ValueDeclared)
with_text = |values, definition| {
	past_end = "Declarations: an annotation's place is past the values"
	value = match values.get(definition.place) {
		Ok(found) => found
		Err(OutOfBounds) => crash past_end
	}
	match values.set(definition.place, { ..value, has_text: definition.has_text }) {
		Ok(changed) => changed
		Err(OutOfBounds) => crash past_end
	}
}

## The annotation at `at`, unless it is only a `where` clause.
annotated_value : Declarations.FileRead, U64 -> Try(Declarations.ValueDeclared, [OnlyWhere])
annotated_value = |file, at| {
	nodes = file.nodes
	node = Tree.node_at(nodes, at)
	match node.kids.first() {
		Err(_) => Err(OnlyWhere)
		Ok(type_at) => {
			type_node = Tree.node_at(nodes, type_at)
			if type_node.kind == "where" {
				return Err(OnlyWhere)
			}
			written_type = Tree.written_type(nodes, type_at)
			is_function = type_node.kind == "ty-fn"
			result_type = if is_function {
				Tree.written_type(nodes, Tree.last_kid(type_node))
			} else {
				written_type
			}
			first_type = if is_function and type_node.kids.len() > 1 {
				Tree.written_type(nodes, Tree.first_kid(type_node))
			} else {
				""
			}
			Ok({
				path: file.path,
				place: at,
				line: node.line,
				name: Tree.attribute(node, "name"),
				owner: owner_of(nodes, at),
				annotated: Bool.True,
				is_function,
				has_text: Bool.False,
				written_type,
				result_type,
				first_type,
			})
		}
	}
}

## The type declared at `at`, when it has a name. Whether it is nominal is
## read from the text just after its name, where `:=` or `::` stands.
type_declared : Declarations.FileRead, U64 -> Try(Declarations.TypeDeclared, [Unnamed])
type_declared = |file, at| {
	nodes = file.nodes
	node = Tree.node_at(nodes, at)
	name_node = Tree.node_at(nodes, Tree.first_kid(node))
	name = Tree.attribute(name_node, "name")
	if name == "" {
		return Err(Unnamed)
	}
	declared = {
		path: file.path,
		place: at,
		line: node.line,
		name,
		owner: owner_of(nodes, at),
		underlying: "",
		shape: "",
		is_nominal: Bool.False,
		is_tag_union: Bool.False,
		is_function_type: Bool.False,
	}
	match node.kids.get(1) {
		Ok(built_at) => {
			built_on = Tree.node_at(nodes, built_at)
			if built_on.kind == "associated" {
				Ok(declared)
			} else {
				name_line = Text.line_at(file.source_lines, name_node.end_line)
				after_name = Text.bytes_of(name_line, { start: name_node.end_column - 1, count: 4 })
				Ok({
					..declared,
					underlying: Tree.written_type(nodes, built_at),
					shape: Tree.shape(nodes, built_at),
					is_nominal: after_name.contains(":=") or after_name.contains("::"),
					is_tag_union: built_on.kind == "ty-tag-union",
					is_function_type: built_on.kind == "ty-fn",
				})
			}
		}
		Err(_) => Ok(declared)
	}
}

## A finding of one of a recipe's checks: the file, the line, the check and
## what is wrong there, ordered as the Shale originals sorted their reports
## and printed as their lines.
import Text

Finding := {
	path : Str,
	line : U64,
	check : Str,
	what : Str,
}.{

	## By path, then line, then what is said, each path and saying bytewise.
	order : Finding, Finding -> [Before, Same, After]
	order = |first, second|
		match Text.byte_order(first.path, second.path) {
			Same =>
				if first.line < second.line {
					Before
				} else if first.line > second.line {
					After
				} else {
					Text.byte_order(first.what, second.what)
				}
			other => other
		}

	## The finding as its line: `path:line: check: what`.
	line_of : Finding -> Str
	line_of = |finding| {
		place = "${finding.path}:${finding.line.to_str()}"
		"${place}: ${finding.check}: ${finding.what}"
	}

	## How many of the findings each check made, in the order of `checks`, as
	## a summary counts them: `family 2, twin_name 0`.
	counts_of : List(Finding), List(Str) -> Str
	counts_of = |findings, checks| {
		count_of = |check| findings.count_if(|finding| finding.check == check).to_str()
		Str.join_with(checks.map(|check| "${check} ${count_of(check)}"), ", ")
	}
}

## Every parsed file's tree joined into one list of nodes, as the Shale
## originals of check_how_failure_is_reported and find_functions_and_parameters_that_dont_earn_their_keep join them: each node's
## `parent` and `kids` are places in the joined list, counted from 1, a
## file's root's parent 0, and each node keeps its file's path. Nodes are in source order, each before its kids, so a
## subtree runs from its root to its last descendant, and a node with no line
## of its own takes the next line in the list, across a file's end.
import Tree

Forest := {
	nodes : List(Tree.Node),
	paths : List(Str),
}.{

	## The files' trees joined, in the files' order.
	join : List({ path : Str, nodes : List(Tree.Node) }) -> Forest
	join = |files| {
		var $nodes = []
		var $paths = []
		for file in files {
			offset = $nodes.len()
			for file_node in file.nodes {
				moved_parent = if file_node.parent > 0 {
					file_node.parent + offset
				} else {
					0
				}
				moved = { ..file_node, kids: file_node.kids.map(|kid_at| kid_at + offset), parent: moved_parent }
				$nodes = $nodes.append(moved)
				$paths = $paths.append(file.path)
			}
		}
		{ nodes: $nodes, paths: $paths }
	}

	## How many nodes there are.
	count : Forest -> U64
	count = |forest| forest.nodes.len()

	## The node at a place.
	node : Forest, U64 -> Tree.Node
	node = |forest, at| Tree.node_at(forest.nodes, at)

	## The path of the file the node at a place is in.
	path : Forest, U64 -> Str
	path = |forest, at|
		match forest.paths.get(at - 1) {
			Ok(found) => found
			Err(_) => crash "Forest.path: no node ${at.to_str()}"
		}

	kind : Forest, U64 -> Str
	kind = |forest, at| Forest.node(forest, at).kind

	kids : Forest, U64 -> List(U64)
	kids = |forest, at| Forest.node(forest, at).kids

	parent : Forest, U64 -> U64
	parent = |forest, at| Forest.node(forest, at).parent

	## A node's attribute, `""` where it has none.
	attribute : Forest, U64, Str -> Str
	attribute = |forest, at, attribute_name| Tree.attribute(Forest.node(forest, at), attribute_name)

	name : Forest, U64 -> Str
	name = |forest, at| Forest.attribute(forest, at, "name")

	raw : Forest, U64 -> Str
	raw = |forest, at| Forest.attribute(forest, at, "raw")

	## The place of a node's kid, the kids counted from 1, 0 when it has no
	## such kid.
	kid : Forest, U64, U64 -> U64
	kid = |forest, at, number| Forest.kids(forest, at).get(number - 1) ?? 0

	## The kind of the node at a place, `""` for place 0, no node.
	kind_or_none : Forest, U64 -> Str
	kind_or_none = |forest, at|
		if at == 0 {
			""
		} else {
			Forest.kind(forest, at)
		}

	## The last node of the subtree at a place.
	last_descendant : Forest, U64 -> U64
	last_descendant = |forest, at| {
		var $last = at
		while !(Forest.kids(forest, $last).is_empty()) {
			$last = Forest.kids(forest, $last).last() ?? $last
		}
		$last
	}

	## The line a node is reported at: its own, else the next node's that has
	## one, 0 when no node after it has one (a finding prints that as `?`).
	line_number : Forest, U64 -> U64
	line_number = |forest, at| {
		var $place = at
		while $place <= Forest.count(forest) and Forest.node(forest, $place).line == 0 {
			$place = $place + 1
		}
		if $place <= Forest.count(forest) {
			Forest.node(forest, $place).line
		} else {
			0
		}
	}

	## Each tag of the tag union at a place, as a whole node (`Some(U64)`).
	union_tags : Forest, U64 -> List(U64)
	union_tags = |forest, at|
		Forest.kids(forest, at)
			.keep_if(|tags| Forest.kind(forest, tags) == "tags")
			.join_map(|tags| Forest.kids(forest, tags))

	## The name of a tag node: `Some` for `Some(U64)`.
	tag_name : Forest, U64 -> Str
	tag_name = |forest, at|
		if Forest.kind(forest, at) == "ty-apply" {
			Forest.name(forest, Forest.kid(forest, at, 1))
		} else {
			Forest.name(forest, at)
		}
}

## Roc's bracket groups read from a file's lines, for check_and_fix_group_stacking and
## wrap_long_lines_with_trailing_commas (ported from the Shale groups.shale): each pair of
## brackets or a lambda's `|...|`, where it opens and closes, its items counted by the
## commas at its own depth and its tokens. A token is a name (a dotted path
## like `Angle.Bearing` is one), a number, a string or a character;
## punctuation and operators are none. The rule the groups are judged by is
## Stacking's. A place in a line is counted from 1 and in bytes, as Shale's
## `sub` counts, so a line's width is its bytes with a tab counting four.
import Bytes

Groups := [].{

	## and column it closes at, its items, its tokens, whether a comment or a

## The scan after one more line.
scanned_line : Scan, { bytes : List(U8), number : U64 } -> Scan
scanned_line = |before, source| {
	l = source.bytes
	line = source.number
	var $scan = before
	var $count = l.len()
	if Bytes.trimmed(l).starts_with(['\\', '\\']) {
		$scan = {
			..$scan,
			fixed: $scan.fixed + 1,
			tokens: $scan.tokens + 1,
			last_code: '\\',
		}
		$count = 0
	}
	$i : U64
	var $i = 1
	while $i <= $count {
		c = Bytes.byte_at(l, $i)
		var $code = c
		if c == ' ' {
			# A space only separates, and spaces are the most common byte.
			$code = ' '
		} else if Bytes.is_word_byte(c) {
			# A name right after a dot continues a path, as in `f(x).name`.
			continues = Bytes.byte_at(
				l,
				$i - 1,
			) == '.' and !($i >= 2 and Bytes.byte_at(l, $i - 2) == '.')
			if !(continues) {
				$scan = { ..$scan, tokens: $scan.tokens + 1 }
			}
			$i = path_end(l, $i) - 1
		} else if c == '"' or c == '\'' {
			$scan = { ..$scan, tokens: $scan.tokens + 1 }
			$i = $i + 1
			while $i <= $count and Bytes.byte_at(l, $i) != c {
				if Bytes.byte_at(l, $i) == '\\' {
					$i = $i + 1
				}
				$i = $i + 1
			}
		} else if c == '#' {
			$scan = { ..$scan, fixed: $scan.fixed + 1 }
			$code = ' '
			$count = 0
		} else if Groups.closer_of(c) != 0 {
			is_where = c == '[' and follows_where(l, $i)
			$scan = opened(
				$scan,
				{
					line,
					at: $i,
					closer: Groups.closer_of(c),
					is_where,
				},
			)
		} else if (c == '-' or c == '=') and Bytes.byte_at(
			l,
			$i + 1,
		) == '>' and Bytes.byte_at(l, $i - 1) == ' ' and !($scan.open.is_empty()) {
			$scan = with_top($scan, |group| { ..group, has_arrow: Bool.True })
			$i = $i + 1
		} else if c == '|' and Bytes.byte_at(
			l,
			$i + 1,
		) != '>' and Bytes.byte_at(l, $i + 1) != '|' and Bytes.byte_at(l, $i - 1) != '|' {
			if top_closer($scan.open) == '|' {
				$scan = closed($scan, { bytes: l, line, close: $i })
			} else {
				# A `|` opens a lambda only where an expression can start.
				b = Bytes.byte_before(l, $i)
				if b == 0 or b == '=' or b == '(' or b == ',' or b == '[' or b == '{' or b == ':' {
					$scan = opened(
						$scan,
						{
							line,
							at: $i,
							closer: '|',
							is_where: Bool.False,
						},
					)
				}
			}
		} else if Groups.is_closer(c) {
			# Pop down to the group this byte closes. A `|` that turned out to
			# be no lambda is dropped on the way.
			var $open = $scan.open
			while !($open.is_empty()) and top_closer($open) != c {
				$open = $open.drop_last(1)
			}
			$scan = { ..$scan, open: $open }
			if !($open.is_empty()) {
				$scan = closed($scan, { bytes: l, line, close: $i })
			}
		} else if c == ',' and !($scan.open.is_empty()) {
			rest = Bytes.rest_from(l, $i + 1)
			in_where = match $scan.open.last() {
				Ok(group) => group.is_where
				Err(_) => Bool.False
			}
			# In a `where` clause's brackets, only a comma before a method
			# constraint, or at its line's end, separates items.
			if !(in_where) or Bytes.trimmed(rest).is_empty() or starts_constraint(rest) {
				comma_at = $i
				$scan = with_top($scan, |group| with_comma(group, comma_at))
			}
		}
		if $code != ' ' and $code != '\t' {
			$scan = { ..$scan, last_code: $code }
		}
		$i = $i + 1
	}
	$scan
}

## Whether the code before column `place` ends in the keyword `where`, so
## that a `[` there opens a `where` clause.
follows_where : List(U8), U64 -> Bool
follows_where = |bytes, place| {
	var $j = place - 1
	while $j > 0 and (Bytes.byte_at(bytes, $j) == ' ' or Bytes.byte_at(bytes, $j) == '\t') {
		$j = $j - 1
	}
	if $j < 5 or Bytes.span(bytes, $j - 4, $j) != "where".to_utf8() {
		Bool.False
	} else if $j == 5 {
		Bool.True
	} else {
		b = Bytes.byte_at(bytes, $j - 5)
		b == ' ' or (b == '\t' and Bytes.trimmed(Bytes.span(bytes, 1, $j - 5)).is_empty())
	}
}

## Whether the bytes start a method constraint, `k.is_eq :`:
## `^ *[a-z_][A-Za-z0-9_]*\.[a-z_][A-Za-z0-9_]*!? *:`. In a `where` clause's
## brackets, only a comma followed by one, or ending its line, separates
## items.
starts_constraint : List(U8) -> Bool
starts_constraint = |bytes| {
	var $at = 1
	while Bytes.byte_at(bytes, $at) == ' ' {
		$at = $at + 1
	}
	match name_end(bytes, $at) {
		Err(_) => Bool.False
		Ok(after_type) =>
			if Bytes.byte_at(bytes, after_type) != '.' {
				Bool.False
			} else {
				match name_end(bytes, after_type + 1) {
					Err(_) => Bool.False
					Ok(after_method) => {
						var $end = after_method
						if Bytes.byte_at(bytes, $end) == '!' {
							$end = $end + 1
						}
						while Bytes.byte_at(bytes, $end) == ' ' {
							$end = $end + 1
						}
						Bytes.byte_at(bytes, $end) == ':'
					}
				}
			}
	}
}

	first = Bytes.byte_at(bytes, place)
	if (first >= 'a' and first <= 'z') or first == '_' {
		var $end = place + 1
		while Bytes.is_word_byte(Bytes.byte_at(bytes, $end)) {
			$end = $end + 1
		}
		Ok($end)
	} else {
		Err(NoName)
	}
}

## Whether an item is a literal: `-?[0-9][0-9A-Za-z_.]*`, `"[^"\\$]*"`,
## `'[^']+'` or a bare tag `[A-Z][A-Za-z0-9]*`.
is_literal_item : List(U8) -> Bool
is_literal_item = |item| {
	count = item.len()
	first = Bytes.byte_at(item, 1)
	last = Bytes.byte_at(item, count)
	middle = Bytes.span(item, 2, count - 1)
	if number_end(item, 1) == Ok(count + 1) {
		Bool.True
	} else if first == '"' {
		is_kept_out = |byte| byte == '"' or byte == '\\' or byte == '$'
		count >= 2 and last == '"' and !(middle.any(is_kept_out))
	} else if first == '\'' {
		count >= 3 and last == '\'' and !(middle.contains('\''))
	} else {
		is_tag_byte = |byte| Bytes.is_letter(byte) or Bytes.is_digit(byte)

## The record a `t` line holds, `t`, the kind, the tag, the span's four
## numbers and the type, tab-separated, or why it is not one. `number` is the
## line's, named in a refusal.
record_of : Str, U64 -> Try(RocTypes.Record, Str)
record_of = |output_line, number| {
	at = |{}| "line ${number.to_str()}"
	fields = output_line.split_on("\t")
	field = |place| fields.get(place - 1) ?? ""
	kind = field(2)
	if fields.len() != 8 {
		Err("`roc_types`: ${at({})}, a record of ${fields.len().to_str()} fields, not 8: ${output_line}")
	} else if kind != "expr" and kind != "pat" {
		Err("`roc_types`: ${at({})}, a record of kind `${kind}`")
	} else {
		var $span = ""
		for number_text in fields.sublist({ start: 3, len: 4 }) {
			match I64.from_str(number_text) {
				Ok(parsed) => {
					$span = if $span == "" {
						parsed.to_str()
					} else {
						"${$span}:${parsed.to_str()}"
					}
				}
				Err(_) => {
					return Err("`roc_types`: ${at({})}, a span's number `${number_text}`")
				}
			}
		}
		written = if field(8).contains("\\") {
			unescaped(field(8))?
		} else {
			field(8)
		}
		Ok({ kind, tag: field(3), span: $span, written })
	}
}

## A type's text with roc-types' escapes of `\`, tab, newline and carriage
## return undone, or why it holds a bad one.
unescaped : Str -> Try(Str, Str)
unescaped = |text| {
	bytes = text.to_utf8()
	var $out = []
	$at : U64
	var $at = 0
	while $at < bytes.len() {
		byte = bytes.get($at) ?? 0
		if byte == '\\' and $at + 1 < bytes.len() {
			escaped = bytes.get($at + 1) ?? 0
			plain = if escaped == 't' {
				Ok('\t')
			} else if escaped == 'n' {
				Ok('\n')
			} else if escaped == 'r' {
				Ok('\r')
			} else if escaped == '\\' {
				Ok('\\')
			} else {
				Err(BadEscape)
			}
			match plain {
				Ok(kept) => {
					$out = $out.append(kept)
				}
				Err(BadEscape) => {
					shown = Str.from_utf8_lossy([escaped])
					return Err("`roc_types`: a type with a bad escape, `\\${shown}`: ${text}")
				}
			}
			$at = $at + 2
		} else if byte == '\\' {
			return Err("`roc_types`: a type ending in a lone `\\`: ${text}")
		} else {
			$out = $out.append(byte)
			$at = $at + 1
		}
	}
	Ok(Str.from_utf8_lossy($out))
}

## The stacking rule's one home, over Groups' groups (ported from the Shale
## groups.shale, since retired): a group of three or more items and `tokens`
## or more tokens stacks, one item a line, and any other group sits on one
## line while that line fits `width`.
## A function type's arguments can't take a trailing comma, and a row of
## literals or a line of numeric data stays packed. The rewrite's own pieces
## are here too: the group a line's rewrite gives a trailing comma, the
## stacked groups whose comma it takes away, the `targets:` section no
## rewrite reaches, and the columns a line's strings take up, which no
## trailing comma can shorten.

	## items stacks, and the widest a line may be.

	## of its trailing comma.

	## data, or beside it, makes `roc fmt` open every group around that one.

	## wider than `width`, the widest group of two or more items whose

					}

		return at.joined

## width alone: the widest group of two or more items, not a function type's

		indent_end = at.bytes.find_first_index(|byte| !(is_indent(byte))) ?? at.bytes.len()

		Bytes.width_of(indent.concat(Bytes.rest_from(at.bytes, at.group.close))) <= width

	bytes_of = |text, span| {
		bytes = text.to_utf8().sublist({ start: span.start, len: span.count })
		Str.from_utf8_lossy(bytes)
	}

	## ends, as Shale's `trim` takes them.
	trim : Str -> Str
	trim = |text| {
		bytes = text.to_utf8()
		is_kept = |byte| !(byte == ' ' or byte == '\t' or byte == '\r' or byte == '\n')
		match bytes.find_first_index(is_kept) {
			Ok(start) => {
				end = bytes.find_last_index(is_kept) ?? start
				Str.from_utf8_lossy(bytes.sublist({ start, len: end + 1 - start }))
			}
			Err(_) => ""
		}
	}

	lower : Str -> Str
	lower = |text| Str.from_utf8_lossy(text.to_utf8().map(lower_byte))

	is_capital = |byte| byte >= 'A' and byte <= 'Z'

	is_small_letter : U8 -> Bool
	is_small_letter = |byte| byte >= 'a' and byte <= 'z'

	is_digit : U8 -> Bool
	is_digit = |byte| byte >= '0' and byte <= '9'

	before_first = |text, separator|
		match text.split_first(separator) {
			Ok(parts) => parts.before
			Err(_) => text
		}

	after_last : Str, Str -> Str
	after_last = |text, separator|
		match text.split_last(separator) {
			Ok(parts) => parts.after
			Err(_) => text
		}

	## before the next `closing`.
	between : Str, Str, Str -> Try(Str, [NotFound])
	between = |text, opening, closing| {
		after_opening = text.split_first(opening)?
		inside = after_opening.after.split_first(closing)?

	}

	basename : Str -> Str
	basename = |path| Text.after_last(path, "/")

	## it has none.
	dirname : Str -> Str
	dirname = |path|
		match path.split_last("/") {
			Ok(parts) => parts.before
			Err(_) => ""
		}

	## Shale's `path_under`: a name below a folder, spelled as every call takes
	## it: in the root (`""`, `"."`) the name, in a mount's folder (`m:`,
	## `m:.`) `m:name`, else `dir/name`.
	path_under : Str, Str -> Str
	path_under = |folder, name|
		if folder == "" or folder == "." {
			name
		} else if folder.ends_with(":") {
			"${folder}${name}"
		} else if folder.ends_with(":.") {
			"${folder.drop_suffix(".")}${name}"
		} else {
			"${folder}/${name}"
		}

	## A name's words, lowercased: split at its underscores and before each
	## capital that follows a letter or digit, so `TiltStep` and `tilt_step`
	## both give `tilt` and `step` (cookbook/roc/declared.shale's
	## `name_words`).
	name_words : Str -> List(Str)
	name_words = |name| {
		var $words = []
		var $word = []
		for byte in name.to_utf8() {
			if byte == '_' {
				$words = with_word($words, $word)
				$word = []
			} else if Text.is_capital(byte) and !($word.is_empty()) {
				$words = with_word($words, $word)
				$word = [byte]
			} else {
				$word = $word.append(byte)
			}
		}
		with_word($words, $word)
	}

	## A text's words, as Shale's `words` splits it: the runs between spaces,
	## tabs, carriage returns and newlines, none empty.
	words : Str -> List(Str)
	words = |text| {
		var $said = []
		var $word = []
		for byte in text.to_utf8() {
			if byte == ' ' or byte == '\t' or byte == '\r' or byte == '\n' {
				$said = with_raw_word($said, $word)
				$word = []
			} else {
				$word = $word.append(byte)
			}
		}
		with_raw_word($said, $word)
	}

	## The texts put in Shale's `sort` order, bytewise.
	sort_bytewise : List(Str) -> List(Str)
	sort_bytewise = |texts| texts.sort_with(Text.byte_order)

	## Two texts in the order of their bytes, a text before every longer one
	## it starts, as Shale's `sort` puts strings.
	byte_order : Str, Str -> [Before, Same, After]
	byte_order = |first, second| {
		pairs = List.map2(
			first.to_utf8(),
			second.to_utf8(),
			|x, y| (x, y),
		)
		match pairs.find_first(|(x, y)| x != y) {
			Ok((x, y)) =>
				if x < y {
					Before
				} else {
					After
				}
			Err(_) =>
				if first.count_utf8_bytes() < second.count_utf8_bytes() {
					Before
				} else if first.count_utf8_bytes() > second.count_utf8_bytes() {
					After
				} else {
					Same
				}
		}
	}
}

	node = Tree.node_at(nodes, at)
	kind = node.kind
	name = Tree.attribute(node, "name")
	texts_of = |kids| kids.map(|kid| type_text(nodes, kid, style))
	joined = |kids| Str.join_with(texts_of(kids), ", ")
	members_joined = |kids|
		match style {
			AsWritten => joined(kids)
			AsShape => Str.join_with(Text.sort_bytewise(texts_of(kids)), ", ")
		}
	first = |{}| type_text(nodes, Tree.first_kid(node), style)
	if kind == "ty-apply" {
		"${first({})}(${joined(node.kids.drop_first(1))})"
	} else if kind == "ty-fn" {
		result = type_text(nodes, Tree.last_kid(node), style)
		"(${joined(node.kids.drop_last(1))} -> ${result})"
	} else if kind == "ty-record" {
		"{ ${members_joined(node.kids)} }"
	} else if kind == "ty-tuple" {
		"(${joined(node.kids)})"
	} else if kind == "tags" {
		members_joined(node.kids)
	} else if kind == "ty-tag-union" {
		"[${first({})}]"
	} else if kind == "anno-record-field" {

			AsShape => Tree.last_part(name)
		}
